   187→   187→  const [selectedStudent, setSelectedStudent] = useState<Student | null>(null)
   188→   188→  const [certifications, setCertifications] = useState<Certification[]>([])
   189→   189→  const [previewCert, setPreviewCert] = useState<Certification | null>(null)
   190→   190→  const [loading, setLoading] = useState(false)
   191→   191→  const [generating, setGenerating] = useState(false)
   192→   192→  const [loadingData, setLoadingData] = useState(false)
   193→   193→  const [dataLoaded, setDataLoaded] = useState(false)
   194→   194→  const [activeTab, setActiveTab] = useState('calificaciones')
   195→   195→  const { toast } = useToast()
   196→   196→
   197→   197→  const [certData, setCertData] = useState<CertData>(emptyCertData())
   198→   198→  const [loadedData, setLoadedData] = useState<CertData | null>(null)
   199→   199→
   200→   200→  // Plan activo basado en los datos cargados
   201→   201→  const activePlan = getActivePlan(certData.planTipo)
   202→   202→
   203→   203→  // Fetch historial de certificaciones del estudiante
   204→   204→  const fetchCertifications = async (studentId: string) => {
   205→   205→    setLoading(true)
   206→   206→    try {
   207→   207→      const res = await fetch(`/api/certifications/${studentId}`)
   208→   208→      if (!res.ok) { setCertifications([]); return }
   209→   209→      const data = await res.json()
   210→   210→      setCertifications(Array.isArray(data) ? data.filter((c: Certification) => c.tipo === 'CERTIFICACION_CALIFICACIONES') : [])
   211→   211→    } catch { setCertifications([]) }
   212→   212→    finally { setLoading(false) }
   213→   213→  }
   214→   214→
   215→   215→  // Cargar datos de calificaciones del rawData del estudiante
   216→   216→  const fetchCertDataFromDB = async (student: Student) => {
   217→   217→    setLoadingData(true)
   218→   218→    setDataLoaded(false)
   219→   219→    try {
   220→   220→      const res = await fetch(`/api/students/${student.id}/cert-data`)
   221→   221→      if (!res.ok) {
   222→   222→        const errorData = await res.json().catch(() => ({}))
   223→   223→        console.error('cert-data API error:', res.status, errorData)
   224→   224→        const reason = errorData.reason || ''
   225→   225→        const detailMsg = reason === 'empty_rawData'
   226→   226→          ? `El alumno ${student.cedula} no tiene rawData (datos de calificaciones) en la base de datos. Se requiere re-importar los datos.`
   227→   227→          : reason === 'parse_error'
   228→   228→            ? `Error al parsear rawData del alumno ${student.cedula} (longitud: ${errorData.rawDataLength || '?'}).`
   229→   229→            : `No se encontraron datos de calificaciones para ${student.cedula}.`
   230→   230→        toast({ title: 'Sin datos de calificaciones', description: detailMsg, variant: 'destructive' })
   231→   231→        setLoadingData(false)
   232→   232→        return
   233→   233→      }
   234→   234→      const result = await res.json()
   235→   235→
   236→   236→      if (result.certData) {
   237→   237→        // Si el estudiante tiene rawData parseado, usar esos datos
   238→   238→        const cd = result.certData as CertData
   239→   239→        // Si la fecha de expedición está vacía, usar la fecha actual
   240→   240→        if (!cd.fechaExpedicion || cd.fechaExpedicion.trim() === '') {
   241→   241→          cd.fechaExpedicion = new Date().toISOString().split('T')[0]
   242→   242→        }
   243→   243→        // Si la fecha viene en DD/MM/YYYY, convertir a YYYY-MM-DD para el input
   244→   244→        if (/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(cd.fechaExpedicion)) {
   245→   245→          const parts = cd.fechaExpedicion.split('/')
   246→   246→          cd.fechaExpedicion = `${parts[2]}-${parts[1].padStart(2, '0')}-${parts[0].padStart(2, '0')}`
   247→   247→        }
   248→   248→        setCertData(cd)
   249→   249→        setDataLoaded(true)
   250→   250→
   251→   251→        const planLabel = result.certData.planTipo === 'derogado' ? 'Plan Derogado (BD2)' : 'Plan Vigente (BD)'
   252→   252→        const gradeCount = result.gradeCount || Object.values(result.certData.calificaciones || {})
   253→   253→          .flat()
   254→   254→          .filter((c: CalificacionRow) => c.nota && c.nota !== '').length
   255→   255→
   256→   256→        toast({
   257→   257→          title: `Datos cargados — ${planLabel}`,
   258→   258→          description: `Se cargaron ${gradeCount} calificaciones del rawData. Verifica y edita los datos en las pestañas.`,
   259→   259→        })
   260→   260→        setActiveTab('calificaciones')
   261→   261→      }
   262→   262→    } catch (err) {
   263→   263→      console.error('Error fetching cert data:', err)
   264→   264→      toast({ title: 'Error', description: 'Error al cargar datos de calificaciones. Verifique su conexión e intente de nuevo.', variant: 'destructive' })
   265→   265→    } finally {
   266→   266→      setLoadingData(false)
   267→   267→    }
   268→   268→  }
   269→   269→
   270→   270→  const handleSelectStudent = (student: Student) => {
   271→   271→    setSelectedStudent(student)
   272→   272→    setPreviewCert(null)
   273→   273→    setLoadedData(null)
   274→   274→    setDataLoaded(false)
   275→   275→    fetchCertifications(student.id)
   276→   276→
   277→   277→    // Normalizar fecha de nacimiento a DD/MM/YYYY
   278→   278→    const normalizeFecha = (f: string | null | undefined): string => {
   279→   279→      if (!f) return ''
   280→   280→      const trimmed = String(f).trim()
   281→   281→      if (!trimmed) return ''
   282→   282→      if (/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(trimmed)) {
   283→   283→        const parts = trimmed.split('/')
   284→   284→        return `${parts[0].padStart(2, '0')}/${parts[1].padStart(2, '0')}/${parts[2]}`
   285→   285→      }
   286→   286→      if (trimmed.includes('T') || /^\d{4}-\d{2}-\d{2}/.test(trimmed)) {
   287→   287→        try {
   288→   288→          const d = new Date(trimmed)
   289→   289→          if (!isNaN(d.getTime())) {
   290→   290→            const day = String(d.getDate()).padStart(2, '0')
   291→   291→            const month = String(d.getMonth() + 1).padStart(2, '0')
   292→   292→            const year = d.getFullYear()
   293→   293→            return `${day}/${month}/${year}`
   294→   294→          }
   295→   295→        } catch { /* ignore */ }
   296→   296→      }
   297→   297→      return trimmed
   298→   298→    }
   299→   299→
   300→   300→    // Iniciar con datos vacíos pero del plan correcto
   301→   301→    const data = emptyCertData(student.plan === 'derogado' ? 'derogado' : 'vigente')

   304→   304→      fechaNacimiento: normalizeFecha(student.fechaNacimiento),
   305→   305→      apellidos: student.apellidos,
   306→   306→      nombres: student.nombres,
   307→   307→      pais: student.pais || 'VENEZUELA',
   308→   308→      estado: student.estado || '',
   309→   309→      municipio: student.municipio || '',
   310→   310→    }
   311→   311→    data.planTipo = student.plan === 'derogado' ? 'derogado' : 'vigente'
   312→   312→    setCertData(data)
   313→   313→
   314→   314→    // Auto-cargar datos de rawData
   315→   315→    fetchCertDataFromDB(student)
   316→   316→  }
   317→   317→
   318→   318→  const updateNota = (anio: string, matIndex: number, field: keyof CalificacionRow, value: string) => {

   357→   357→
   358→   358→  const handleGenerate = async () => {

   362→   362→      const res = await fetch('/api/certifications', {

   368→   368→      const cert = await res.json()
   369→   369→      setPreviewCert(cert)
   370→   370→      setCertifications(prev => [cert, ...prev])

   372→   372→      toast({ title: 'Certificación de Calificaciones Generada', description: `Numero: ${cert.numero}` })
   373→   373→    } catch {
   374→   374→      toast({ title: 'Error', description: 'Error al generar certificación', variant: 'destructive' })
   375→   375→    } finally { setGenerating(false) }
   376→   376→  }
   377→   377→
   378→   378→  const handleViewCert = (cert: Certification) => {
   379→   379→    try {
   380→   380→      const data = JSON.parse(cert.datos || '{}')
   381→   381→      // Corregir literales viejos: IN→INASISTENTE, PE→PENDIENTE
   382→   382→      if (data.calificaciones) {
   383→   383→        for (const anio of Object.keys(data.calificaciones)) {
   384→   384→          for (const cal of data.calificaciones[anio]) {
   385→   385→            if (cal.nota === 'IN' && cal.literal !== 'INASISTENTE') cal.literal = 'INASISTENTE'
   386→   386→            if (cal.nota === 'PE' && cal.literal !== 'PENDIENTE') cal.literal = 'PENDIENTE'
   387→   387→            // Si el literal está vacío pero la nota tiene valor, regenerar
   388→   388→            if (!cal.literal && cal.nota) cal.literal = notaEnLetras(cal.nota)
   389→   389→          }
   390→   390→        }
   391→   391→      }
   392→   392→      setLoadedData(data)
   393→   393→      setPreviewCert(cert)
   394→   394→      setActiveTab('vista')
   395→   395→    } catch {
   396→   396→      toast({ title: 'Error', description: 'Error al cargar datos de certificación', variant: 'destructive' })
   397→   397→    }
   398→   398→  }
   399→   399→
