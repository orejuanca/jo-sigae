    53→function formatDateLong(dateStr: string): string {
    54→  const m = dateStr.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
    55→  if (!m) return dateStr
    56→  const months = ['enero','febrero','marzo','abril','mayo','junio','julio','agosto','septiembre','octubre','noviembre','diciembre']
    57→  const day = parseInt(m[1])
    58→  const month = months[parseInt(m[2]) - 1] || m[2]
    59→  const year = m[3]
    60→  return `${day} de ${month} de ${year}`
    61→}

    63→function resolveTemplateToken(token: string, data: CertData | null): string {
    64→  if (!data) return token
    65→  const t = token.trim()
    66→  const key = t.replace(/^\{\{|\}\}$/g, '').trim()
    67→  const parts = key.split('.')
    68→  let value = ''
    69→  if (parts[0] === 'estudiante') {
    70→    const obj = data.estudiante as any
    71→    value = parts.slice(1).reduce((o: any, k: string) => o?.[k], obj) || ''
    72→  } else if (parts[0] === 'director') {
    73→    const obj = data.director as any
    74→    value = parts.slice(1).reduce((o: any, k: string) => o?.[k], obj) || ''
    75→  } else if (parts[0] === 'directorCdcce') {
    76→    const obj = data.directorCdcce as any
    77→    value = parts.slice(1).reduce((o: any, k: string) => o?.[k], obj) || ''
    78→  } else {
    79→    const top = (data as any)[key]
    80→    if (top !== undefined && top !== null) value = String(top)
    81→  }
    82→  if (key === 'fechaExpedicion' && value) value = formatDateLong(value)
    83→  return value
    84→}

    86→function processTemplateLine(line: string, data: CertData | null): React.ReactNode[] {
    87→  const parts: React.ReactNode[] = []
    88→  const regex = /\{\{[^}]+\}\}/g
    89→  let lastIndex = 0
    90→  let match: RegExpExecArray | null
    91→  while ((match = regex.exec(line)) !== null) {
    92→    if (match.index > lastIndex) parts.push(line.substring(lastIndex, match.index))
    93→    const resolved = resolveTemplateToken(match[0], data)
    94→    parts.push(<span key={match.index} className="font-semibold">{resolved}</span>)
    95→    lastIndex = regex.lastIndex
    96→  }
    97→  if (lastIndex < line.length) parts.push(line.substring(lastIndex))
    98→  return parts.length > 0 ? parts : [line]
    99→}

   149→// Inline editable line component
   150→function EditableLine({
   151→  value,
   152→  onChange,
   153→  onRemove,
   154→  onAddAfter,
   155→  canRemove,
   156→  className,
   157→  style,
   158→  placeholder,
   159→  isTextArea,
   160→}: {
   161→  value: string
   162→  onChange: (val: string) => void
   163→  onRemove: () => void
   164→  onAddAfter: () => void
   165→  canRemove: boolean
   166→  className?: string
   167→  style?: React.CSSProperties
   168→  placeholder?: string
   169→  isTextArea?: boolean
   170→}) {
   171→  const [editing, setEditing] = useState(false)
   172→  const [localValue, setLocalValue] = useState(value)
   173→  const ref = useRef<HTMLDivElement>(null)
   174→
   175→  useEffect(() => { setLocalValue(value) }, [value])
   176→
   177→  const handleBlur = () => {
   178→    setEditing(false)
   179→    if (localValue !== value) onChange(localValue)
   180→  }
   181→
   182→  const handleKeyDown = (e: React.KeyboardEvent) => {
   183→    if (e.key === 'Enter' && !e.shiftKey) {
   184→      e.preventDefault()
   185→      handleBlur()
   186→    }
   187→  }
   188→
   189→  return (
   190→    <div className="group relative flex items-start gap-1" style={style}>
   191→      {/* Editable content */}
   192→      {editing ? (
   193→        <textarea
   194→          ref={ref as any}
   195→          className="flex-1 border border-blue-400 bg-blue-50/80 rounded px-1 py-0.5 text-[inherit] font-[inherit] text-left outline-none resize-none focus:ring-1 focus:ring-blue-400"
   196→          style={{ fontFamily: 'inherit', fontSize: 'inherit', lineHeight: 'inherit', color: '#000', minHeight: '1.4em' }}
   197→          value={localValue}
   198→          onChange={(e) => setLocalValue(e.target.value)}
   199→          onBlur={handleBlur}
   200→          onKeyDown={handleKeyDown}
   201→          autoFocus
   202→          rows={1}
   203→        />
   204→      ) : (
   205→        <div
   206→          className={`flex-1 cursor-text hover:bg-blue-50/60 rounded px-1 py-0.5 transition-colors min-h-[1.4em] ${className || ''}`}
   207→          style={style}
   208→          onClick={() => setEditing(true)}
   209→        >
   210→          {value || <span className="text-gray-300 italic">{placeholder || 'Doble clic para editar...'}</span>}
   211→        </div>
   212→      )}
   213→      {/* Action buttons - show on hover */}
   214→      <div className="flex gap-0.5 opacity-0 group-hover:opacity-100 transition-opacity shrink-0 mt-px">
   215→        <button
   216→          onClick={onAddAfter}
   217→          className="w-4 h-4 flex items-center justify-center rounded bg-green-100 hover:bg-green-200 text-green-700"
   218→          title="Agregar linea"
   219→        >
   220→          <Plus className="w-2.5 h-2.5" />
   221→        </button>
   222→        {canRemove && (
   223→          <button
   224→            onClick={onRemove}
   225→            className="w-4 h-4 flex items-center justify-center rounded bg-red-100 hover:bg-red-200 text-red-700"
   226→            title="Eliminar linea"
   227→          >
   228→            <Trash2 className="w-2.5 h-2.5" />
   229→          </button>
   230→        )}
   231→      </div>
   232→    </div>
   233→  )
   234→}

   462→          docHtml += `<td style="border:1px solid #333;padding:4px 6px">${g.materia}</td><td style="border:1px solid #333;padding:4px 6px;text-align:center">${g.nota}</td><td style="border:1px solid #333;padding:4px 6px;text-align:center">${g.literal}</td><td style="border:1px solid #333;padding:4px 6px;text-align:center">${g.tipoEvaluacion || ''}</td></tr>`
   463→        }
   464→      }
   465→      docHtml += '</tbody></table>'
   466→    }
   467→
   468→    docHtml += '<div class="signatures">'
   469→    for (const line of template.footerLines) {
   470→      const resolved = line.replace(/\{\{[^}]+\}\}/g, (tok) => resolveTemplateToken(tok, data))
   471→      if (resolved.trim() === '') { docHtml += '<br/>'; continue }
   472→      if (/^_{3,}/.test(resolved.trim())) { docHtml += `<div class="sig-line">${resolved}</div>`; continue }
   473→      if (line.includes('{{director.apellidosNombres}}') && !line.includes('C.I.')) { docHtml += `<div class="sig-name">${resolved}</div>`; continue }
   474→      if (line.includes('C.I. {{director.cedula}}')) { docHtml += `<div class="sig-role">${resolved}</div>`; continue }
   475→      if (line.includes('Secretaria') || line.includes('Director')) { docHtml += `<div class="sig-role">${resolved}</div>`; continue }
   476→      docHtml += `<p class="body-text">${resolved}</p>`
   477→    }
   478→    docHtml += '</div></body></html>'
   479→
   480→    let iframe = document.getElementById('text-editor-print-frame') as HTMLIFrameElement | null
   481→    if (!iframe) {
   482→      iframe = document.createElement('iframe')
   483→      iframe.id = 'text-editor-print-frame'
   484→      iframe.style.cssText = `position:fixed;left:-9999px;top:0;width:${pageWidths[pageSize]};height:${pageHeights[pageSize]};border:none`
   485→      document.body.appendChild(iframe)
   486→    }
   487→    const doc = iframe.contentDocument!
   488→    doc.open(); doc.write(docHtml); doc.close()
   489→    setTimeout(() => { iframe!.contentWindow!.print() }, 300)
   490→  }
   491→
   492→  const previewData = previewCertData || SAMPLE_CERT_DATA
   493→  const yearOrder = ['Primer Año', 'Segundo Año', 'Tercer Año', 'Cuarto Año', 'Quinto Año']
   494→
   495→  if (loading || !template) {
   496→    return (
   497→      <AppShell>
   498→        <div className="flex items-center justify-center py-20">
   499→          <Loader2 className="h-6 w-6 animate-spin text-gray-400" />
   500→          <span className="ml-2 text-sm text-gray-400">Cargando plantilla...</span>
   501→        </div>
   502→      </AppShell>
   503→    )
   504→  }
   505→
   506→  const gradeCount = previewCertData
   507→    ? Object.values(previewCertData.calificaciones).flat().filter(c => c.nota && c.nota !== '' && !/^\*+$/.test(c.nota)).length
   508→    : 0
   509→
   510→  return (
   511→    <AppShell>
   512→      <div className="space-y-4">
   513→        {/* Top bar - SAME layout as /validar */}
   514→        <div className="flex items-center gap-3 flex-wrap">
   515→          <Button size="sm" variant="outline" onClick={() => router.push('/editor-formatos')} className="h-8 text-xs">
   516→            <ArrowLeft className="h-3 w-3 mr-1" /> Volver
   517→          </Button>
   518→          <div className="flex-1 min-w-[250px]">
   519→            <StudentSearch

   521→              placeholder="Buscar alumno para vista previa con datos reales..."
   522→              plan={plan}
   523→              autoFocus
   524→            />
   525→          </div>
   526→          {previewStudent && (
   527→            <div className="flex items-center gap-2">
   528→              <Badge variant={plan === 'derogado' ? 'destructive' : 'default'}>
   529→                {plan === 'derogado' ? 'Plan Derogado' : 'Plan Vigente'}
   530→              </Badge>
   531→              <span className="text-sm font-medium text-white">
   532→                {previewStudent.apellidos}, {previewStudent.nombres}
   533→              </span>
   534→              <span className="text-xs text-gray-400">
   535→                C.I.: {formatCedulaFinal(previewStudent.cedula)}
   536→              </span>
   537→              {gradeCount > 0 && (
   538→                <Badge variant="outline" className="text-emerald-400 border-emerald-700">
   539→                  {gradeCount} notas
   540→                </Badge>
   541→              )}
   542→            </div>
   543→          )}
   544→          <Button size="sm" variant="outline" onClick={handlePrint} className="h-8 text-xs">
   545→            <Printer className="h-3.5 w-3.5 mr-1" /> Imprimir
   546→          </Button>
   547→          <Button size="sm" onClick={handleSave} disabled={saving} className="h-8 text-xs bg-blue-600 hover:bg-blue-500">
   548→            {saving ? <Loader2 className="h-3 w-3 mr-1 animate-spin" /> : <Save className="h-3 w-3 mr-1" />}
   549→            {saving ? 'Guardando...' : 'Guardar'}
   550→          </Button>
   551→          {loadingPreview && <Loader2 className="h-4 w-4 animate-spin text-gray-400" />}
   552→        </div>
   553→
   554→        {/* Config bar */}
   555→        <div className="flex items-center gap-3 flex-wrap">
   556→          <div className="flex items-center gap-2">
   557→            <Label className="text-xs text-gray-400">Tamano hoja:</Label>

   565→              <option value="a4">A4</option>
   566→            </select>

   568→          <div className="flex items-center gap-2">

   575→            />
   576→            <Label htmlFor="show-grades-edit" className="text-xs text-gray-400">Tabla de calificaciones</Label>

   578→          {template.showGradesTable && (
   579→            <Input
   580→              className="bg-gray-900 text-white border-gray-700 text-xs h-8 max-w-[250px]"

   591→        {/* Document - SAME structure as /validar but every line is editable */}
   592→        <div className="bg-white rounded border shadow-lg mx-auto" style={{ maxWidth: '760px' }}>
   593→          <div ref={previewRef} className="p-8" style={{ fontFamily: 'Arial, sans-serif', fontSize: '11pt', lineHeight: '1.6', color: '#000' }}>
   594→
   595→            {/* Header */}

   614→

   616→            {template.bodyParagraphs.map((para, i) => {

   618→              const isTitle = i === 0 && !para.includes('{{') && para.trim().length > 0
   619→
   620→              if (isTitle) {
   621→                return (
   622→                  <div key={`b-${i}`} style={{ margin: '16pt 0 10pt' }}>
   623→                    <EditableLine
   624→                      value={para}
   625→                      onChange={(val) => updateBodyLine(i, val)}
   626→                      onRemove={() => removeBodyLine(i)}
   627→                      onAddAfter={() => addBodyLine(i)}
   628→                      canRemove={template.bodyParagraphs.length > 1}
   629→                      className="text-center font-bold underline"
   630→                      style={{ fontSize: '12pt' }}
   631→                      placeholder="Titulo del documento"
   632→                    />
   633→                  </div>
   634→                )
   635→              }
   636→
   637→              if (isStudentLine) {
   638→                return (
   639→                  <div key={`b-${i}`} className="text-center" style={{ margin: '10pt 0' }}>
   640→                    <EditableLine
   641→                      value={para}
   642→                      onChange={(val) => updateBodyLine(i, val)}
   643→                      onRemove={() => removeBodyLine(i)}
   644→                      onAddAfter={() => addBodyLine(i)}
   645→                      canRemove={template.bodyParagraphs.length > 1}
   646→                      className="font-bold"
   647→                      style={{ fontSize: '12pt', display: 'inline-block', border: '1px solid #999', padding: '6pt 20pt', minWidth: '280pt' }}
   648→                      placeholder="Datos del estudiante"
   649→                    />

   652→              }
   653→
   654→              if (para.trim() === '') {

   656→                  <EditableLine
   657→                    key={`b-${i}`}
   658→                    value={para}
   659→                    onChange={(val) => updateBodyLine(i, val)}
   660→                    onRemove={() => removeBodyLine(i)}
   661→                    onAddAfter={() => addBodyLine(i)}
   662→                    canRemove={template.bodyParagraphs.length > 1}

   664→                  />
   665→                )
   666→              }
   667→
   668→              return (
   669→                <EditableLine
   670→                  key={`b-${i}`}
   671→                  value={para}
   672→                  onChange={(val) => updateBodyLine(i, val)}
   673→                  onRemove={() => removeBodyLine(i)}
   674→                  onAddAfter={() => addBodyLine(i)}
   675→                  canRemove={template.bodyParagraphs.length > 1}

   685→                <div className="text-center font-bold" style={{ margin: '10pt 0' }}>

   693→                        <th style={{ border: '1px solid #333', padding: '3px 5px', textAlign: 'center' }}>Asignatura</th>
   694→                        <th style={{ border: '1px solid #333', padding: '3px 5px', textAlign: 'center' }}>Nota</th>
   695→                        <th style={{ border: '1px solid #333', padding: '3px 5px', textAlign: 'center' }}>Literal</th>
   696→                        <th style={{ border: '1px solid #333', padding: '3px 5px', textAlign: 'center' }}>Eval.</th>

   702→                        if (!grades || grades.length === 0) return null

   704→                        if (validGrades.length === 0) return null

   706→                          <tr key={`${yearName}-${gi}`}>
   707→                            {gi === 0 && (
   708→                              <td style={{ border: '1px solid #333', padding: '3px 5px', textAlign: 'center', verticalAlign: 'top', fontWeight: 'bold' }}
   709→                                rowSpan={validGrades.length}>

   716→                            <td style={{ border: '1px solid #333', padding: '3px 5px', textAlign: 'center' }}>{g.tipoEvaluacion || ''}</td>

   718→                        ))

   721→                  </table>
   722→                </div>
   723→              </>

   726→            {/* Footer */}
   727→            <div style={{ marginTop: '28pt' }}>
   728→              {template.footerLines.map((line, i) => {
   729→                if (line.trim() === '') {
   730→                  return (
   731→                    <EditableLine
   732→                      key={`f-${i}`}
   733→                      value={line}
   734→                      onChange={(val) => updateFooterLine(i, val)}
   735→                      onRemove={() => removeFooterLine(i)}
   736→                      onAddAfter={() => addFooterLine(i)}
   737→                      canRemove={template.footerLines.length > 1}
   738→                      style={{ height: '8pt' }}
   739→                    />
   740→                  )
   741→                }
   742→
   743→                const isSigLine = /^_{3,}/.test(line.trim())
   744→                const isDirectorName = line.includes('{{director.apellidosNombres}}') && !line.includes('C.I.')
   745→                const isCedula = line.includes('{{director.cedula}}')
   746→                const isRole = line.includes('Secretaria') || (line.includes('Director') && !line.includes('{{director'))
   747→
   748→                return (
   749→                  <EditableLine
   750→                    key={`f-${i}`}
   751→                    value={line}
   752→                    onChange={(val) => updateFooterLine(i, val)}
   753→                    onRemove={() => removeFooterLine(i)}
   754→                    onAddAfter={() => addFooterLine(i)}
   755→                    canRemove={template.footerLines.length > 1}

   757→                    style={isSigLine ? { marginTop: '28pt' } : isCedula || isRole ? {} : { margin: '4pt 0' }}
   758→                  />
   759→                )
   760→              })}
   761→            </div>
   762→          </div>
   763→        </div>
   764→
   765→        {/* Hint */}
   766→        {!previewStudent && (
   767→          <div className="text-center">
   768→            <span className="text-[10px] text-gray-500">
   769→              Mostrando datos de ejemplo. Busca un alumno arriba para ver con datos reales.
   770→            </span>
   771→          </div>
   772→        )}
   773→      </div>
   774→    </AppShell>
   775→  )
   776→}
