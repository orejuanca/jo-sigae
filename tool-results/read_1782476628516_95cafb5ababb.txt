    65→// === Grid helpers ===
    66→function loadGridConfig(): GridConfig {
    67→  if (typeof window === 'undefined') return createDefaultTemplate()
    68→  try {
    69→    const raw = localStorage.getItem(STORAGE_KEY)
    70→    if (raw) {
    71→      const parsed = JSON.parse(raw) as GridConfig
    72→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) return patchDataBindings(parsed)
    73→    }
    74→  } catch { /* ignore */ }
    75→  return createDefaultTemplate()
    76→}
    77→
    78→function saveGridConfig(config: GridConfig) {
    79→  try {
    80→    const patched = patchDataBindings(config)
    81→    const json = JSON.stringify(patched)
    82→    console.log('[saveGridConfig] size:', (json.length / 1024).toFixed(1), 'KB')
    83→    localStorage.setItem(STORAGE_KEY, json)
    84→  } catch (e) {
    85→    console.error('[saveGridConfig] FAILED:', e)
    86→  }
    87→}

    89→function updateCellInConfig(
    90→  config: GridConfig,
    91→  rowIdx: number,
    92→  colIdx: number,
    93→  updates: Partial<CellConfig>
    94→): GridConfig {
    95→  const newRows = config.rows.map((r, i) => {
    96→    if (i !== rowIdx) return r
    97→    const cell = r.cells[colIdx] || emptyCell()
    98→    return { cells: { ...r.cells, [colIdx]: { ...cell, ...updates } } }
    99→  })
   100→  return { ...config, rows: newRows }
   101→}

   103→// === Grid Rendering Component ===
   104→function GridTable({
   105→  config,
   106→  selectedCell,
   107→  selectionRange,
   108→  onCellMouseDown,
   109→  onCellMouseEnter,
   110→  onCellMouseUp,
   111→  isPreview,
   112→  displayData,
   113→  onCellEdit,
   114→  savingDraft,
   115→}: {
   116→  config: GridConfig
   117→  selectedCell: { row: number; col: number } | null
   118→  selectionRange: { r1: number; c1: number; r2: number; c2: number } | null
   119→  onCellMouseDown: (row: number, col: number) => void
   120→  onCellMouseEnter: (row: number, col: number) => void
   121→  onCellMouseUp: () => void
   122→  isPreview: boolean
   123→  displayData: DisplayData | null
   124→  onCellEdit?: (binding: string, newValue: string) => void
   125→  savingDraft?: boolean
   126→}) {
   127→  // Recompute occupied set on every render to track rowspan AND colspan correctly
   128→  const occupied = useMemo(() => {
   129→    const occ = new Set<string>()
   130→    for (let r = 0; r < config.rows.length; r++) {
   131→      const row = config.rows[r]
   132→      if (!row) continue
   133→      for (const [key, cell] of Object.entries(row.cells)) {
   134→        const c = Number(key)
   135→        const rs = cell.rowspan || 1
   136→        const cs = cell.colspan || 1
   137→        // Mark cells consumed by rowspan (rows below)
   138→        if (rs > 1) {
   139→          for (let dr = 1; dr < rs; dr++) {
   140→            occ.add(`${r + dr}-${c}`)
   141→          }
   142→        }
   143→        // Mark cells consumed by colspan (columns to the right, same row)
   144→        if (cs > 1) {
   145→          for (let dc = 1; dc < cs; dc++) {
   146→            occ.add(`${r}-${c + dc}`)
   147→          }
   148→        }
   149→      }
   150→    }
   151→    return occ
   152→  }, [config.rows])
   153→
   154→  const inRange = (r: number, c: number) => {
   155→    if (!selectionRange) return false
   156→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   157→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   158→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   159→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   160→    return r >= minR && r <= maxR && c >= minC && c <= maxC
   161→  }
   162→
   163→  const isRangeActive = selectionRange !== null
   164→  const isSingleSelected = selectedCell && !isRangeActive
   165→
   166→  return (
   167→    <div
   168→      className="overflow-auto flex-1 bg-white p-2 rounded border"
   169→      style={{ maxHeight: 'calc(100vh - 200px)' }}
   170→      onMouseUp={() => !isPreview && onCellMouseUp()}
   171→    >
   172→      <div style={{ width: '816px', height: '1344px', maxWidth: '100%', margin: '0 auto', boxShadow: '0 1px 3px rgba(0,0,0,0.12)', position: 'relative', overflow: 'hidden' }}>
   173→        <table style={{ borderCollapse: 'collapse', width: '100%', height: '100%', fontSize: '9pt', fontFamily: 'Arial, sans-serif', lineHeight: '1.2', tableLayout: 'fixed' }}>
   174→          <colgroup>
   175→            {config.columnWidths.map((w, i) => (
   176→              <col key={i} style={{ width: w || `${100 / config.totalCols}%` }} />
   177→            ))}
   178→          </colgroup>
   179→          <tbody>
   180→            {config.rows.map((gridRow, r) => {
   181→              const cells: React.ReactNode[] = []
   182→              for (let c = 0; c < config.totalCols; c++) {
   183→                const key = `${r}-${c}`
   184→                if (occupied.has(key)) continue
   185→
   186→                const cell = gridRow.cells[c] || emptyCell()
   187→
   188→                let displayContent = cell.content
   189→                if (isPreview && cell.dataBinding && displayData) {
   190→                  displayContent = resolveBinding(cell.dataBinding, displayData)
   191→                }
   192→
   193→                const borderStyle = (enabled: boolean) =>
   194→                  enabled ? `1px solid ${cell.borderColor}` : 'none'
   195→
   196→                const cellIsSelected = isSingleSelected && selectedCell?.row === r && selectedCell?.col === c
   197→                const cellInRange = isRangeActive && inRange(r, c)
   198→                const hasContent = cell.content || cell.dataBinding
   199→
   200→                cells.push(
   201→                  <td
   202→                    key={key}
   203→                    colSpan={cell.colspan > 1 ? cell.colspan : undefined}
   204→                    rowSpan={cell.rowspan > 1 ? cell.rowspan : undefined}
   205→                    style={{
   206→                      borderTop: borderStyle(cell.borderTop),
   207→                      borderRight: borderStyle(cell.borderRight),
   208→                      borderBottom: borderStyle(cell.borderBottom),
   209→                      borderLeft: borderStyle(cell.borderLeft),
   210→                      width: cell.width || undefined,
   211→                      height: cell.height || undefined,
   212→                      fontSize: `${cell.fontSize}pt`,
   213→                      fontWeight: cell.fontWeight,
   214→                      fontStyle: cell.fontStyle,
   215→                      textAlign: cell.textAlign,
   216→                      verticalAlign: cell.verticalAlign,
   217→                      color: cell.color || undefined,
   218→                      whiteSpace: cell.whiteSpace,
   219→                      padding: cell.padding,
   220→                      cursor: isPreview ? 'default' : 'cell',
   221→                      outline: cellIsSelected ? '2px solid #3b82f6' : undefined,
   222→                      outlineOffset: '-2px',
   223→                      minWidth: hasContent ? undefined : '0px',
   224→                      background: cellInRange
   225→                        ? 'rgba(59,130,246,0.15)'
   226→                        : cellIsSelected
   227→                          ? 'rgba(59,130,246,0.06)'
   228→                          : cell.bgColor || undefined,
   229→                      userSelect: 'none',
   230→                      position: 'relative',
   231→                    }}
   232→                    onMouseDown={() => !isPreview && onCellMouseDown(r, c)}
   233→                    onMouseEnter={() => !isPreview && onCellMouseEnter(r, c)}
   234→                    title={cell.dataBinding ? `[${cell.dataBinding}]` : undefined}
   235→                    onDoubleClick={(e) => {
   236→                      if (isPreview && cell.dataBinding && onCellEdit) {
   237→                        e.stopPropagation()
   238→                        const target = e.currentTarget.querySelector('[data-editable]') as HTMLElement | null
   239→                        if (target) {
   240→                          target.setAttribute('contentEditable', 'true')
   241→                          target.focus()
   242→                          // Select all text
   243→                          const range = document.createRange()
   244→                          range.selectNodeContents(target)
   245→                          const sel = window.getSelection()
   246→                          sel?.removeAllRanges()
   247→                          sel?.addRange(range)
   248→                        }
   249→                      }
   250→                    }}
   251→                  >
   252→                    {!isPreview && cell.dataBinding && (
   253→                      <span style={{
   254→                        position: 'absolute', top: '1px', right: '2px',
   255→                        width: '5px', height: '5px', borderRadius: '50%',
   256→                        background: '#f97316', display: 'inline-block',
   257→                        pointerEvents: 'none',
   258→                      }} />
   259→                    )}
   260→                    {isPreview && cell.dataBinding && !displayContent ? (
   261→                      <span style={{ color: '#ccc' }}>—</span>
   262→                    ) : isPreview && cell.dataBinding && onCellEdit ? (
   263→                      <span
   264→                        data-editable
   265→                        style={{ cursor: 'text', outline: 'none', minWidth: '20px', display: 'inline-block', minHeight: '1em' }}
   266→                        onBlur={(e) => {
   267→                          e.currentTarget.setAttribute('contentEditable', 'false')
   268→                          const newVal = e.currentTarget.textContent || ''
   269→                          if (newVal !== resolveBinding(cell.dataBinding!, displayData!)) {
   270→                            onCellEdit(cell.dataBinding, newVal)
   271→                          }
   272→                        }}
   273→                        onKeyDown={(e) => {
   274→                          if (e.key === 'Enter' && !e.shiftKey) {
   275→                            e.preventDefault()
   276→                            e.currentTarget.blur()
   277→                          }
   278→                          if (e.key === 'Escape') {
   279→                            e.currentTarget.blur()
   280→                          }
   281→                        }}
   282→                        suppressContentEditableWarning
   283→                      >
   284→                        {displayContent}
   285→                      </span>
   286→                    ) : displayContent.startsWith('##LOGO_') && displayContent.endsWith('##') ? (
   287→                      <img
   288→                        src="/logo-gob-mppe.png"
   289→                        alt="Logo Gobierno Bolivariano"
   290→                        style={{ maxWidth: '100%', maxHeight: '100%', width: '100%', height: 'auto', objectFit: 'contain', display: 'block' }}
   291→                      />
   292→                    ) : (
   293→                      displayContent
   294→                    )}
   295→                  </td>
   296→                )
   297→              }
   298→              return <tr key={r}>{cells}</tr>
   299→            })}
   300→          </tbody>
   301→        </table>
   302→      </div>
   303→    </div>
   304→  )
   305→}

   307→// === Column Width Editor ===
   308→function ColumnWidthEditor({
   309→  config,
   310→  onUpdate,
   311→}: {
   312→  config: GridConfig
   313→  onUpdate: (config: GridConfig) => void
   314→}) {
   315→  const [editing, setEditing] = useState<number | null>(null)
   316→  const [value, setValue] = useState('')
   317→
   318→  const startEdit = (idx: number) => {
   319→    setEditing(idx)
   320→    setValue(config.columnWidths[idx] || '')
   321→  }
   322→
   323→  const commitEdit = () => {
   324→    if (editing !== null && value.trim()) {
   325→      const newWidths = [...config.columnWidths]
   326→      newWidths[editing] = value.trim()
   327→      onUpdate({ ...config, columnWidths: newWidths })
   328→    }
   329→    setEditing(null)
   330→  }
   331→
   332→  return (
   333→    <div className="flex gap-0.5 flex-wrap p-1">
   334→      {config.columnWidths.map((w, i) => (
   335→        <div
   336→          key={i}
   337→          className="text-center border rounded px-0.5 py-0.5 min-w-[44px]"
   338→          style={{ fontSize: '7pt', backgroundColor: '#f8f8f8' }}
   339→        >
   340→          <div className="text-muted-foreground" style={{ fontSize: '6pt' }}>C{i}</div>
   341→          {editing === i ? (
   342→            <input
   343→              autoFocus
   344→              value={value}
   345→              onChange={(e) => setValue(e.target.value)}
   346→              onBlur={commitEdit}
   347→              onKeyDown={(e) => { if (e.key === 'Enter') commitEdit() }}
   348→              className="w-10 h-4 text-center text-[7pt] border rounded"
   349→              style={{ padding: '0 1px' }}
   350→            />
   351→          ) : (
   352→            <div
   353→              onClick={() => startEdit(i)}
   354→              className="cursor-pointer hover:bg-accent rounded"
   355→              title="Clic para editar"
   356→            >
   357→              {w}
   358→            </div>
   359→          )}
   360→        </div>
   361→      ))}
   362→    </div>
   363→  )
   364→}

   993→        }
   994→      }
   995→      const newRows = [...prev.rows]
   996→      newRows[mergeRow] = { cells: mergedCells }
   997→      newRows.splice(mergeRow + 1, 1)
   998→      if (selectedCell.row >= newRows.length) {
   999→        setSelectedCell(null)
  1000→      }
