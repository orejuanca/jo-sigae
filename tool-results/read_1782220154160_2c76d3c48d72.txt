    65→    65→function loadGridConfig(): GridConfig {
    66→    66→  if (typeof window === 'undefined') return createDefaultTemplate()
    67→    67→  try {
    68→    68→    const raw = localStorage.getItem(STORAGE_KEY)
    69→    69→    if (raw) {
    70→    70→      const parsed = JSON.parse(raw) as GridConfig
    71→    71→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) return patchDataBindings(parsed)
    72→    72→    }
    73→    73→  } catch { /* ignore */ }
    74→    74→  return createDefaultTemplate()
    75→    75→}

    77→    77→function saveGridConfig(config: GridConfig) {
    78→    78→  try {
    79→    79→    const patched = patchDataBindings(config)
    80→    80→    const json = JSON.stringify(patched)
    81→    81→    console.log('[saveGridConfig] size:', (json.length / 1024).toFixed(1), 'KB')
    82→    82→    localStorage.setItem(STORAGE_KEY, json)
    83→    83→  } catch (e) {
    84→    84→    console.error('[saveGridConfig] FAILED:', e)
    85→    85→  }
    86→    86→}

   103→   103→function GridTable({
   104→   104→  config,
   105→   105→  selectedCell,
   106→   106→  selectionRange,
   107→   107→  onCellMouseDown,
   108→   108→  onCellMouseEnter,
   109→   109→  onCellMouseUp,
   110→   110→  isPreview,
   111→   111→  displayData,
   112→   112→}: {
   113→   113→  config: GridConfig
   114→   114→  selectedCell: { row: number; col: number } | null
   115→   115→  selectionRange: { r1: number; c1: number; r2: number; c2: number } | null
   116→   116→  onCellMouseDown: (row: number, col: number) => void
   117→   117→  onCellMouseEnter: (row: number, col: number) => void
   118→   118→  onCellMouseUp: () => void
   119→   119→  isPreview: boolean
   120→   120→  displayData: DisplayData | null
   121→   121→}) {
   122→   122→  // Recompute occupied set on every render to track rowspan AND colspan correctly
   123→   123→  const occupied = useMemo(() => {
   124→   124→    const occ = new Set<string>()
   125→   125→    for (let r = 0; r < config.rows.length; r++) {
   126→   126→      const row = config.rows[r]
   127→   127→      if (!row) continue
   128→   128→      for (const [key, cell] of Object.entries(row.cells)) {
   129→   129→        const c = Number(key)
   130→   130→        const rs = cell.rowspan || 1
   131→   131→        const cs = cell.colspan || 1
   132→   132→        // Mark cells consumed by rowspan (rows below)
   133→   133→        if (rs > 1) {
   134→   134→          for (let dr = 1; dr < rs; dr++) {
   135→   135→            occ.add(`${r + dr}-${c}`)
   136→   136→          }
   137→   137→        }
   138→   138→        // Mark cells consumed by colspan (columns to the right, same row)
   139→   139→        if (cs > 1) {
   140→   140→          for (let dc = 1; dc < cs; dc++) {
   141→   141→            occ.add(`${r}-${c + dc}`)
   142→   142→          }
   143→   143→        }
   144→   144→      }
   145→   145→    }
   146→   146→    return occ
   147→   147→  }, [config.rows])
   148→   148→
   149→   149→  const inRange = (r: number, c: number) => {
   150→   150→    if (!selectionRange) return false
   151→   151→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   152→   152→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   153→   153→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   154→   154→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   155→   155→    return r >= minR && r <= maxR && c >= minC && c <= maxC
   156→   156→  }
   157→   157→
   158→   158→  const isRangeActive = selectionRange !== null
   159→   159→  const isSingleSelected = selectedCell && !isRangeActive
   160→   160→
   161→   161→  return (
   162→   162→    <div
   163→   163→      className="overflow-auto flex-1 bg-white p-2 rounded border"
   164→   164→      style={{ maxHeight: 'calc(100vh - 200px)' }}
   165→   165→      onMouseUp={() => !isPreview && onCellMouseUp()}
   166→   166→    >
   167→   167→      <div style={{ width: '816px', height: '1344px', maxWidth: '100%', margin: '0 auto', boxShadow: '0 1px 3px rgba(0,0,0,0.12)', position: 'relative', overflow: 'hidden' }}>
   168→   168→        <table style={{ borderCollapse: 'collapse', width: '100%', height: '100%', fontSize: '9pt', fontFamily: 'Arial, sans-serif', lineHeight: '1.2', tableLayout: 'fixed' }}>
   169→   169→          <colgroup>
   170→   170→            {config.columnWidths.map((w, i) => (
   171→   171→              <col key={i} style={{ width: w || `${100 / config.totalCols}%` }} />
   172→   172→            ))}
   173→   173→          </colgroup>
   174→   174→          <tbody>
   175→   175→            {config.rows.map((gridRow, r) => {
   176→   176→              const cells: React.ReactNode[] = []
   177→   177→              for (let c = 0; c < config.totalCols; c++) {
   178→   178→                const key = `${r}-${c}`
   179→   179→                if (occupied.has(key)) continue
   180→   180→
   181→   181→                const cell = gridRow.cells[c] || emptyCell()
   182→   182→
   183→   183→                let displayContent = cell.content
   184→   184→                if (isPreview && cell.dataBinding && displayData) {
   185→   185→                  displayContent = resolveBinding(cell.dataBinding, displayData)
   186→   186→                }
   187→   187→
   188→   188→                const borderStyle = (enabled: boolean) =>
   189→   189→                  enabled ? `1px solid ${cell.borderColor}` : 'none'
   190→   190→
   191→   191→                const cellIsSelected = isSingleSelected && selectedCell?.row === r && selectedCell?.col === c
   192→   192→                const cellInRange = isRangeActive && inRange(r, c)
   193→   193→                const hasContent = cell.content || cell.dataBinding
   194→   194→
   195→   195→                cells.push(
   196→   196→                  <td
   197→   197→                    key={key}
   198→   198→                    colSpan={cell.colspan > 1 ? cell.colspan : undefined}
   199→   199→                    rowSpan={cell.rowspan > 1 ? cell.rowspan : undefined}
   200→   200→                    style={{
   201→   201→                      borderTop: borderStyle(cell.borderTop),
   202→   202→                      borderRight: borderStyle(cell.borderRight),
   203→   203→                      borderBottom: borderStyle(cell.borderBottom),
   204→   204→                      borderLeft: borderStyle(cell.borderLeft),
   205→   205→                      width: cell.width || undefined,
   206→   206→                      height: cell.height || undefined,
   207→   207→                      fontSize: `${cell.fontSize}pt`,
   208→   208→                      fontWeight: cell.fontWeight,
   209→   209→                      fontStyle: cell.fontStyle,
   210→   210→                      textAlign: cell.textAlign,
   211→   211→                      verticalAlign: cell.verticalAlign,
   212→   212→                      color: cell.color || undefined,
   213→   213→                      whiteSpace: cell.whiteSpace,
   214→   214→                      padding: cell.padding,
   215→   215→                      cursor: isPreview ? 'default' : 'cell',
   216→   216→                      outline: cellIsSelected ? '2px solid #3b82f6' : undefined,
   217→   217→                      outlineOffset: '-2px',
   218→   218→                      minWidth: hasContent ? undefined : '0px',
   219→   219→                      background: cellInRange
   220→   220→                        ? 'rgba(59,130,246,0.15)'
   221→   221→                        : cellIsSelected
   222→   222→                          ? 'rgba(59,130,246,0.06)'
   223→   223→                          : cell.bgColor || undefined,
   224→   224→                      userSelect: 'none',
   225→   225→                      position: 'relative',
   226→   226→                    }}
   227→   227→                    onMouseDown={() => !isPreview && onCellMouseDown(r, c)}
   228→   228→                    onMouseEnter={() => !isPreview && onCellMouseEnter(r, c)}
   229→   229→                    title={cell.dataBinding ? `[${cell.dataBinding}]` : undefined}
   230→   230→                  >
   231→   231→                    {!isPreview && cell.dataBinding && (
   232→   232→                      <span style={{
   233→   233→                        position: 'absolute', top: '1px', right: '2px',
   234→   234→                        width: '5px', height: '5px', borderRadius: '50%',
   235→   235→                        background: '#f97316', display: 'inline-block',
   236→   236→                        pointerEvents: 'none',
   237→   237→                      }} />
   238→   238→                    )}
   239→   239→                    {isPreview && cell.dataBinding && !displayContent ? (
   240→   240→                      <span style={{ color: '#ccc' }}>—</span>
   241→   241→                    ) : (
   242→   242→                      displayContent
   243→   243→                    )}
   244→   244→                  </td>
   245→   245→                )
   246→   246→              }
   247→   247→              return <tr key={r}>{cells}</tr>
   248→   248→            })}
   249→   249→          </tbody>
   250→   250→        </table>
   251→   251→      </div>
   252→   252→    </div>
   253→   253→  )
   254→   254→}

   257→   257→function ColumnWidthEditor({
   258→   258→  config,
   259→   259→  onUpdate,
   260→   260→}: {
   261→   261→  config: GridConfig
   262→   262→  onUpdate: (config: GridConfig) => void
   263→   263→}) {
   264→   264→  const [editing, setEditing] = useState<number | null>(null)
   265→   265→  const [value, setValue] = useState('')
   266→   266→
   267→   267→  const startEdit = (idx: number) => {
   268→   268→    setEditing(idx)
   269→   269→    setValue(config.columnWidths[idx] || '')
   270→   270→  }
   271→   271→
   272→   272→  const commitEdit = () => {
   273→   273→    if (editing !== null && value.trim()) {
   274→   274→      const newWidths = [...config.columnWidths]
   275→   275→      newWidths[editing] = value.trim()
   276→   276→      onUpdate({ ...config, columnWidths: newWidths })
   277→   277→    }
   278→   278→    setEditing(null)
   279→   279→  }
   280→   280→
   281→   281→  return (
   282→   282→    <div className="flex gap-0.5 flex-wrap p-1">
   283→   283→      {config.columnWidths.map((w, i) => (
   284→   284→        <div
   285→   285→          key={i}
   286→   286→          className="text-center border rounded px-0.5 py-0.5 min-w-[44px]"
   287→   287→          style={{ fontSize: '7pt', backgroundColor: '#f8f8f8' }}
   288→   288→        >
   289→   289→          <div className="text-muted-foreground" style={{ fontSize: '6pt' }}>C{i}</div>
   290→   290→          {editing === i ? (
   291→   291→            <input
   292→   292→              autoFocus
   293→   293→              value={value}
   294→   294→              onChange={(e) => setValue(e.target.value)}
   295→   295→              onBlur={commitEdit}
   296→   296→              onKeyDown={(e) => { if (e.key === 'Enter') commitEdit() }}
   297→   297→              className="w-10 h-4 text-center text-[7pt] border rounded"
   298→   298→              style={{ padding: '0 1px' }}
   299→   299→            />
   300→   300→          ) : (
   301→   301→            <div
   302→   302→              onClick={() => startEdit(i)}
   303→   303→              className="cursor-pointer hover:bg-accent rounded"
   304→   304→              title="Clic para editar"
   305→   305→            >
   306→   306→              {w}
   307→   307→            </div>
   308→   308→          )}
   309→   309→        </div>
   310→   310→      ))}
   311→   311→    </div>
   312→   312→  )
