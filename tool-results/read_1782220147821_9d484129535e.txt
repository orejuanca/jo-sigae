    64→// === Grid helpers ===
    65→function loadGridConfig(): GridConfig {
    66→  if (typeof window === 'undefined') return createDefaultTemplate()
    67→  try {
    68→    const raw = localStorage.getItem(STORAGE_KEY)
    69→    if (raw) {
    70→      const parsed = JSON.parse(raw) as GridConfig
    71→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) return patchDataBindings(parsed)
    72→    }
    73→  } catch { /* ignore */ }
    74→  return createDefaultTemplate()
    75→}
    76→
    77→function saveGridConfig(config: GridConfig) {
    78→  try {
    79→    const patched = patchDataBindings(config)
    80→    const json = JSON.stringify(patched)
    81→    console.log('[saveGridConfig] size:', (json.length / 1024).toFixed(1), 'KB')
    82→    localStorage.setItem(STORAGE_KEY, json)
    83→  } catch (e) {
    84→    console.error('[saveGridConfig] FAILED:', e)
    85→  }
    86→}

    88→function updateCellInConfig(
    89→  config: GridConfig,
    90→  rowIdx: number,
    91→  colIdx: number,
    92→  updates: Partial<CellConfig>
    93→): GridConfig {
    94→  const newRows = config.rows.map((r, i) => {
    95→    if (i !== rowIdx) return r
    96→    const cell = r.cells[colIdx] || emptyCell()
    97→    return { cells: { ...r.cells, [colIdx]: { ...cell, ...updates } } }
    98→  })
    99→  return { ...config, rows: newRows }
   100→}

   102→// === Grid Rendering Component ===
   103→function GridTable({
   104→  config,
   105→  selectedCell,
   106→  selectionRange,
   107→  onCellMouseDown,
   108→  onCellMouseEnter,
   109→  onCellMouseUp,
   110→  isPreview,
   111→  displayData,
   112→}: {
   113→  config: GridConfig
   114→  selectedCell: { row: number; col: number } | null
   115→  selectionRange: { r1: number; c1: number; r2: number; c2: number } | null
   116→  onCellMouseDown: (row: number, col: number) => void
   117→  onCellMouseEnter: (row: number, col: number) => void
   118→  onCellMouseUp: () => void
   119→  isPreview: boolean
   120→  displayData: DisplayData | null
   121→}) {
   122→  // Recompute occupied set on every render to track rowspan AND colspan correctly
   123→  const occupied = useMemo(() => {
   124→    const occ = new Set<string>()
   125→    for (let r = 0; r < config.rows.length; r++) {
   126→      const row = config.rows[r]
   127→      if (!row) continue
   128→      for (const [key, cell] of Object.entries(row.cells)) {
   129→        const c = Number(key)
   130→        const rs = cell.rowspan || 1
   131→        const cs = cell.colspan || 1
   132→        // Mark cells consumed by rowspan (rows below)
   133→        if (rs > 1) {
   134→          for (let dr = 1; dr < rs; dr++) {
   135→            occ.add(`${r + dr}-${c}`)
   136→          }
   137→        }
   138→        // Mark cells consumed by colspan (columns to the right, same row)
   139→        if (cs > 1) {
   140→          for (let dc = 1; dc < cs; dc++) {
   141→            occ.add(`${r}-${c + dc}`)
   142→          }
   143→        }
   144→      }
   145→    }
   146→    return occ
   147→  }, [config.rows])
   148→
   149→  const inRange = (r: number, c: number) => {
   150→    if (!selectionRange) return false
   151→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   152→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   153→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   154→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   155→    return r >= minR && r <= maxR && c >= minC && c <= maxC
   156→  }
   157→
   158→  const isRangeActive = selectionRange !== null
   159→  const isSingleSelected = selectedCell && !isRangeActive
   160→
   161→  return (
   162→    <div
   163→      className="overflow-auto flex-1 bg-white p-2 rounded border"
   164→      style={{ maxHeight: 'calc(100vh - 200px)' }}
   165→      onMouseUp={() => !isPreview && onCellMouseUp()}
   166→    >
   167→      <div style={{ width: '816px', height: '1344px', maxWidth: '100%', margin: '0 auto', boxShadow: '0 1px 3px rgba(0,0,0,0.12)', position: 'relative', overflow: 'hidden' }}>
   168→        <table style={{ borderCollapse: 'collapse', width: '100%', height: '100%', fontSize: '9pt', fontFamily: 'Arial, sans-serif', lineHeight: '1.2', tableLayout: 'fixed' }}>
   169→          <colgroup>
   170→            {config.columnWidths.map((w, i) => (
   171→              <col key={i} style={{ width: w || `${100 / config.totalCols}%` }} />
   172→            ))}
   173→          </colgroup>
   174→          <tbody>
   175→            {config.rows.map((gridRow, r) => {
   176→              const cells: React.ReactNode[] = []
   177→              for (let c = 0; c < config.totalCols; c++) {
   178→                const key = `${r}-${c}`
   179→                if (occupied.has(key)) continue
   180→
   181→                const cell = gridRow.cells[c] || emptyCell()
   182→
   183→                let displayContent = cell.content
   184→                if (isPreview && cell.dataBinding && displayData) {
   185→                  displayContent = resolveBinding(cell.dataBinding, displayData)
   186→                }
   187→
   188→                const borderStyle = (enabled: boolean) =>
   189→                  enabled ? `1px solid ${cell.borderColor}` : 'none'
   190→
   191→                const cellIsSelected = isSingleSelected && selectedCell?.row === r && selectedCell?.col === c
   192→                const cellInRange = isRangeActive && inRange(r, c)
   193→                const hasContent = cell.content || cell.dataBinding
   194→
   195→                cells.push(
   196→                  <td
   197→                    key={key}
   198→                    colSpan={cell.colspan > 1 ? cell.colspan : undefined}
   199→                    rowSpan={cell.rowspan > 1 ? cell.rowspan : undefined}
   200→                    style={{
   201→                      borderTop: borderStyle(cell.borderTop),
   202→                      borderRight: borderStyle(cell.borderRight),
   203→                      borderBottom: borderStyle(cell.borderBottom),
   204→                      borderLeft: borderStyle(cell.borderLeft),
   205→                      width: cell.width || undefined,
   206→                      height: cell.height || undefined,
   207→                      fontSize: `${cell.fontSize}pt`,
   208→                      fontWeight: cell.fontWeight,
   209→                      fontStyle: cell.fontStyle,
   210→                      textAlign: cell.textAlign,
   211→                      verticalAlign: cell.verticalAlign,
   212→                      color: cell.color || undefined,
   213→                      whiteSpace: cell.whiteSpace,
   214→                      padding: cell.padding,
   215→                      cursor: isPreview ? 'default' : 'cell',
   216→                      outline: cellIsSelected ? '2px solid #3b82f6' : undefined,
   217→                      outlineOffset: '-2px',
   218→                      minWidth: hasContent ? undefined : '0px',
   219→                      background: cellInRange
   220→                        ? 'rgba(59,130,246,0.15)'
   221→                        : cellIsSelected
   222→                          ? 'rgba(59,130,246,0.06)'
   223→                          : cell.bgColor || undefined,
   224→                      userSelect: 'none',
   225→                      position: 'relative',
   226→                    }}
   227→                    onMouseDown={() => !isPreview && onCellMouseDown(r, c)}
   228→                    onMouseEnter={() => !isPreview && onCellMouseEnter(r, c)}
   229→                    title={cell.dataBinding ? `[${cell.dataBinding}]` : undefined}
   230→                  >
   231→                    {!isPreview && cell.dataBinding && (
   232→                      <span style={{
   233→                        position: 'absolute', top: '1px', right: '2px',
   234→                        width: '5px', height: '5px', borderRadius: '50%',
   235→                        background: '#f97316', display: 'inline-block',
   236→                        pointerEvents: 'none',
   237→                      }} />
   238→                    )}
   239→                    {isPreview && cell.dataBinding && !displayContent ? (
   240→                      <span style={{ color: '#ccc' }}>—</span>
   241→                    ) : (
   242→                      displayContent
   243→                    )}
   244→                  </td>
   245→                )
   246→              }
   247→              return <tr key={r}>{cells}</tr>
   248→            })}
   249→          </tbody>
   250→        </table>
   251→      </div>
   252→    </div>
   253→  )
   254→}

   256→// === Column Width Editor ===
   257→function ColumnWidthEditor({
   258→  config,
   259→  onUpdate,
   260→}: {
   261→  config: GridConfig
   262→  onUpdate: (config: GridConfig) => void
   263→}) {
   264→  const [editing, setEditing] = useState<number | null>(null)
   265→  const [value, setValue] = useState('')
   266→
   267→  const startEdit = (idx: number) => {
   268→    setEditing(idx)
   269→    setValue(config.columnWidths[idx] || '')
   270→  }
   271→
   272→  const commitEdit = () => {
   273→    if (editing !== null && value.trim()) {
   274→      const newWidths = [...config.columnWidths]
   275→      newWidths[editing] = value.trim()
   276→      onUpdate({ ...config, columnWidths: newWidths })
   277→    }
   278→    setEditing(null)
   279→  }
   280→
   281→  return (
   282→    <div className="flex gap-0.5 flex-wrap p-1">
   283→      {config.columnWidths.map((w, i) => (
   284→        <div
   285→          key={i}
   286→          className="text-center border rounded px-0.5 py-0.5 min-w-[44px]"
   287→          style={{ fontSize: '7pt', backgroundColor: '#f8f8f8' }}
   288→        >
   289→          <div className="text-muted-foreground" style={{ fontSize: '6pt' }}>C{i}</div>
   290→          {editing === i ? (
   291→            <input
   292→              autoFocus
   293→              value={value}
   294→              onChange={(e) => setValue(e.target.value)}
   295→              onBlur={commitEdit}
   296→              onKeyDown={(e) => { if (e.key === 'Enter') commitEdit() }}
   297→              className="w-10 h-4 text-center text-[7pt] border rounded"
   298→              style={{ padding: '0 1px' }}
   299→            />
   300→          ) : (
   301→            <div
   302→              onClick={() => startEdit(i)}
   303→              className="cursor-pointer hover:bg-accent rounded"
   304→              title="Clic para editar"
   305→            >
   306→              {w}
   307→            </div>
   308→          )}
   309→        </div>
   310→      ))}
   311→    </div>
   312→  )
   313→}

   538→    setSelEnd(null)

   540→  }, [])
   541→
   542→  const handleCellUpdate = useCallback((updates: Partial<CellConfig>) => {
   543→    if (!selectedCell) return
   544→    setGridConfig((prev) => updateCellInConfig(prev, selectedCell.row, selectedCell.col, updates))
   545→  }, [selectedCell])
   546→
   547→  const handleApplyColumns = () => {
   548→    const n = parseInt(colInput)
   549→    if (isNaN(n) || n < 1 || n > 100) {
   550→      toast({ title: 'Error', description: 'Ingrese un número válido de columnas (1-100).', variant: 'destructive' })
   551→      return
   552→    }
   553→    const newConfig: GridConfig = {
   554→      totalCols: n,
   555→      columnWidths: Array(n).fill(`${(100 / n).toFixed(2)}%`),
   556→      rows: [emptyRow(n)],
   557→    }
   558→    setGridConfig(newConfig)
   559→    setSelectedCell(null)
   560→    toast({ title: `Grilla reiniciada con ${n} columnas` })
   561→  }
   562→
   563→  const handleAddRow = () => {
   564→    setGridConfig((prev) => ({
   565→      ...prev,
   566→      rows: [...prev.rows, emptyRow(prev.totalCols)],
   567→    }))
   568→  }
   569→
   570→  const handleInsertRowAbove = () => {
   571→    if (!selectedCell) {
   572→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar fila arriba.', variant: 'destructive' })
   573→      return
   574→    }
   575→    setGridConfig((prev) => {
   576→      const newRow = emptyRow(prev.totalCols)
   577→      const newRows = [...prev.rows]
   578→
   579→      // Adjust rowspan of cells in the row above (selectedCell.row - 1)
   580→      // that span across the insertion point
   581→      const aboveIdx = selectedCell.row - 1
   582→      if (aboveIdx >= 0 && aboveIdx < prev.rows.length) {
   583→        const aboveRow = prev.rows[aboveIdx]
   584→        const adjustedCells: Record<number, CellConfig> = {}
   585→        for (const [key, cell] of Object.entries(aboveRow.cells)) {
   586→          const col = Number(key)
   587→          const rs = cell.rowspan || 1
   588→          // If this cell spans to or past the insertion row, reduce rowspan
   589→          if (col === 0 || true) { // check all cells
   590→            if (rs > 1 && (aboveIdx + rs) > selectedCell.row) {
   591→              adjustedCells[col] = { ...cell, rowspan: rs - 1 }
   592→              continue
   593→            }
   594→          }
   595→          adjustedCells[col] = cell
   596→        }
   597→        newRows[aboveIdx] = { cells: adjustedCells }
   598→      }
   599→
   600→      // Also adjust rowspan of cells in the CURRENT row (selectedCell.row)
   601→      // since it shifts down by 1, but their rowspan origin stays the same
   602→      const currentRow = prev.rows[selectedCell.row]
   603→      if (currentRow) {
   604→        const adjustedCurrent: Record<number, CellConfig> = {}
   605→        for (const [key, cell] of Object.entries(currentRow.cells)) {
   606→          const col = Number(key)
   607→          const rs = cell.rowspan || 1
   608→          // Cells that started in a row above and were part of a rowspan
   609→          // don't need adjustment - they're already handled by the row above
   610→          adjustedCurrent[col] = cell
   611→        }
   612→        // Keep the current row as-is (it just moves down)
   613→      }
   614→
   615→      newRows.splice(selectedCell.row, 0, newRow)
   616→      setSelectedCell({ row: selectedCell.row + 1, col: selectedCell.col })
   617→      return { ...prev, rows: newRows }
   618→    })
   619→  }
   620→
   621→  const handleInsertRowBelow = () => {
   622→    if (!selectedCell) {
   623→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar fila abajo.', variant: 'destructive' })
   624→      return
   625→    }
   626→    setGridConfig((prev) => {
   627→      const newRow = emptyRow(prev.totalCols)
   628→      const insertAt = selectedCell.row + 1
   629→      const newRows = [...prev.rows]
   630→
   631→      // Adjust rowspan of cells in selectedCell.row that span across insertAt
   632→      const currentRow = prev.rows[selectedCell.row]
   633→      if (currentRow) {
   634→        const adjustedCells: Record<number, CellConfig> = {}
   635→        for (const [key, cell] of Object.entries(currentRow.cells)) {
   636→          const col = Number(key)
   637→          const rs = cell.rowspan || 1
   638→          if (rs > 1 && (selectedCell.row + rs) > insertAt) {
   639→            adjustedCells[col] = { ...cell, rowspan: rs - 1 }
   640→            continue
   641→          }
   642→          adjustedCells[col] = cell
   643→        }
   644→        newRows[selectedCell.row] = { cells: adjustedCells }
   645→      }
   646→
   647→      newRows.splice(insertAt, 0, newRow)
   648→      return { ...prev, rows: newRows }
   649→    })
   650→  }
   651→
   652→  const handleDeleteLastRow = () => {
   653→    setGridConfig((prev) => {
   654→      if (prev.rows.length <= 1) {
   655→        toast({ title: 'No se puede eliminar la última fila', variant: 'destructive' })
   656→        return prev
   657→      }
   658→      const newRows = prev.rows.slice(0, -1)
   659→      // Clear selection if it was on the deleted row
   660→      if (selectedCell && selectedCell.row >= newRows.length) {
   661→        setSelectedCell(null)
   662→      }
   663→      return { ...prev, rows: newRows }
   664→    })
   665→  }
   666→
   667→  // === Column Operations ===
   668→  const handleInsertColLeft = () => {
   669→    if (!selectedCell) {
   670→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar columna a la izquierda.', variant: 'destructive' })
   671→      return
   672→    }
   673→    setGridConfig((prev) => {
   674→      const insertAt = selectedCell.col
   675→      // Shift cell indices >= insertAt in every row
   676→      const newRows = prev.rows.map((row) => {
   677→        const newCells: Record<number, CellConfig> = {}
   678→        for (const [key, cell] of Object.entries(row.cells)) {
   679→          const idx = Number(key)
   680→          if (idx >= insertAt) {
   681→            newCells[idx + 1] = cell
   682→          } else {
   683→            newCells[idx] = cell
   684→          }
   685→        }
   686→        newCells[insertAt] = emptyCell()
   687→        return { cells: newCells }
   688→      })
   689→      // Insert column width
   690→      const newWidths = [...prev.columnWidths]
   691→      newWidths.splice(insertAt, 0, '3%')
   692→      // Adjust selection
   693→      setSelectedCell({ row: selectedCell.row, col: selectedCell.col + 1 })
   694→      return { ...prev, totalCols: prev.totalCols + 1, columnWidths: newWidths, rows: newRows }
   695→    })
   696→    setColInput(String(gridConfig.totalCols + 1))
   697→  }
   698→
   699→  const handleInsertColRight = () => {
   700→    if (!selectedCell) {
   701→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar columna a la derecha.', variant: 'destructive' })
   702→      return
   703→    }
   704→    setGridConfig((prev) => {
   705→      const insertAt = selectedCell.col + 1
   706→      // Shift cell indices >= insertAt in every row
   707→      const newRows = prev.rows.map((row) => {
   708→        const newCells: Record<number, CellConfig> = {}
   709→        for (const [key, cell] of Object.entries(row.cells)) {
   710→          const idx = Number(key)
   711→          if (idx >= insertAt) {
   712→            newCells[idx + 1] = cell
   713→          } else {
   714→            newCells[idx] = cell
   715→          }
   716→        }
   717→        newCells[insertAt] = emptyCell()
   718→        return { cells: newCells }
   719→      })
   720→      // Insert column width
   721→      const newWidths = [...prev.columnWidths]
   722→      newWidths.splice(insertAt, 0, '3%')
   723→      // Selection stays on same column
   724→      return { ...prev, totalCols: prev.totalCols + 1, columnWidths: newWidths, rows: newRows }
   725→    })
   726→    setColInput(String(gridConfig.totalCols + 1))
   727→  }
   728→
   729→  const handleDeleteLastCol = () => {
   730→    setGridConfig((prev) => {
   731→      if (prev.totalCols <= 1) {
   732→        toast({ title: 'No se puede eliminar la última columna', variant: 'destructive' })
   733→        return prev
   734→      }
   735→      const lastCol = prev.totalCols - 1
   736→      const newRows = prev.rows.map((row) => {
   737→        const { [lastCol]: _removed, ...rest } = row.cells
   738→        return { cells: rest }
   739→      })
   740→      const newWidths = prev.columnWidths.slice(0, -1)
   741→      if (selectedCell && selectedCell.col >= prev.totalCols - 1) {
   742→        setSelectedCell(null)
   743→      }
   744→      return { ...prev, totalCols: prev.totalCols - 1, columnWidths: newWidths, rows: newRows }
   745→    })
   746→    setColInput(String(gridConfig.totalCols - 1))
   747→  }
   748→
   749→  // === Merge Selection (drag range) ===
   750→  const handleMergeSelection = () => {
   751→    if (!selectionRange) {
   752→      toast({ title: 'Sin selección', description: 'Selecciona varias celdas arrastrando primero.', variant: 'destructive' })
   753→      return
   754→    }
   755→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   756→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   757→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   758→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   759→    const spanCols = maxC - minC + 1
   760→    const spanRows = maxR - minR + 1
   761→    if (spanCols === 1 && spanRows === 1) {
   762→      toast({ title: 'Una sola celda', description: 'Selecciona al menos 2 celdas para combinar.', variant: 'destructive' })
   763→      return
   764→    }
   765→    setGridConfig((prev) => {
   766→      const updated = updateCellInConfig(prev, minR, minC, {
   767→        colspan: spanCols,
   768→        rowspan: spanRows,
   769→      })
   770→      const newRows = updated.rows.map((row, ri) => {
   771→        if (ri < minR || ri > maxR) return row
   772→        const newCells = { ...row.cells }
   773→        for (let ci = minC; ci <= maxC; ci++) {
   774→          if (ri === minR && ci === minC) continue
   775→          delete newCells[ci]
   776→        }
   777→        return { cells: newCells }
   778→      })
   779→      return { ...updated, rows: newRows }
   780→    })
   781→    setSelectedCell({ row: minR, col: minC })
   782→    clearRangeSelection()
   783→    toast({ title: 'Celdas combinadas', description: `${spanCols}x${spanRows} celdas fusionadas en una.` })
   784→  }
   785→
   786→  // === Merge / Split Operations ===
   787→  // Combina columnas: elimina la columna de la derecha y suma su ancho a la actual.
   788→  // El total de columnas baja (ej: 27 → 26).
   789→  const handleMergeColumns = () => {
   790→    if (!selectedCell) {
   791→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   792→      return
   793→    }
   794→    if (gridConfig.totalCols <= 1) {
   795→      toast({ title: 'No se puede', description: 'Solo queda 1 columna.', variant: 'destructive' })
   796→      return
   797→    }
   798→    const mergeCol = selectedCell.col
   799→    if (mergeCol >= gridConfig.totalCols - 1) {
   800→      toast({ title: 'No se puede', description: 'No hay columna a la derecha para combinar.', variant: 'destructive' })
   801→      return
   802→    }
   803→    setGridConfig((prev) => {
   804→      // Sumar ancho de la columna derecha a la actual
   805→      const newWidths = [...prev.columnWidths]
   806→      const leftW = parseFloat(newWidths[mergeCol]) || 0
   807→      const rightW = parseFloat(newWidths[mergeCol + 1]) || 0
   808→      newWidths[mergeCol] = `${leftW + rightW}%`
   809→      newWidths.splice(mergeCol + 1, 1)
   810→      // En cada fila, remover la columna mergeCol+1 y reindexar
   811→      const newRows = prev.rows.map((row) => {
   812→        const newCells: Record<number, CellConfig> = {}
   813→        for (const [key, cell] of Object.entries(row.cells)) {
   814→          const idx = Number(key)
   815→          if (idx === mergeCol + 1) continue // descartar columna derecha
   816→          if (idx > mergeCol + 1) {
   817→            newCells[idx - 1] = cell // reindexar
   818→          } else {
   819→            newCells[idx] = cell
   820→          }
   821→        }
   822→        return { cells: newCells }
   823→      })
   824→      // Ajustar seleccion si estaba mas alla de la columna eliminada
   825→      if (selectedCell.col > mergeCol) {
   826→        setSelectedCell({ row: selectedCell.row, col: Math.max(0, selectedCell.col - 1) })
   827→      }
   828→      const newTotal = prev.totalCols - 1
   829→      return { ...prev, totalCols: newTotal, columnWidths: newWidths, rows: newRows }
   830→    })
   831→    setColInput(String(gridConfig.totalCols - 1))
   832→  }
   833→
   834→  // Combina filas: elimina la fila de abajo y la suma a la actual.
   835→  // El total de filas baja en 1.
   836→  const handleMergeRows = () => {
   837→    if (!selectedCell) {
   838→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   839→      return
   840→    }
   841→    if (gridConfig.rows.length <= 1) {
   842→      toast({ title: 'No se puede', description: 'Solo queda 1 fila.', variant: 'destructive' })
   843→      return
   844→    }
   845→    const mergeRow = selectedCell.row
   846→    if (mergeRow >= gridConfig.rows.length - 1) {
   847→      toast({ title: 'No se puede', description: 'No hay fila debajo para combinar.', variant: 'destructive' })
   848→      return
   849→    }
   850→    setGridConfig((prev) => {
   851→      const currentRow = prev.rows[mergeRow]
   852→      const nextRow = prev.rows[mergeRow + 1]
   853→      // Fusionar contenido de celdas: si la celda de abajo tiene contenido y la actual no, copiarlo
   854→      const mergedCells: Record<number, CellConfig> = { ...currentRow.cells }
   855→      for (const [key, cell] of Object.entries(nextRow.cells)) {
   856→        const idx = Number(key)
   857→        const existing = mergedCells[idx]
   858→        if (!existing || (!existing.content && !existing.dataBinding)) {
   859→          mergedCells[idx] = { ...cell }
   860→        } else if (cell.content && !existing.content) {
   861→          mergedCells[idx] = { ...existing, content: existing.content + ' ' + cell.content }
   862→        }
   863→      }
   864→      const newRows = [...prev.rows]
   865→      newRows[mergeRow] = { cells: mergedCells }
   866→      newRows.splice(mergeRow + 1, 1)
   867→      if (selectedCell.row >= newRows.length) {
   868→        setSelectedCell(null)
   869→      }
   870→      return { ...prev, rows: newRows }
   871→    })
   872→  }
   873→
   874→  // Separar celda: reset colspan/rowspan a 1
   875→  const handleSplitCell = () => {
   876→    if (!selectedCell) {
   877→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   878→      return
   879→    }
   880→    const cell = gridConfig.rows[selectedCell.row]?.cells[selectedCell.col]
   881→    if (!cell || (cell.colspan === 1 && cell.rowspan === 1)) {
   882→      toast({ title: 'Sin combinación', description: 'Esta celda no está combinada (colspan/rowspan ya es 1).', variant: 'destructive' })
   883→      return
   884→    }
   885→    setGridConfig((prev) => updateCellInConfig(prev, selectedCell.row, selectedCell.col, { colspan: 1, rowspan: 1 }))
   886→  }

   891→    setShowSaveDialog(true)
