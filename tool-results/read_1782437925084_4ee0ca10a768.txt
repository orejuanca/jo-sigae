    65→// === Grid helpers ===
    66→function loadGridConfig(): GridConfig {
    67→  if (typeof window === 'undefined') return createDefaultTemplate()
    68→  try {
    69→    const raw = localStorage.getItem(STORAGE_KEY)
    70→    if (raw) {
    71→      const parsed = JSON.parse(raw) as GridConfig
    72→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) return patchDataBindings(parsed)
    73→    }
    74→  } catch { /* ignore */ }
    75→  return createDefaultTemplate()
    76→}
    77→
    78→function saveGridConfig(config: GridConfig) {
    79→  try {
    80→    const patched = patchDataBindings(config)
    81→    const json = JSON.stringify(patched)
    82→    console.log('[saveGridConfig] size:', (json.length / 1024).toFixed(1), 'KB')
    83→    localStorage.setItem(STORAGE_KEY, json)
    84→  } catch (e) {
    85→    console.error('[saveGridConfig] FAILED:', e)
    86→  }
    87→}

    89→function updateCellInConfig(
    90→  config: GridConfig,
    91→  rowIdx: number,
    92→  colIdx: number,
    93→  updates: Partial<CellConfig>
    94→): GridConfig {
    95→  const newRows = config.rows.map((r, i) => {
    96→    if (i !== rowIdx) return r
    97→    const cell = r.cells[colIdx] || emptyCell()
    98→    return { cells: { ...r.cells, [colIdx]: { ...cell, ...updates } } }
    99→  })
   100→  return { ...config, rows: newRows }
   101→}

   103→// === Grid Rendering Component ===
   104→function GridTable({
   105→  config,
   106→  selectedCell,
   107→  selectionRange,
   108→  onCellMouseDown,
   109→  onCellMouseEnter,
   110→  onCellMouseUp,
   111→  isPreview,
   112→  displayData,
   113→}: {
   114→  config: GridConfig
   115→  selectedCell: { row: number; col: number } | null
   116→  selectionRange: { r1: number; c1: number; r2: number; c2: number } | null
   117→  onCellMouseDown: (row: number, col: number) => void
   118→  onCellMouseEnter: (row: number, col: number) => void
   119→  onCellMouseUp: () => void
   120→  isPreview: boolean
   121→  displayData: DisplayData | null
   122→}) {
   123→  // Recompute occupied set on every render to track rowspan AND colspan correctly
   124→  const occupied = useMemo(() => {
   125→    const occ = new Set<string>()
   126→    for (let r = 0; r < config.rows.length; r++) {
   127→      const row = config.rows[r]
   128→      if (!row) continue
   129→      for (const [key, cell] of Object.entries(row.cells)) {
   130→        const c = Number(key)
   131→        const rs = cell.rowspan || 1
   132→        const cs = cell.colspan || 1
   133→        // Mark cells consumed by rowspan (rows below)
   134→        if (rs > 1) {
   135→          for (let dr = 1; dr < rs; dr++) {
   136→            occ.add(`${r + dr}-${c}`)
   137→          }
   138→        }
   139→        // Mark cells consumed by colspan (columns to the right, same row)
   140→        if (cs > 1) {
   141→          for (let dc = 1; dc < cs; dc++) {
   142→            occ.add(`${r}-${c + dc}`)
   143→          }
   144→        }
   145→      }
   146→    }
   147→    return occ
   148→  }, [config.rows])
   149→
   150→  const inRange = (r: number, c: number) => {
   151→    if (!selectionRange) return false
   152→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   153→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   154→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   155→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   156→    return r >= minR && r <= maxR && c >= minC && c <= maxC
   157→  }
   158→
   159→  const isRangeActive = selectionRange !== null
   160→  const isSingleSelected = selectedCell && !isRangeActive
   161→
   162→  return (
   163→    <div
   164→      className="overflow-auto flex-1 bg-white p-2 rounded border"
   165→      style={{ maxHeight: 'calc(100vh - 200px)' }}
   166→      onMouseUp={() => !isPreview && onCellMouseUp()}
   167→    >
   168→      <div style={{ width: '816px', height: '1344px', maxWidth: '100%', margin: '0 auto', boxShadow: '0 1px 3px rgba(0,0,0,0.12)', position: 'relative', overflow: 'hidden' }}>
   169→        <table style={{ borderCollapse: 'collapse', width: '100%', height: '100%', fontSize: '9pt', fontFamily: 'Arial, sans-serif', lineHeight: '1.2', tableLayout: 'fixed' }}>
   170→          <colgroup>
   171→            {config.columnWidths.map((w, i) => (
   172→              <col key={i} style={{ width: w || `${100 / config.totalCols}%` }} />
   173→            ))}
   174→          </colgroup>
   175→          <tbody>
   176→            {config.rows.map((gridRow, r) => {
   177→              const cells: React.ReactNode[] = []
   178→              for (let c = 0; c < config.totalCols; c++) {
   179→                const key = `${r}-${c}`
   180→                if (occupied.has(key)) continue
   181→
   182→                const cell = gridRow.cells[c] || emptyCell()
   183→
   184→                let displayContent = cell.content
   185→                if (isPreview && cell.dataBinding && displayData) {
   186→                  displayContent = resolveBinding(cell.dataBinding, displayData)
   187→                }
   188→
   189→                const borderStyle = (enabled: boolean) =>
   190→                  enabled ? `1px solid ${cell.borderColor}` : 'none'
   191→
   192→                const cellIsSelected = isSingleSelected && selectedCell?.row === r && selectedCell?.col === c
   193→                const cellInRange = isRangeActive && inRange(r, c)
   194→                const hasContent = cell.content || cell.dataBinding
   195→
   196→                cells.push(
   197→                  <td
   198→                    key={key}
   199→                    colSpan={cell.colspan > 1 ? cell.colspan : undefined}
   200→                    rowSpan={cell.rowspan > 1 ? cell.rowspan : undefined}
   201→                    style={{
   202→                      borderTop: borderStyle(cell.borderTop),
   203→                      borderRight: borderStyle(cell.borderRight),
   204→                      borderBottom: borderStyle(cell.borderBottom),
   205→                      borderLeft: borderStyle(cell.borderLeft),
   206→                      width: cell.width || undefined,
   207→                      height: cell.height || undefined,
   208→                      fontSize: `${cell.fontSize}pt`,
   209→                      fontWeight: cell.fontWeight,
   210→                      fontStyle: cell.fontStyle,
   211→                      textAlign: cell.textAlign,
   212→                      verticalAlign: cell.verticalAlign,
   213→                      color: cell.color || undefined,
   214→                      whiteSpace: cell.whiteSpace,
   215→                      padding: cell.padding,
   216→                      cursor: isPreview ? 'default' : 'cell',
   217→                      outline: cellIsSelected ? '2px solid #3b82f6' : undefined,
   218→                      outlineOffset: '-2px',
   219→                      minWidth: hasContent ? undefined : '0px',
   220→                      background: cellInRange
   221→                        ? 'rgba(59,130,246,0.15)'
   222→                        : cellIsSelected
   223→                          ? 'rgba(59,130,246,0.06)'
   224→                          : cell.bgColor || undefined,
   225→                      userSelect: 'none',
   226→                      position: 'relative',
   227→                    }}
   228→                    onMouseDown={() => !isPreview && onCellMouseDown(r, c)}
   229→                    onMouseEnter={() => !isPreview && onCellMouseEnter(r, c)}
   230→                    title={cell.dataBinding ? `[${cell.dataBinding}]` : undefined}
   231→                  >
   232→                    {!isPreview && cell.dataBinding && (
   233→                      <span style={{
   234→                        position: 'absolute', top: '1px', right: '2px',
   235→                        width: '5px', height: '5px', borderRadius: '50%',
   236→                        background: '#f97316', display: 'inline-block',
   237→                        pointerEvents: 'none',
   238→                      }} />
   239→                    )}
   240→                    {isPreview && cell.dataBinding && !displayContent ? (
   241→                      <span style={{ color: '#ccc' }}>—</span>
   242→                    ) : (
   243→                      displayContent
   244→                    )}
   245→                  </td>
   246→                )
   247→              }
   248→              return <tr key={r}>{cells}</tr>
   249→            })}
   250→          </tbody>
   251→        </table>
   252→      </div>
   253→    </div>
   254→  )
   255→}

   257→// === Column Width Editor ===
   258→function ColumnWidthEditor({
   259→  config,
   260→  onUpdate,
   261→}: {
   262→  config: GridConfig
   263→  onUpdate: (config: GridConfig) => void
   264→}) {
   265→  const [editing, setEditing] = useState<number | null>(null)
   266→  const [value, setValue] = useState('')
   267→
   268→  const startEdit = (idx: number) => {
   269→    setEditing(idx)
   270→    setValue(config.columnWidths[idx] || '')
   271→  }
   272→
   273→  const commitEdit = () => {
   274→    if (editing !== null && value.trim()) {
   275→      const newWidths = [...config.columnWidths]
   276→      newWidths[editing] = value.trim()
   277→      onUpdate({ ...config, columnWidths: newWidths })
   278→    }
   279→    setEditing(null)
   280→  }
   281→
   282→  return (
   283→    <div className="flex gap-0.5 flex-wrap p-1">
   284→      {config.columnWidths.map((w, i) => (
   285→        <div
   286→          key={i}
   287→          className="text-center border rounded px-0.5 py-0.5 min-w-[44px]"
   288→          style={{ fontSize: '7pt', backgroundColor: '#f8f8f8' }}
   289→        >
   290→          <div className="text-muted-foreground" style={{ fontSize: '6pt' }}>C{i}</div>
   291→          {editing === i ? (
   292→            <input
   293→              autoFocus
   294→              value={value}
   295→              onChange={(e) => setValue(e.target.value)}
   296→              onBlur={commitEdit}
   297→              onKeyDown={(e) => { if (e.key === 'Enter') commitEdit() }}
   298→              className="w-10 h-4 text-center text-[7pt] border rounded"
   299→              style={{ padding: '0 1px' }}
   300→            />
   301→          ) : (
   302→            <div
   303→              onClick={() => startEdit(i)}
   304→              className="cursor-pointer hover:bg-accent rounded"
   305→              title="Clic para editar"
   306→            >
   307→              {w}
   308→            </div>
   309→          )}
   310→        </div>
   311→      ))}
   312→    </div>
   313→  )
   314→}

   534→    if (!isDragging) return
   535→    setSelEnd({ row: r, col: c })
   536→  }, [isDragging])
   537→
   538→  const handleCellMouseUp = useCallback(() => {
   539→    setIsDragging(false)
   540→  }, [])
   541→
   542→  const clearRangeSelection = useCallback(() => {
   543→    setSelAnchor(null)
   544→    setSelEnd(null)
   545→    setIsDragging(false)
   546→  }, [])
   547→
   548→  const handleCellUpdate = useCallback((updates: Partial<CellConfig>) => {
   549→    if (!selectedCell) return
   550→    setGridConfig((prev) => updateCellInConfig(prev, selectedCell.row, selectedCell.col, updates))
   551→  }, [selectedCell])
   552→
   553→  const handleApplyColumns = () => {
   554→    const n = parseInt(colInput)
   555→    if (isNaN(n) || n < 1 || n > 100) {
   556→      toast({ title: 'Error', description: 'Ingrese un número válido de columnas (1-100).', variant: 'destructive' })
   557→      return
   558→    }
   559→    const newConfig: GridConfig = {
   560→      totalCols: n,
   561→      columnWidths: Array(n).fill(`${(100 / n).toFixed(2)}%`),
   562→      rows: [emptyRow(n)],
   563→    }
   564→    setGridConfig(newConfig)
   565→    setSelectedCell(null)
   566→    toast({ title: `Grilla reiniciada con ${n} columnas` })
   567→  }
   568→
   569→  const handleAddRow = () => {
   570→    setGridConfig((prev) => ({
   571→      ...prev,
   572→      rows: [...prev.rows, emptyRow(prev.totalCols)],
   573→    }))
   574→  }
   575→
   576→  const handleInsertRowAbove = () => {
   577→    if (!selectedCell) {
   578→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar fila arriba.', variant: 'destructive' })
   579→      return
   580→    }
   581→    setGridConfig((prev) => {
   582→      const newRow = emptyRow(prev.totalCols)
   583→      const newRows = [...prev.rows]
   584→
   585→      // Adjust rowspan of cells in the row above (selectedCell.row - 1)
   586→      // that span across the insertion point
   587→      const aboveIdx = selectedCell.row - 1
   588→      if (aboveIdx >= 0 && aboveIdx < prev.rows.length) {
   589→        const aboveRow = prev.rows[aboveIdx]
   590→        const adjustedCells: Record<number, CellConfig> = {}
   591→        for (const [key, cell] of Object.entries(aboveRow.cells)) {
   592→          const col = Number(key)
   593→          const rs = cell.rowspan || 1
   594→          // If this cell spans to or past the insertion row, reduce rowspan
   595→          if (col === 0 || true) { // check all cells
   596→            if (rs > 1 && (aboveIdx + rs) > selectedCell.row) {
   597→              adjustedCells[col] = { ...cell, rowspan: rs - 1 }
   598→              continue
   599→            }
   600→          }
   601→          adjustedCells[col] = cell
   602→        }
   603→        newRows[aboveIdx] = { cells: adjustedCells }
   604→      }
   605→
   606→      // Also adjust rowspan of cells in the CURRENT row (selectedCell.row)
   607→      // since it shifts down by 1, but their rowspan origin stays the same
   608→      const currentRow = prev.rows[selectedCell.row]
   609→      if (currentRow) {
   610→        const adjustedCurrent: Record<number, CellConfig> = {}
   611→        for (const [key, cell] of Object.entries(currentRow.cells)) {
   612→          const col = Number(key)
   613→          const rs = cell.rowspan || 1
   614→          // Cells that started in a row above and were part of a rowspan
   615→          // don't need adjustment - they're already handled by the row above
   616→          adjustedCurrent[col] = cell
   617→        }
   618→        // Keep the current row as-is (it just moves down)
   619→      }
   620→
   621→      newRows.splice(selectedCell.row, 0, newRow)
   622→      setSelectedCell({ row: selectedCell.row + 1, col: selectedCell.col })
   623→      return { ...prev, rows: newRows }
   624→    })
   625→  }
   626→
   627→  const handleInsertRowBelow = () => {
   628→    if (!selectedCell) {
   629→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar fila abajo.', variant: 'destructive' })
   630→      return
   631→    }
   632→    setGridConfig((prev) => {
   633→      const newRow = emptyRow(prev.totalCols)
   634→      const insertAt = selectedCell.row + 1
   635→      const newRows = [...prev.rows]
   636→
   637→      // Adjust rowspan of cells in selectedCell.row that span across insertAt
   638→      const currentRow = prev.rows[selectedCell.row]
   639→      if (currentRow) {
   640→        const adjustedCells: Record<number, CellConfig> = {}
   641→        for (const [key, cell] of Object.entries(currentRow.cells)) {
   642→          const col = Number(key)
   643→          const rs = cell.rowspan || 1
   644→          if (rs > 1 && (selectedCell.row + rs) > insertAt) {
   645→            adjustedCells[col] = { ...cell, rowspan: rs - 1 }
   646→            continue
   647→          }
   648→          adjustedCells[col] = cell
   649→        }
   650→        newRows[selectedCell.row] = { cells: adjustedCells }
   651→      }
   652→
   653→      newRows.splice(insertAt, 0, newRow)
   654→      return { ...prev, rows: newRows }
   655→    })
   656→  }
   657→
   658→  const handleDeleteLastRow = () => {
   659→    setGridConfig((prev) => {
   660→      if (prev.rows.length <= 1) {
   661→        toast({ title: 'No se puede eliminar la última fila', variant: 'destructive' })
   662→        return prev
   663→      }
   664→      const newRows = prev.rows.slice(0, -1)
   665→      // Clear selection if it was on the deleted row
   666→      if (selectedCell && selectedCell.row >= newRows.length) {
   667→        setSelectedCell(null)
   668→      }
   669→      return { ...prev, rows: newRows }
   670→    })
   671→  }
   672→
   673→  // === Column Operations ===
   674→  const handleInsertColLeft = () => {
   675→    if (!selectedCell) {
   676→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar columna a la izquierda.', variant: 'destructive' })
   677→      return
   678→    }
   679→    setGridConfig((prev) => {
   680→      const insertAt = selectedCell.col
   681→      // Shift cell indices >= insertAt in every row
   682→      const newRows = prev.rows.map((row) => {
   683→        const newCells: Record<number, CellConfig> = {}
   684→        for (const [key, cell] of Object.entries(row.cells)) {
   685→          const idx = Number(key)
   686→          if (idx >= insertAt) {
   687→            newCells[idx + 1] = cell
   688→          } else {
   689→            newCells[idx] = cell
   690→          }
   691→        }
   692→        newCells[insertAt] = emptyCell()
   693→        return { cells: newCells }
   694→      })
   695→      // Insert column width
   696→      const newWidths = [...prev.columnWidths]
   697→      newWidths.splice(insertAt, 0, '3%')
   698→      // Adjust selection
   699→      setSelectedCell({ row: selectedCell.row, col: selectedCell.col + 1 })
   700→      return { ...prev, totalCols: prev.totalCols + 1, columnWidths: newWidths, rows: newRows }
   701→    })
   702→    setColInput(String(gridConfig.totalCols + 1))
   703→  }
   704→
   705→  const handleInsertColRight = () => {
   706→    if (!selectedCell) {
   707→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar columna a la derecha.', variant: 'destructive' })
   708→      return
   709→    }
   710→    setGridConfig((prev) => {
   711→      const insertAt = selectedCell.col + 1
   712→      // Shift cell indices >= insertAt in every row
   713→      const newRows = prev.rows.map((row) => {
   714→        const newCells: Record<number, CellConfig> = {}
   715→        for (const [key, cell] of Object.entries(row.cells)) {
   716→          const idx = Number(key)
   717→          if (idx >= insertAt) {
   718→            newCells[idx + 1] = cell
   719→          } else {
   720→            newCells[idx] = cell
   721→          }
   722→        }
   723→        newCells[insertAt] = emptyCell()
   724→        return { cells: newCells }
   725→      })
   726→      // Insert column width
   727→      const newWidths = [...prev.columnWidths]
   728→      newWidths.splice(insertAt, 0, '3%')
   729→      // Selection stays on same column
   730→      return { ...prev, totalCols: prev.totalCols + 1, columnWidths: newWidths, rows: newRows }
   731→    })
   732→    setColInput(String(gridConfig.totalCols + 1))
   733→  }
   734→
   735→  const handleDeleteLastCol = () => {
   736→    setGridConfig((prev) => {
   737→      if (prev.totalCols <= 1) {
   738→        toast({ title: 'No se puede eliminar la última columna', variant: 'destructive' })
   739→        return prev
   740→      }
   741→      const lastCol = prev.totalCols - 1
   742→      const newRows = prev.rows.map((row) => {
   743→        const { [lastCol]: _removed, ...rest } = row.cells
   744→        return { cells: rest }
   745→      })
   746→      const newWidths = prev.columnWidths.slice(0, -1)
   747→      if (selectedCell && selectedCell.col >= prev.totalCols - 1) {
   748→        setSelectedCell(null)
   749→      }
   750→      return { ...prev, totalCols: prev.totalCols - 1, columnWidths: newWidths, rows: newRows }
   751→    })
   752→    setColInput(String(gridConfig.totalCols - 1))
   753→  }
   754→
   755→  // === Merge Selection (drag range) ===
   756→  const handleMergeSelection = () => {
   757→    if (!selectionRange) {
   758→      toast({ title: 'Sin selección', description: 'Selecciona varias celdas arrastrando primero.', variant: 'destructive' })
   759→      return
   760→    }
   761→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   762→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   763→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   764→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   765→    const spanCols = maxC - minC + 1
   766→    const spanRows = maxR - minR + 1
   767→    if (spanCols === 1 && spanRows === 1) {
   768→      toast({ title: 'Una sola celda', description: 'Selecciona al menos 2 celdas para combinar.', variant: 'destructive' })
   769→      return
   770→    }
   771→    setGridConfig((prev) => {
   772→      const updated = updateCellInConfig(prev, minR, minC, {
   773→        colspan: spanCols,
   774→        rowspan: spanRows,
   775→      })
   776→      const newRows = updated.rows.map((row, ri) => {
   777→        if (ri < minR || ri > maxR) return row
   778→        const newCells = { ...row.cells }
   779→        for (let ci = minC; ci <= maxC; ci++) {
   780→          if (ri === minR && ci === minC) continue
   781→          delete newCells[ci]
   782→        }
   783→        return { cells: newCells }
   784→      })
   785→      return { ...updated, rows: newRows }
   786→    })
   787→    setSelectedCell({ row: minR, col: minC })
   788→    clearRangeSelection()
   789→    toast({ title: 'Celdas combinadas', description: `${spanCols}x${spanRows} celdas fusionadas en una.` })
   790→  }
   791→
   792→  // === Merge / Split Operations ===
   793→  // Combina columnas: elimina la columna de la derecha y suma su ancho a la actual.
   794→  // El total de columnas baja (ej: 27 → 26).
   795→  const handleMergeColumns = () => {
   796→    if (!selectedCell) {
   797→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   798→      return
   799→    }
   800→    if (gridConfig.totalCols <= 1) {
   801→      toast({ title: 'No se puede', description: 'Solo queda 1 columna.', variant: 'destructive' })
   802→      return
   803→    }
   804→    const mergeCol = selectedCell.col
   805→    if (mergeCol >= gridConfig.totalCols - 1) {
   806→      toast({ title: 'No se puede', description: 'No hay columna a la derecha para combinar.', variant: 'destructive' })
   807→      return
   808→    }
   809→    setGridConfig((prev) => {
   810→      // Sumar ancho de la columna derecha a la actual
   811→      const newWidths = [...prev.columnWidths]
   812→      const leftW = parseFloat(newWidths[mergeCol]) || 0
   813→      const rightW = parseFloat(newWidths[mergeCol + 1]) || 0
   814→      newWidths[mergeCol] = `${leftW + rightW}%`
   815→      newWidths.splice(mergeCol + 1, 1)
   816→      // En cada fila, remover la columna mergeCol+1 y reindexar
   817→      const newRows = prev.rows.map((row) => {
   818→        const newCells: Record<number, CellConfig> = {}
   819→        for (const [key, cell] of Object.entries(row.cells)) {
   820→          const idx = Number(key)
   821→          if (idx === mergeCol + 1) continue // descartar columna derecha
   822→          if (idx > mergeCol + 1) {
   823→            newCells[idx - 1] = cell // reindexar
   824→          } else {
   825→            newCells[idx] = cell
   826→          }
   827→        }
   828→        return { cells: newCells }
   829→      })
   830→      // Ajustar seleccion si estaba mas alla de la columna eliminada
   831→      if (selectedCell.col > mergeCol) {
   832→        setSelectedCell({ row: selectedCell.row, col: Math.max(0, selectedCell.col - 1) })
   833→      }
   834→      const newTotal = prev.totalCols - 1
   835→      return { ...prev, totalCols: newTotal, columnWidths: newWidths, rows: newRows }
   836→    })
   837→    setColInput(String(gridConfig.totalCols - 1))
   838→  }
   839→
   840→  // Combina filas: elimina la fila de abajo y la suma a la actual.
   841→  // El total de filas baja en 1.
   842→  const handleMergeRows = () => {
   843→    if (!selectedCell) {
   844→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   845→      return
   846→    }
   847→    if (gridConfig.rows.length <= 1) {
   848→      toast({ title: 'No se puede', description: 'Solo queda 1 fila.', variant: 'destructive' })
   849→      return
   850→    }
   851→    const mergeRow = selectedCell.row
   852→    if (mergeRow >= gridConfig.rows.length - 1) {
   853→      toast({ title: 'No se puede', description: 'No hay fila debajo para combinar.', variant: 'destructive' })
   854→      return
   855→    }
   856→    setGridConfig((prev) => {
   857→      const currentRow = prev.rows[mergeRow]
   858→      const nextRow = prev.rows[mergeRow + 1]
   859→      // Fusionar contenido de celdas: si la celda de abajo tiene contenido y la actual no, copiarlo
   860→      const mergedCells: Record<number, CellConfig> = { ...currentRow.cells }
   861→      for (const [key, cell] of Object.entries(nextRow.cells)) {
   862→        const idx = Number(key)
   863→        const existing = mergedCells[idx]
   864→        if (!existing || (!existing.content && !existing.dataBinding)) {
   865→          mergedCells[idx] = { ...cell }
   866→        } else if (cell.content && !existing.content) {
   867→          mergedCells[idx] = { ...existing, content: existing.content + ' ' + cell.content }
   868→        }
   869→      }
   870→      const newRows = [...prev.rows]
   871→      newRows[mergeRow] = { cells: mergedCells }
   872→      newRows.splice(mergeRow + 1, 1)
   873→      if (selectedCell.row >= newRows.length) {
   874→        setSelectedCell(null)
   875→      }
   876→      return { ...prev, rows: newRows }
   877→    })
   878→  }
   879→
   880→  // Separar celda: reset colspan/rowspan a 1
   881→  const handleSplitCell = () => {
   882→    if (!selectedCell) {
   883→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   884→      return
   885→    }
   886→    const cell = gridConfig.rows[selectedCell.row]?.cells[selectedCell.col]
   887→    if (!cell || (cell.colspan === 1 && cell.rowspan === 1)) {
   888→      toast({ title: 'Sin combinación', description: 'Esta celda no está combinada (colspan/rowspan ya es 1).', variant: 'destructive' })
   889→      return
   890→    }
   891→    setGridConfig((prev) => updateCellInConfig(prev, selectedCell.row, selectedCell.col, { colspan: 1, rowspan: 1 }))
   892→  }
   893→
   894→  // === Save to DB (opens dialog for name) ===
   895→  const handleOpenSaveDialog = () => {
   896→    setSaveName('')
   897→    setShowSaveDialog(true)
   898→  }

   901→    if (!saveName.trim()) {
   902→      toast({ title: 'Nombre requerido', description: 'Ingresa un nombre para el layout.', variant: 'destructive' })
   903→      return
   904→    }
   905→
   906→    setSaving(true)
   907→    try {
   908→      const res = await fetch('/api/cert-layouts', {
   909→        method: 'POST',
   910→        headers: { 'Content-Type': 'application/json' },
   911→        body: JSON.stringify({
   912→          nombre: saveName.trim(),
   913→          datos: gridConfig,
   914→        }),
   915→      })
   916→
   917→      if (!res.ok) {
   918→        const err = await res.json()

   920→      }
   921→
   922→      setShowSaveDialog(false)
   923→      toast({
   924→        title: '¡Guardado exitoso!',
   925→        description: `El layout "${saveName.trim()}" se guardó correctamente en la base de datos.`,
   926→      })
   927→    } catch (error: unknown) {
   928→      const msg = error instanceof Error ? error.message : 'Error desconocido'
   929→      toast({ title: 'Error al guardar', description: msg, variant: 'destructive' })
   930→    } finally {
   931→      setSaving(false)
   932→    }
   933→  }
   934→

   936→  const handleOpenLayoutsDialog = async () => {

   938→    setLoadingLayouts(true)
   939→    try {
   940→      const res = await fetch('/api/cert-layouts')
   941→      if (res.ok) {
   942→        const data = await res.json()

   944→      }
   945→    } catch {

   947→    } finally {
   948→      setLoadingLayouts(false)
   949→    }
   950→  }
   951→
   952→  const handleLoadLayoutFromDB = async (id: string) => {
   953→    setLoadingLayout(true)
   954→    try {
   955→      const res = await fetch(`/api/cert-layouts/${id}`)
   956→      if (!res.ok) {
   957→        throw new Error('Layout no encontrado')

   959→      const layout = await res.json()

   967→        setShowLayoutsDialog(false)
   968→        toast({ title: 'Layout cargado', description: `"${layout.nombre}" se cargó correctamente.` })
   969→      } else {
   970→        throw new Error('Datos de layout inválidos')
   971→      }
   972→    } catch (error: unknown) {
   973→      const msg = error instanceof Error ? error.message : 'Error desconocido'
