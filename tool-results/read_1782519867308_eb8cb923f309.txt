    66→    66→function loadGridConfig(): GridConfig {
    67→    67→  if (typeof window === 'undefined') return createDefaultTemplate()
    68→    68→  try {
    69→    69→    const raw = localStorage.getItem(STORAGE_KEY)
    70→    70→    if (raw) {
    71→    71→      const parsed = JSON.parse(raw) as GridConfig
    72→    72→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) return patchDataBindings(parsed)
    73→    73→    }
    74→    74→  } catch { /* ignore */ }
    75→    75→  return createDefaultTemplate()
    76→    76→}

   104→   104→function GridTable({
   105→   105→  config,
   106→   106→  selectedCell,
   107→   107→  selectionRange,
   108→   108→  onCellMouseDown,
   109→   109→  onCellMouseEnter,
   110→   110→  onCellMouseUp,
   111→   111→  isPreview,
   112→   112→  displayData,
   113→   113→  onCellEdit,
   114→   114→  savingDraft,
   115→   115→}: {
   116→   116→  config: GridConfig
   117→   117→  selectedCell: { row: number; col: number } | null
   118→   118→  selectionRange: { r1: number; c1: number; r2: number; c2: number } | null
   119→   119→  onCellMouseDown: (row: number, col: number) => void
   120→   120→  onCellMouseEnter: (row: number, col: number) => void
   121→   121→  onCellMouseUp: () => void
   122→   122→  isPreview: boolean
   123→   123→  displayData: DisplayData | null
   124→   124→  onCellEdit?: (binding: string, newValue: string) => void
   125→   125→  savingDraft?: boolean
   126→   126→}) {
   127→   127→  // Recompute occupied set on every render to track rowspan AND colspan correctly
   128→   128→  const occupied = useMemo(() => {
   129→   129→    const occ = new Set<string>()
   130→   130→    for (let r = 0; r < config.rows.length; r++) {
   131→   131→      const row = config.rows[r]
   132→   132→      if (!row) continue
   133→   133→      for (const [key, cell] of Object.entries(row.cells)) {
   134→   134→        const c = Number(key)
   135→   135→        const rs = cell.rowspan || 1
   136→   136→        const cs = cell.colspan || 1
   137→   137→        // Mark cells consumed by rowspan (rows below)
   138→   138→        if (rs > 1) {
   139→   139→          for (let dr = 1; dr < rs; dr++) {
   140→   140→            occ.add(`${r + dr}-${c}`)
   141→   141→          }
   142→   142→        }
   143→   143→        // Mark cells consumed by colspan (columns to the right, same row)
   144→   144→        if (cs > 1) {
   145→   145→          for (let dc = 1; dc < cs; dc++) {
   146→   146→            occ.add(`${r}-${c + dc}`)
   147→   147→          }
   148→   148→        }
   149→   149→      }
   150→   150→    }
   151→   151→    return occ
   152→   152→  }, [config.rows])
   153→   153→
   154→   154→  const inRange = (r: number, c: number) => {
   155→   155→    if (!selectionRange) return false
   156→   156→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   157→   157→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   158→   158→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   159→   159→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   160→   160→    return r >= minR && r <= maxR && c >= minC && c <= maxC
   161→   161→  }
   162→   162→
   163→   163→  const isRangeActive = selectionRange !== null
   164→   164→  const isSingleSelected = selectedCell && !isRangeActive
   165→   165→
   166→   166→  return (
   167→   167→    <div
   168→   168→      className="overflow-auto flex-1 bg-white p-2 rounded border"
   169→   169→      style={{ maxHeight: 'calc(100vh - 200px)' }}
   170→   170→      onMouseUp={() => !isPreview && onCellMouseUp()}
   171→   171→    >
   172→   172→      <div style={{ width: '816px', height: '1344px', maxWidth: '100%', margin: '0 auto', boxShadow: '0 1px 3px rgba(0,0,0,0.12)', position: 'relative', overflow: 'hidden' }}>
   173→   173→        <table style={{ borderCollapse: 'collapse', width: '100%', height: '100%', fontSize: '9pt', fontFamily: 'Arial, sans-serif', lineHeight: '1.2', tableLayout: 'fixed' }}>
   174→   174→          <colgroup>
   175→   175→            {config.columnWidths.map((w, i) => (
   176→   176→              <col key={i} style={{ width: w || `${100 / config.totalCols}%` }} />
   177→   177→            ))}
   178→   178→          </colgroup>
   179→   179→          <tbody>
   180→   180→            {config.rows.map((gridRow, r) => {
   181→   181→              const cells: React.ReactNode[] = []
   182→   182→              for (let c = 0; c < config.totalCols; c++) {
   183→   183→                const key = `${r}-${c}`
   184→   184→                if (occupied.has(key)) continue
   185→   185→
   186→   186→                const cell = gridRow.cells[c] || emptyCell()
   187→   187→
   188→   188→                let displayContent = cell.content
   189→   189→                if (isPreview && cell.dataBinding && displayData) {
   190→   190→                  displayContent = resolveBinding(cell.dataBinding, displayData)
   191→   191→                }
   192→   192→
   193→   193→                const borderStyle = (enabled: boolean) =>
   194→   194→                  enabled ? `1px solid ${cell.borderColor}` : 'none'
   195→   195→
   196→   196→                const cellIsSelected = isSingleSelected && selectedCell?.row === r && selectedCell?.col === c
   197→   197→                const cellInRange = isRangeActive && inRange(r, c)
   198→   198→                const hasContent = cell.content || cell.dataBinding
   199→   199→
   200→   200→                cells.push(
   201→   201→                  <td
   202→   202→                    key={key}
   203→   203→                    colSpan={cell.colspan > 1 ? cell.colspan : undefined}
   204→   204→                    rowSpan={cell.rowspan > 1 ? cell.rowspan : undefined}
   205→   205→                    style={{
   206→   206→                      borderTop: borderStyle(cell.borderTop),
   207→   207→                      borderRight: borderStyle(cell.borderRight),
   208→   208→                      borderBottom: borderStyle(cell.borderBottom),
   209→   209→                      borderLeft: borderStyle(cell.borderLeft),
   210→   210→                      width: cell.width || undefined,
   211→   211→                      height: cell.height || undefined,
   212→   212→                      fontSize: `${cell.fontSize}pt`,
   213→   213→                      fontWeight: cell.fontWeight,
   214→   214→                      fontStyle: cell.fontStyle,
   215→   215→                      textAlign: cell.textAlign,
   216→   216→                      verticalAlign: cell.verticalAlign,
   217→   217→                      color: cell.color || undefined,
   218→   218→                      whiteSpace: cell.whiteSpace,
   219→   219→                      padding: cell.padding,
   220→   220→                      cursor: isPreview ? 'default' : 'cell',
   221→   221→                      outline: cellIsSelected ? '2px solid #3b82f6' : undefined,
   222→   222→                      outlineOffset: '-2px',
   223→   223→                      minWidth: hasContent ? undefined : '0px',
   224→   224→                      background: cellInRange
   225→   225→                        ? 'rgba(59,130,246,0.15)'
   226→   226→                        : cellIsSelected
   227→   227→                          ? 'rgba(59,130,246,0.06)'
   228→   228→                          : cell.bgColor || undefined,
   229→   229→                      userSelect: 'none',
   230→   230→                      position: 'relative',
   231→   231→                    }}
   232→   232→                    onMouseDown={() => !isPreview && onCellMouseDown(r, c)}
   233→   233→                    onMouseEnter={() => !isPreview && onCellMouseEnter(r, c)}
   234→   234→                    title={cell.dataBinding ? `[${cell.dataBinding}]` : undefined}
   235→   235→                    onDoubleClick={(e) => {
   236→   236→                      if (isPreview && cell.dataBinding && onCellEdit) {
   237→   237→                        e.stopPropagation()
   238→   238→                        const target = e.currentTarget.querySelector('[data-editable]') as HTMLElement | null
   239→   239→                        if (target) {
   240→   240→                          target.setAttribute('contentEditable', 'true')
   241→   241→                          target.focus()
   242→   242→                          // Select all text
   243→   243→                          const range = document.createRange()
   244→   244→                          range.selectNodeContents(target)
   245→   245→                          const sel = window.getSelection()
   246→   246→                          sel?.removeAllRanges()
   247→   247→                          sel?.addRange(range)
   248→   248→                        }
   249→   249→                      }
   250→   250→                    }}
   251→   251→                  >
   252→   252→                    {!isPreview && cell.dataBinding && (
   253→   253→                      <span style={{
   254→   254→                        position: 'absolute', top: '1px', right: '2px',
   255→   255→                        width: '5px', height: '5px', borderRadius: '50%',
   256→   256→                        background: '#f97316', display: 'inline-block',
   257→   257→                        pointerEvents: 'none',
   258→   258→                      }} />
   259→   259→                    )}
   260→   260→                    {isPreview && cell.dataBinding && !displayContent ? (
   261→   261→                      <span style={{ color: '#ccc' }}>—</span>
   262→   262→                    ) : isPreview && cell.dataBinding && onCellEdit ? (
   263→   263→                      <span
   264→   264→                        data-editable
   265→   265→                        style={{ cursor: 'text', outline: 'none', minWidth: '20px', display: 'inline-block', minHeight: '1em' }}
   266→   266→                        onBlur={(e) => {
   267→   267→                          e.currentTarget.setAttribute('contentEditable', 'false')
   268→   268→                          const newVal = e.currentTarget.textContent || ''
   269→   269→                          if (newVal !== resolveBinding(cell.dataBinding!, displayData!)) {
   270→   270→                            onCellEdit(cell.dataBinding, newVal)
   271→   271→                          }
   272→   272→                        }}
   273→   273→                        onKeyDown={(e) => {
   274→   274→                          if (e.key === 'Enter' && !e.shiftKey) {
   275→   275→                            e.preventDefault()
   276→   276→                            e.currentTarget.blur()
   277→   277→                          }
   278→   278→                          if (e.key === 'Escape') {
   279→   279→                            e.currentTarget.blur()
   280→   280→                          }
   281→   281→                        }}
   282→   282→                        suppressContentEditableWarning
   283→   283→                      >
   284→   284→                        {displayContent}
   285→   285→                      </span>
   286→   286→                    ) : displayContent.startsWith('##LOGO_') && displayContent.endsWith('##') ? (
   287→   287→                      <img
   288→   288→                        src="/logo-gob-mppe.png"
   289→   289→                        alt="Logo Gobierno Bolivariano"
   290→   290→                        style={{ maxWidth: '100%', maxHeight: '100%', width: '100%', height: 'auto', objectFit: 'contain', display: 'block' }}
   291→   291→                      />
   292→   292→                    ) : (
   293→   293→                      displayContent
   294→   294→                    )}
   295→   295→                  </td>
   296→   296→                )
   297→   297→              }
   298→   298→              return <tr key={r}>{cells}</tr>
   299→   299→            })}
   300→   300→          </tbody>
   301→   301→        </table>
   302→   302→      </div>
   303→   303→    </div>
   304→   304→  )
   305→   305→}

   308→   308→function ColumnWidthEditor({
   309→   309→  config,
   310→   310→  onUpdate,
   311→   311→}: {
   312→   312→  config: GridConfig
   313→   313→  onUpdate: (config: GridConfig) => void
   314→   314→}) {
   315→   315→  const [editing, setEditing] = useState<number | null>(null)
   316→   316→  const [value, setValue] = useState('')
   317→   317→
   318→   318→  const startEdit = (idx: number) => {
   319→   319→    setEditing(idx)
   320→   320→    setValue(config.columnWidths[idx] || '')
   321→   321→  }
   322→   322→
   323→   323→  const commitEdit = () => {
   324→   324→    if (editing !== null && value.trim()) {
   325→   325→      const newWidths = [...config.columnWidths]
   326→   326→      newWidths[editing] = value.trim()
   327→   327→      onUpdate({ ...config, columnWidths: newWidths })
   328→   328→    }
   329→   329→    setEditing(null)
   330→   330→  }
   331→   331→
   332→   332→  return (
   333→   333→    <div className="flex gap-0.5 flex-wrap p-1">
   334→   334→      {config.columnWidths.map((w, i) => (
   335→   335→        <div
   336→   336→          key={i}
   337→   337→          className="text-center border rounded px-0.5 py-0.5 min-w-[44px]"
   338→   338→          style={{ fontSize: '7pt', backgroundColor: '#f8f8f8' }}
   339→   339→        >

   350→   350→            />

   359→   359→          )}

   993→   993→        }
   994→   994→      }
   995→   995→      const newRows = [...prev.rows]
   996→   996→      newRows[mergeRow] = { cells: mergedCells }
