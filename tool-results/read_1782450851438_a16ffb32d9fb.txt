   147→const emptyCertData = (planTipo?: string): CertData => {
   148→  const activePlan = getActivePlan(planTipo)
   149→  const calificaciones: Record<string, CalificacionRow[]> = {}
   150→  activePlan.forEach(plan => {
   151→    calificaciones[plan.anio] = plan.materias.map(m => ({
   152→      materia: m.nombre, numero: m.numero, nota: '', literal: '',
   153→      tipoEvaluacion: '', fechaMes: '', fechaAnio: '', instEduc: '',
   154→    }))
   155→  })
   156→
   157→  return {
   158→    lugar: schoolConfig.estado,
   159→    fechaExpedicion: new Date().toISOString().split('T')[0], // YYYY-MM-DD para input type="date"
   160→    planEstudio: planTipo === 'derogado'
   161→      ? 'EDUCACIÓN MEDIA GENERAL (PLAN DEROGADO)'
   162→      : schoolConfig.planEstudio,
   163→    od: schoolConfig.od,
   164→    denominacion: schoolConfig.nombreCompleto,
   165→    direccion: schoolConfig.direccion,
   166→    telefono: schoolConfig.telefono,
   167→    municipio: schoolConfig.municipio,
   168→    estado: schoolConfig.estado,
   169→    cdcce: schoolConfig.cdcceEstado,
   170→    estudiante: { cedula: '', fechaNacimiento: '', apellidos: '', nombres: '', pais: 'VENEZUELA', estado: '', municipio: '' },
   171→    instituciones: Array.from({ length: 5 }, (_, i) => ({ numero: i + 1, denominacion: '', localidad: '', ef: '' })),
   172→    calificaciones,
   173→    orientacion: Array.from({ length: 5 }, () => ({ anio: '', literal: '' })),
   174→    grupos: Array.from({ length: 5 }, () => ({ anio: '', grupo: '', literal: '' })),
   175→    observaciones: '',
   176→    promedioAcumulado: '',
   177→    director: { apellidosNombres: schoolConfig.director.apellidosNombres, cedula: schoolConfig.director.cedula },
   178→    directorCdcce: { apellidosNombres: schoolConfig.directorCdcce.apellidosNombres, cedula: schoolConfig.directorCdcce.cedula },
   179→    planTipo: planTipo || 'vigente',
   180→    aniosEscolares: [],
   181→  }
   182→}

   186→export default function CertificacionesPage() {
   187→  const [selectedStudent, setSelectedStudent] = useState<Student | null>(null)
   188→  const [certifications, setCertifications] = useState<Certification[]>([])
   189→  const [previewCert, setPreviewCert] = useState<Certification | null>(null)
   190→  const [loading, setLoading] = useState(false)
   191→  const [generating, setGenerating] = useState(false)
   192→  const [savingTab, setSavingTab] = useState<string | null>(null)
   193→  const [lastSavedTab, setLastSavedTab] = useState<string | null>(null)
   194→  const [loadingData, setLoadingData] = useState(false)
   195→  const [dataLoaded, setDataLoaded] = useState(false)
   196→  const [activeTab, setActiveTab] = useState('calificaciones')
   197→  const { toast } = useToast()
   198→
   199→  const [certData, setCertData] = useState<CertData>(emptyCertData())
   200→  const [loadedData, setLoadedData] = useState<CertData | null>(null)
   201→
   202→  // Plan activo basado en los datos cargados
   203→  const activePlan = getActivePlan(certData.planTipo)
   204→
   205→  // Fetch historial de certificaciones del estudiante
   206→  const fetchCertifications = async (studentId: string) => {
   207→    setLoading(true)
   208→    try {
   209→      const res = await fetch(`/api/certifications/${studentId}`)
   210→      if (!res.ok) { setCertifications([]); return }
   211→      const data = await res.json()
   212→      setCertifications(Array.isArray(data) ? data.filter((c: Certification) => c.tipo === 'CERTIFICACION_CALIFICACIONES') : [])
   213→    } catch { setCertifications([]) }
   214→    finally { setLoading(false) }
   215→  }
   216→
   217→  // Cargar datos de calificaciones del rawData del estudiante
   218→  const fetchCertDataFromDB = async (student: Student) => {
   219→    setLoadingData(true)
   220→    setDataLoaded(false)
   221→    try {
   222→      const res = await fetch(`/api/students/${student.id}/cert-data`)
   223→      if (!res.ok) {
   224→        const errorData = await res.json().catch(() => ({}))
   225→        console.error('cert-data API error:', res.status, errorData)
   226→        const reason = errorData.reason || ''
   227→        const detailMsg = reason === 'empty_rawData'
   228→          ? `El alumno ${student.cedula} no tiene rawData (datos de calificaciones) en la base de datos. Se requiere re-importar los datos.`
   229→          : reason === 'parse_error'
   230→            ? `Error al parsear rawData del alumno ${student.cedula} (longitud: ${errorData.rawDataLength || '?'}).`
   231→            : `No se encontraron datos de calificaciones para ${student.cedula}.`
   232→        toast({ title: 'Sin datos de calificaciones', description: detailMsg, variant: 'destructive' })
   233→        setLoadingData(false)
   234→        return
   235→      }
   236→      const result = await res.json()
   237→
   238→      if (result.certData) {
   239→        // Si el estudiante tiene rawData parseado, usar esos datos
   240→        const cd = result.certData as CertData
   241→        // Si la fecha de expedición está vacía, usar la fecha actual
   242→        if (!cd.fechaExpedicion || cd.fechaExpedicion.trim() === '') {
   243→          cd.fechaExpedicion = new Date().toISOString().split('T')[0]
   244→        }
   245→        // Si la fecha viene en DD/MM/YYYY, convertir a YYYY-MM-DD para el input
   246→        if (/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(cd.fechaExpedicion)) {
   247→          const parts = cd.fechaExpedicion.split('/')
   248→          cd.fechaExpedicion = `${parts[2]}-${parts[1].padStart(2, '0')}-${parts[0].padStart(2, '0')}`
   249→        }
   250→        setCertData(cd)
   251→        setDataLoaded(true)
   252→
   253→        const planLabel = result.certData.planTipo === 'derogado' ? 'Plan Derogado (BD2)' : 'Plan Vigente (BD)'
   254→        const gradeCount = result.gradeCount || Object.values(result.certData.calificaciones || {})
   255→          .flat()
   256→          .filter((c: CalificacionRow) => c.nota && c.nota !== '').length
   257→
   258→        toast({
   259→          title: `Datos cargados — ${planLabel}`,
   260→          description: `Se cargaron ${gradeCount} calificaciones del rawData. Verifica y edita los datos en las pestañas.`,
   261→        })
   262→        setActiveTab('calificaciones')
   263→      }
   264→    } catch (err) {
   265→      console.error('Error fetching cert data:', err)
   266→      toast({ title: 'Error', description: 'Error al cargar datos de calificaciones. Verifique su conexión e intente de nuevo.', variant: 'destructive' })
   267→    } finally {
   268→      setLoadingData(false)
   269→    }
   270→  }
   271→
   272→  const handleSelectStudent = async (student: Student) => {
   273→    setSelectedStudent(student)
   274→    setPreviewCert(null)
   275→    setLoadedData(null)
   276→    setDataLoaded(false)
   277→    setLastSavedTab(null)
   278→    fetchCertifications(student.id)
   279→
   280→    // Normalizar fecha de nacimiento a DD/MM/YYYY
   281→    const normalizeFecha = (f: string | null | undefined): string => {
   282→      if (!f) return ''
   283→      const trimmed = String(f).trim()
   284→      if (!trimmed) return ''
   285→      if (/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(trimmed)) {
   286→        const parts = trimmed.split('/')
   287→        return `${parts[0].padStart(2, '0')}/${parts[1].padStart(2, '0')}/${parts[2]}`
   288→      }
   289→      if (trimmed.includes('T') || /^\d{4}-\d{2}-\d{2}/.test(trimmed)) {
   290→        try {
   291→          const d = new Date(trimmed)
   292→          if (!isNaN(d.getTime())) {
   293→            const day = String(d.getDate()).padStart(2, '0')
   294→            const month = String(d.getMonth() + 1).padStart(2, '0')
   295→            const year = d.getFullYear()
   296→            return `${day}/${month}/${year}`
   297→          }
   298→        } catch { /* ignore */ }
   299→      }
   300→      return trimmed
   301→    }
   302→
   303→    // Iniciar con datos vacíos pero del plan correcto
   304→    const baseData = emptyCertData(student.plan === 'derogado' ? 'derogado' : 'vigente')
   305→    baseData.estudiante = {
   306→      cedula: formatCedulaFinal(student.cedula),
   307→      fechaNacimiento: normalizeFecha(student.fechaNacimiento),
   308→      apellidos: student.apellidos,
   309→      nombres: student.nombres,
   310→      pais: student.pais || 'VENEZUELA',
   311→      estado: student.estado || '',
   312→      municipio: student.municipio || '',
   313→    }
   314→    baseData.planTipo = student.plan === 'derogado' ? 'derogado' : 'vigente'
   315→    setCertData(baseData)
   316→
   317→    // Intentar cargar borrador guardado primero
   318→    const draft = await loadDraft(student.id)
   319→    if (draft) {
   320→      // Asegurar que los campos del estudiante estén actualizados con los datos del modelo
   321→      draft.estudiante = {
   322→        ...draft.estudiante,
   323→        cedula: formatCedulaFinal(student.cedula),
   324→        apellidos: student.apellidos,
   325→        nombres: student.nombres,
   326→      }
   327→      setCertData(draft)
   328→      setDataLoaded(true)
   329→      setActiveTab('calificaciones')
   330→      toast({
   331→        title: 'Borrador recuperado',
   332→        description: 'Se cargó la última versión guardada de los datos editados.',
   333→      })
   334→    } else {
   335→      // Si no hay borrador, cargar del rawData como antes
   336→      fetchCertDataFromDB(student)
   337→    }
   338→  }
   339→
   340→  const updateNota = (anio: string, matIndex: number, field: keyof CalificacionRow, value: string) => {

   347→        updated.calificaciones[anio][matIndex].literal = notaEnLetras(value)

   350→    })
   351→  }

   358→      return updated
   359→    })
   360→  }

   366→      updated.orientacion[index] = { ...updated.orientacion[index], [field]: value }
   367→      return updated
   368→    })

   371→  const updateGrupo = (index: number, field: keyof GrupoRow, value: string) => {

   373→      const updated = { ...prev }
   374→      updated.grupos = [...prev.grupos]

   383→    try {
   384→      const res = await fetch('/api/certifications', {
   385→        method: 'POST',

   396→      toast({ title: 'Error', description: 'Error al generar certificación', variant: 'destructive' })

   400→  const handleViewCert = (cert: Certification) => {
   401→    try {
   402→      const data = JSON.parse(cert.datos || '{}')
   403→      // Corregir literales viejos: IN→INASISTENTE, PE→PENDIENTE
   404→      if (data.calificaciones) {
   405→        for (const anio of Object.keys(data.calificaciones)) {
   406→          for (const cal of data.calificaciones[anio]) {
   407→            if (cal.nota === 'IN' && cal.literal !== 'INASISTENTE') cal.literal = 'INASISTENTE'
   408→            if (cal.nota === 'PE' && cal.literal !== 'PENDIENTE') cal.literal = 'PENDIENTE'
   409→            // Si el literal está vacío pero la nota tiene valor, regenerar
   410→            if (!cal.literal && cal.nota) cal.literal = notaEnLetras(cal.nota)
   411→          }
   412→        }
   413→      }
   414→      setLoadedData(data)
   415→      setPreviewCert(cert)
   416→      setActiveTab('vista')
   417→    } catch {
   418→      toast({ title: 'Error', description: 'Error al cargar datos de certificación', variant: 'destructive' })
   419→    }
   420→  }
   421→
   422→  // Guardar borrador del certData completo en la base de datos
   423→  const handleSaveTab = async (tabName: string) => {
   424→    if (!selectedStudent) return
   425→    setSavingTab(tabName)
   426→    setLastSavedTab(null)

   428→      const res = await fetch(`/api/students/${selectedStudent.id}/cert-draft`, {
   429→        method: 'PUT',
   430→        headers: { 'Content-Type': 'application/json' },
   431→        body: JSON.stringify({ datos: certData }),
   432→      })
   433→      if (!res.ok) throw new Error('Error al guardar')
   434→      setLastSavedTab(tabName)
   435→      toast({ title: 'Borrador guardado', description: `Los cambios de la pestaña "${tabName}" se guardaron correctamente.` })
   436→    } catch {
   437→      toast({ title: 'Error', description: 'No se pudo guardar el borrador. Intente de nuevo.', variant: 'destructive' })
   438→    } finally {
   439→      setSavingTab(null)
   440→    }
   441→  }

   443→  // Cargar borrador guardado previamente
   444→  const loadDraft = async (studentId: string): Promise<CertData | null> => {
   445→    try {
   446→      const res = await fetch(`/api/students/${studentId}/cert-draft`)
   447→      if (!res.ok) return null
   448→      const { draft } = await res.json()
   449→      return draft as CertData | null
   450→    } catch {
   451→      return null
   452→    }
   453→  }

   477→  // Convertir YYYY-MM-DD → DD/MM/YYYY para mostrar en la vista de impresión
   478→  const displayFechaExpedicion = (() => {
