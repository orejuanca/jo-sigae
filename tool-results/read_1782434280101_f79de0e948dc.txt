     3→export interface CellConfig {
     4→  content: string           // Static text label (e.g., "Código:")
     5→  dataBinding: string       // Dot-path to data field (e.g., "student.cedula", "school.denominacion")
     6→  colspan: number           // Default 1
     7→  rowspan: number           // Default 1
     8→  width: string             // e.g., '10%', '100px', '' (auto)
     9→  height: string            // e.g., '20px', 'auto'
    10→  fontSize: number          // Default 9
    11→  fontWeight: 'normal' | 'bold'
    12→  textAlign: 'left' | 'center' | 'right'
    13→  verticalAlign: 'top' | 'middle' | 'bottom'
    14→  borderTop: boolean        // Default true
    15→  borderRight: boolean      // Default true
    16→  borderBottom: boolean     // Default true
    17→  borderLeft: boolean       // Default true
    18→  borderColor: string       // Default '#000000'
    19→  bgColor: string           // Default '' (transparent)
    20→  color: string             // Default '' (inherit)
    21→  whiteSpace: 'normal' | 'nowrap' | 'pre'
    22→  padding: string           // Default '1px 2px'
    23→  fontStyle: 'normal' | 'italic'
    24→}
    25→

    36→// Factory for empty cell with defaults
    37→export function emptyCell(overrides?: Partial<CellConfig>): CellConfig {
    38→  return {
    39→    content: '', dataBinding: '', colspan: 1, rowspan: 1,
    40→    width: '', height: '', fontSize: 9, fontWeight: 'normal',
    41→    textAlign: 'left', verticalAlign: 'middle',
    42→    borderTop: true, borderRight: true, borderBottom: true, borderLeft: true,
    43→    borderColor: '#000000', bgColor: '', color: '', whiteSpace: 'normal',
    44→    padding: '1px 2px', fontStyle: 'normal',
    45→    ...overrides,
    46→  }
    47→}

    49→// Factory for empty row
    50→export function emptyRow(totalCols: number): GridRow {
    51→  const cells: Record<number, CellConfig> = {}
    52→  for (let c = 0; c < totalCols; c++) {
    53→    cells[c] = emptyCell()
    54→  }
    55→  return { cells }
    56→}

    70→// Generar bindings de calificaciones por año
    71→function generateCalifBindings() {
    72→  const groups: { group: string; bindings: { value: string; label: string }[] }[] = []
    73→  for (let y = 1; y <= 5; y++) {
    74→    const materias = MATERIAS_VIGENTE[y] || []
    75→    const bindings: { value: string; label: string }[] = []
    76→    for (let s = 0; s < materias.length; s++) {
    77→      const nombre = materias[s]
    78→      bindings.push(
    79→        { value: `calif.${y}.${s}.materia`, label: `${s + 1}. ${nombre}` },
    80→        { value: `calif.${y}.${s}.nota`, label: `${s + 1}. ${nombre} — Nota` },
    81→        { value: `calif.${y}.${s}.literal`, label: `${s + 1}. ${nombre} — Literal` },
    82→        { value: `calif.${y}.${s}.te`, label: `${s + 1}. ${nombre} — T-E` },
    83→        { value: `calif.${y}.${s}.mes`, label: `${s + 1}. ${nombre} — Mes` },
    84→        { value: `calif.${y}.${s}.anio`, label: `${s + 1}. ${nombre} — Año` },
    85→        { value: `calif.${y}.${s}.inst`, label: `${s + 1}. ${nombre} — Inst. Educ.` },
    86→        { value: `calif.${y}.${s}.numero`, label: `${s + 1}. ${nombre} — N°` },
    87→      )
    88→    }
    89→    groups.push({ group: `Calificaciones — ${YEAR_NAMES[y - 1]}`, bindings })
    90→  }
    91→  return groups
    92→}

   215→// === Data Resolution for Preview ===
   216→const YEAR_NAME_MAP: Record<string, string> = {
   217→  '1': 'Primer Año', '2': 'Segundo Año', '3': 'Tercer Año',
   218→  '4': 'Cuarto Año', '5': 'Quinto Año',
   219→}

   221→export function resolveBinding(path: string, data: DisplayData): string {
   222→  if (!path || !data) return ''
   223→  const [domain, ...rest] = path.split('.')
   224→  switch (domain) {
   225→    case 'student': return rest.reduce((o: any, k: string) => o?.[k], data.estudiante) || ''
   226→    case 'school': {
   227→      const schoolMap: Record<string, string> = {
   228→        codigo: data.od, denominacion: data.denominacion, direccion: data.direccion,
   229→        telefono: data.telefono, municipio: data.municipio, estado: data.estado, cdcce: data.cdcce
   230→      }
   231→      return schoolMap[rest[0]] || ''
   232→    }
   233→    case 'doc': {
   234→      // Soporte para doc.literalFinal.N
   235→      if (rest[0] === 'literalFinal') {
   236→        const idx = parseInt(rest[1])
   237→        return data.literalesFinales?.[idx] || ''
   238→      }
   239→      const docMap: Record<string, string> = {
   240→        planEstudio: data.planEstudio, codigo: data.planCodigo,
   241→        lugar: data.lugar, fechaExpedicion: data.fechaExpedicion,
   242→        observaciones: data.observaciones, promedioAcumulado: data.promedioAcumulado,
   243→        acta: data.acta || '', actaFecha: data.actaFecha || '', actaAnio: data.actaAnio || '',
   244→      }
   245→      return docMap[rest[0]] || ''
   246→    }
   247→    case 'director':
   248→      return rest[0] === 'nombre' ? (data.director?.apellidosNombres || '') : (data.director?.cedula || '')
   249→    case 'cdcee':
   250→      return rest[0] === 'nombre' ? (data.directorCdcce?.apellidosNombres || '') : (data.directorCdcce?.cedula || '')
   251→    case 'inst': {
   252→      const idx = parseInt(rest[0])
   253→      const field = rest[1]
   254→      return data.instituciones[idx]?.[field as keyof InstitucionEducativa] || '*'
   255→    }
   256→    case 'calif': {
   257→      const yearKey = YEAR_NAME_MAP[rest[0]] || rest[0]
   258→      const idx = parseInt(rest[1])
   259→      const field = rest[2]
   260→      const yearData = data.calificaciones?.[yearKey]
   261→      if (!yearData) return ''
   262→      const subj = yearData[idx]
   263→      if (!subj) return ''
   264→      const map: Record<string, string> = {
   265→        materia: subj.materia, numero: String(subj.numero),
   266→        nota: subj.nota, literal: subj.literal, te: subj.tipoEvaluacion,
   267→        mes: subj.fechaMes, anio: subj.fechaAnio, inst: subj.instEduc,
   268→      }
   269→      return map[field] || ''
   270→    }
   271→    case 'orient': {
   272→      const idx = parseInt(rest[0])
   273→      const entry = data.orientacion?.[idx]
   274→      if (!entry) return ''
   275→      if (rest[1] === 'literal') return entry.literal
   276→      return entry.anio || ''
   277→    }
   278→    case 'grupo': {
   279→      const idx = parseInt(rest[0])
   280→      const entry = data.grupos?.[idx]
   281→      if (!entry) return ''
   282→      if (rest[1] === 'literal') return entry.literal
   283→      if (rest[1] === 'grupo') return entry.grupo
   284→      return entry.anio || ''
   285→    }
   286→    case 'obsLine': {
   287→      const idx = parseInt(rest[0])
   288→      return data.observacionesLines?.[idx] || ''
   289→    }
   290→    default: return ''
   291→  }
   292→}

   294→// === Helpers for building template ===
   295→function c(content: string, dataBinding?: string, overrides?: Partial<CellConfig>): CellConfig {
   296→  return emptyCell({ content, dataBinding, ...overrides })
   297→}

   299→function row(cells: [number, CellConfig][]): GridRow {
   300→  const allCells: Record<number, CellConfig> = {}
   301→  cells.forEach(([col, cell]) => { allCells[col] = cell })
   302→  return { cells: allCells }
   303→}
   304→
   305→// Bold cell shorthand
   306→function b(content: string, dataBinding?: string, overrides?: Partial<CellConfig>): CellConfig {
   307→  return emptyCell({ content, dataBinding, fontWeight: 'bold', ...overrides })
   308→}
   309→
   310→// Header cell (bold, centered, middle)
   311→function h(content: string, overrides?: Partial<CellConfig>): CellConfig {
   312→  return emptyCell({ content, fontWeight: 'bold', textAlign: 'center', verticalAlign: 'middle', ...overrides })
   313→}
