   165→   165→// Check if a value is an asterisk placeholder (empty slot in the spreadsheet)
   166→   166→function isAsterisk(val: string | undefined): boolean {
   167→   167→  if (!val) return true
   168→   168→  const v = String(val).trim()
   169→   169→  return v === '' || /^\*+$/.test(v) || /^\*\s+\*/.test(v)
   170→   170→}
   171→   171→
   172→   172→// Check if a nota value represents a valid grade
   173→   173→function isValidGrade(val: string | undefined): boolean {
   174→   174→  if (!val) return false
   175→   175→  const v = String(val).trim()
   176→   176→  if (isAsterisk(v)) return false
   177→   177→  if (/^\d{1,2}$/.test(v)) {
   178→   178→    const n = parseInt(v, 10)
   179→   179→    return n >= 1 && n <= 20
   180→   180→  }
   181→   181→  return ['PE', 'IN', 'EX'].includes(v.toUpperCase())
   182→   182→}

   184→   184→function cleanSchoolName(val: string): string {
   185→   185→  return String(val).replace(/^\*/, '').trim()
   186→   186→}

   188→   188→// Find the acta key by scanning backwards from the end of the data
   189→   189→function findActaKey(rawData: Record<string, string>, startSearch: number, endSearch: number): string {
   190→   190→  for (let k = endSearch; k >= startSearch; k--) {
   191→   191→    const val = rawData[String(k)]
   192→   192→    if (val && !isAsterisk(val)) {
   193→   193→      return val.trim()
   194→   194→    }
   195→   195→  }
   196→   196→  return ''
   197→   197→}

   214→   214→function parseStructuredVigente(data: Record<string, unknown>): ParsedCertData {
   215→   215→  const result: ParsedCertData = {
   216→   216→    plan: 'vigente',
   217→   217→    acta: String(data['acta'] || '').trim(),
   218→   218→    actaFecha: formatDateVal(String(data['actaFecha'] || '')),
   219→   219→    actaAnio: String(data['actaAnio'] || '').trim(),
   220→   220→    instituciones: [],
   221→   221→    calificaciones: {},
   222→   222→    aniosEscolares: [],
   223→   223→    orientacion: [],
   224→   224→    grupos: [],
   225→   225→    especializaciones: [],
   226→   226→    observaciones: [],
   227→   227→    observacionCompleta: '',
   228→   228→    literalesFinales: [],
   229→   229→  }
   230→   230→
   231→   231→  // Instituciones
   232→   232→  const insts = data['instituciones'] as Array<{ denominacion: string; localidad: string; ef: string }> | undefined
   233→   233→  if (Array.isArray(insts)) {
   234→   234→    result.instituciones = insts.map((inst, i) => ({
   235→   235→      numero: i + 1,
   236→   236→      denominacion: inst.denominacion,
   237→   237→      localidad: inst.localidad,
   238→   238→      ef: inst.ef,
   239→   239→    }))
   240→   240→  }
   241→   241→
   242→   242→  // Calificaciones — agrupar por anioEscolar
   243→   243→  const grades = data['calificaciones'] as StructuredGrade[] | undefined
   244→   244→  if (Array.isArray(grades)) {
   245→   245→    const gradesByYear: Record<number, StructuredGrade[]> = {}
   246→   246→    for (const g of grades) {
   247→   247→      const y = g.anioEscolar || 0
   248→   248→      if (!gradesByYear[y]) gradesByYear[y] = []
   249→   249→      gradesByYear[y].push(g)
   250→   250→    }
   251→   251→
   252→   252→    const yearNames = ['Primer Año', 'Segundo Año', 'Tercer Año', 'Cuarto Año', 'Quinto Año']
   253→   253→    const sortedYears = Object.keys(gradesByYear).map(Number).sort((a, b) => a - b)
   254→   254→
   255→   255→    sortedYears.forEach((yearNum, yearIdx) => {
   256→   256→      if (yearIdx >= 5) return
   257→   257→      const yearName = yearNames[yearIdx]
   258→   258→      const yearGrades = gradesByYear[yearNum]
   259→   259→      const planIdx = Math.min(yearIdx, planEMG.length - 1)
   260→   260→      const subjects = planEMG[planIdx].materias
   261→   261→
   262→   262→      // Recoger años escolares únicos de las notas
   263→   263→      const aniosSet = new Set<string>()
   264→   264→
   265→   265→      result.calificaciones[yearName] = yearGrades.map((g, sIdx) => {
   266→   266→        const literal = notaEnLetras(g.nota)
   267→   267→        const subjectIndex = sIdx % subjects.length
   268→   268→        const materia = subjects[subjectIndex]?.nombre || g.materia || `Materia ${sIdx + 1}`
   269→   269→
   270→   270→        if (g.anio) aniosSet.add(g.anio)
   271→   271→
   272→   272→        return {
   273→   273→          materia,
   274→   274→          numero: sIdx + 1,
   275→   275→          nota: g.nota,
   276→   276→          literal,
   277→   277→          tipoEvaluacion: g.eval || '',
   278→   278→          fechaMes: parseMes(g.mes),
   279→   279→          fechaAnio: g.anio,
   280→   280→          instEduc: g.inst && !isAsterisk(g.inst) ? g.inst : '',
   281→   281→        }
   282→   282→      })
   283→   283→
   284→   284→      aniosSet.forEach(a => {
   285→   285→        if (!result.aniosEscolares.includes(a)) result.aniosEscolares.push(a)
   286→   286→      })
   287→   287→    })
   288→   288→  }
   289→   289→
   290→   290→  // Orientación
   291→   291→  const orient = data['orientacion'] as Array<{ anio: string; literal: string }> | undefined
   292→   292→  if (Array.isArray(orient)) {
   293→   293→    result.orientacion = orient.map(o => ({
   294→   294→      anio: o.anio || '',
   295→   295→      literal: o.literal || '',
   296→   296→    }))
   297→   297→  }
   298→   298→  while (result.orientacion.length < 5) {
   299→   299→    result.orientacion.push({ anio: result.aniosEscolares[result.orientacion.length] || '', literal: '' })
   300→   300→  }
   301→   301→
   302→   302→  // Grupos
   303→   303→  const grps = data['grupos'] as Array<{ anio: string; grupo: string; literal: string }> | undefined
   304→   304→  if (Array.isArray(grps)) {
   305→   305→    result.grupos = grps.map(g => ({
   306→   306→      anio: g.anio || '',
   307→   307→      grupo: g.grupo || '',
   308→   308→      literal: g.literal || '',
   309→   309→    }))
   310→   310→  }
   311→   311→  while (result.grupos.length < 5) {
   312→   312→    result.grupos.push({ anio: result.aniosEscolares[result.grupos.length] || '', grupo: '', literal: '' })
   313→   313→  }
   314→   314→
   315→   315→  // Observaciones
   316→   316→  const obs = data['observaciones'] as string[] | undefined
   317→   317→  if (Array.isArray(obs)) {
   318→   318→    result.observaciones = obs.filter(o => o && o.trim())
   319→   319→  }
   320→   320→  result.observacionCompleta = result.observaciones.join(' ')
   321→   321→
   322→   322→  // Literales finales
   323→   323→  const lits = data['literalesFinales'] as string[] | undefined
   324→   324→  if (Array.isArray(lits)) {
   325→   325→    result.literalesFinales = lits.filter(l => l && l.trim())
   326→   326→  }
   327→   327→
   328→   328→  return result
   329→   329→}

   774→   774→export function parseCertData(rawDataStr: string | null | undefined, plan: string | null | undefined): ParsedCertData | null {
   775→   775→  if (!rawDataStr) return null
   776→   776→
   777→   777→  try {
   778→   778→    const rawData = typeof rawDataStr === 'string' ? JSON.parse(rawDataStr) : rawDataStr
   779→   779→    if (!rawData || typeof rawData !== 'object') return null
   780→   780→
   781→   781→    // Detectar formato estructurado
   782→   782→    if (rawData._format === 'structured_v1') {
   783→   783→      const p = rawData._plan || plan
   784→   784→      if (p === 'derogado') return parseStructuredDerogado(rawData)
   785→   785→      return parseStructuredVigente(rawData)
   786→   786→    }
   787→   787→
   788→   788→    // Detectar formato plano por claves numéricas
   789→   789→    const isBD2 = rawData['9°'] !== undefined || (plan === 'derogado')
   790→   790→    const isBD = rawData['8'] !== undefined && !rawData['9°']
   791→   791→
   792→   792→    if (isBD2) return parseBD2RawDataLegacy(rawData)
   793→   793→    if (isBD) return parseBDRawDataLegacy(rawData)
   794→   794→
   795→   795→    // Fallback: buscar claves numéricas
   796→   796→    const numKeys = Object.keys(rawData).filter(k => {
   797→   797→      const n = parseInt(k)
   798→   798→      return !isNaN(n) && n >= 8 && n <= 50
   799→   799→    })
   800→   800→
   801→   801→    if (numKeys.length > 0) return parseBDRawDataLegacy(rawData)
   802→   802→    return null
   803→   803→  } catch {
   804→   804→    return null
   805→   805→  }
   806→   806→}

   808→   808→// Helper para formatear fechas a DD/MM/YYYY
   809→   809→function formatDateVal(dateStr: string): string {
   810→   810→  if (!dateStr) return ''
   811→   811→  const trimmed = String(dateStr).trim()
   812→   812→  if (!trimmed) return ''
   813→   813→  
   814→   814→  if (/^\d{1,2}\/\d{1,2}\/\d{4}$/.test(trimmed)) {
   815→   815→    const parts = trimmed.split('/')
   816→   816→    const day = parts[0].padStart(2, '0')
   817→   817→    const month = parts[1].padStart(2, '0')
   818→   818→    const year = parts[2]
   819→   819→    return `${day}/${month}/${year}`
   820→   820→  }
   821→   821→  
   822→   822→  if (trimmed.includes('T') || /^\d{4}-\d{2}-\d{2}/.test(trimmed)) {
   823→   823→    try {
   824→   824→      const d = new Date(trimmed)
   825→   825→      if (!isNaN(d.getTime())) {
   826→   826→        const day = String(d.getDate()).padStart(2, '0')
   827→   827→        const month = String(d.getMonth() + 1).padStart(2, '0')
   828→   828→        const year = d.getFullYear()
   829→   829→        return `${day}/${month}/${year}`
   830→   830→      }
   831→   831→    } catch { /* ignore */ }
   832→   832→  }
   833→   833→  return trimmed
   834→   834→}
