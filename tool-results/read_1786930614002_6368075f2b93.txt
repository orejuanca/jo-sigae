   125→   125→const MESES = ['ENERO','FEBRERO','MARZO','ABRIL','MAYO','JUNIO','JULIO','AGOSTO','SEPTIEMBRE','OCTUBRE','NOVIEMBRE','DICIEMBRE'] as const

   129→   129→function formatDateLong(dateStr: string): string {
   130→   130→  const m = dateStr.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   131→   131→  if (!m) return dateStr
   132→   132→  return `${parseInt(m[1])} DE ${MESES[parseInt(m[2]) - 1]} DE ${m[3]}`
   133→   133→}
   134→   134→
   135→   135→/** Convierte fecha DD/MM/YYYY a otros formatos según dateFormat */
   136→   136→export function formatDateWithStyle(dateStr: string, dateFormat: string): string {
   137→   137→  if (!dateStr || !dateFormat) return dateStr
   138→   138→  const m = dateStr.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/)
   139→   139→  if (!m) return dateStr
   140→   140→  const day = parseInt(m[1])
   141→   141→  const month = parseInt(m[2])
   142→   142→  const year = m[3]
   143→   143→  switch (dateFormat) {
   144→   144→    case 'DD DE MES DE YYYY':
   145→   145→      return `${day} DE ${MESES[month - 1]} DE ${year}`
   146→   146→    case 'DD/MM/YYYY':
   147→   147→      return `${String(day).padStart(2,'0')}/${String(month).padStart(2,'0')}/${year}`
   148→   148→    case 'YYYY-MM-DD':
   149→   149→      return `${year}-${String(month).padStart(2,'0')}-${String(day).padStart(2,'0')}`
   150→   150→    case 'MES DD, YYYY':
   151→   151→      return `${MESES_INGLES[month - 1]} ${day}, ${year}`
   152→   152→    default:
   153→   153→      return dateStr
   154→   154→  }
   155→   155→}

   159→   159→function resolveSingleBinding(path: string, data: DisplayData, gridConfig?: GridConfig): string {
   160→   160→  const [domain, ...rest] = path.split('.')
   161→   161→  switch (domain) {
   162→   162→    case 'student': return rest.reduce((o: any, k: string) => o?.[k], data.estudiante) || ''
   163→   163→    case 'school': {
   164→   164→      const schoolMap: Record<string, string> = {
   165→   165→        codigo: data.od, denominacion: data.denominacion, direccion: data.direccion,
   166→   166→        telefono: data.telefono, municipio: data.municipio, estado: data.estado, cdcce: data.cdcce
   167→   167→      }
   168→   168→      return schoolMap[rest[0]] || ''
   169→   169→    }
   170→   170→    case 'doc': {
   171→   171→      // Soporte para doc.literalFinal.N
   172→   172→      if (rest[0] === 'literalFinal') {
   173→   173→        const idx = parseInt(rest[1])
   174→   174→        return data.literalesFinales?.[idx] || ''
   175→   175→      }
   176→   176→      const docMap: Record<string, string> = {
   177→   177→        planEstudio: data.planEstudio, codigo: data.planCodigo,
   178→   178→        lugar: data.lugar, fechaExpedicion: data.fechaExpedicion,
   179→   179→        observaciones: data.observaciones, promedioAcumulado: data.promedioAcumulado,
   180→   180→        acta: data.acta || '', actaFecha: data.actaFecha || '', actaAnio: data.actaAnio || '',
   181→   181→      }
   182→   182→      return docMap[rest[0]] || ''
   183→   183→    }
   184→   184→    case 'director': {
   185→   185→      if (rest[0] === 'nombre') {
   186→   186→        return data.rawDataMap?.['DIRECTOR.NOMBRE'] || data.director?.apellidosNombres || ''
   187→   187→      }
   188→   188→      if (rest[0] === 'cedula') {
   189→   189→        return data.rawDataMap?.['DIRECTOR.CEDULA'] || data.director?.cedula || ''
   190→   190→      }
   191→   191→      return ''
   192→   192→    }
   193→   193→    case 'cdcee':
   194→   194→      return rest[0] === 'nombre' ? (data.directorCdcce?.apellidosNombres || '') : (data.directorCdcce?.cedula || '')
   195→   195→    case 'inst': {
   196→   196→      const idx = parseInt(rest[0])
   197→   197→      const field = rest[1]
   198→   198→      return data.instituciones[idx]?.[field as keyof InstitucionEducativa] || '*'
   199→   199→    }
   200→   200→    case 'calif': {
   201→   201→      const yearKey = YEAR_NAME_MAP[rest[0]] || rest[0]
   202→   202→      const idx = parseInt(rest[1])
   203→   203→      const field = rest[2]
   204→   204→      const yearData = data.calificaciones?.[yearKey]
   205→   205→      if (!yearData) return ''
   206→   206→      const subj = yearData[idx]
   207→   207→      if (!subj) return ''
   208→   208→      const map: Record<string, string> = {
   209→   209→        materia: subj.materia, numero: String(subj.numero),
   210→   210→        nota: subj.nota, literal: subj.literal, te: subj.tipoEvaluacion,
   211→   211→        mes: subj.fechaMes, anio: subj.fechaAnio, inst: subj.instEduc,
   212→   212→      }
   213→   213→      return map[field] || ''
   214→   214→    }
   215→   215→    case 'orient': {
   216→   216→      const idx = parseInt(rest[0])
   217→   217→      const entry = data.orientacion?.[idx]
   218→   218→      if (!entry) return ''
   219→   219→      if (rest[1] === 'literal') return entry.literal
   220→   220→      return entry.anio || ''
   221→   221→    }
   222→   222→    case 'grupo': {
   223→   223→      const idx = parseInt(rest[0])
   224→   224→      const entry = data.grupos?.[idx]
   225→   225→      if (!entry) return ''
   226→   226→      if (rest[1] === 'literal') return entry.literal
   227→   227→      if (rest[1] === 'grupo') return entry.grupo
   228→   228→      return entry.anio || ''
   229→   229→    }
   230→   230→    case 'obsCert': {
   231→   231→      const idx = parseInt(rest[0])
   232→   232→      return data.observacionesLines?.[idx] || ''
   233→   233→    }
   234→   234→    case 'rawData': {
   235→   235→      // Plan derogado: resolve flat keys like rawData.CEDULA, rawData.NOTA.CA.1
   236→   236→      if (!data.rawDataMap) return ''
   237→   237→      const key = rest.join('.')
   238→   238→      const val = data.rawDataMap[key] || ''
   239→   239→      // Pad MES values to 2 digits (01-12)
   240→   240→      if (key.startsWith('MES.') && /^\d{1,2}$/.test(val)) {
   241→   241→        return val.padStart(2, '0')
   242→   242→      }
   243→   243→      return val
   244→   244→    }
   245→   245→    case 'expedicion': {
   246→   246→      if (rest[0] === 'fecha') {
   247→   247→        return data.rawDataMap?.['EXPEDICION.FECHA'] || data.fechaExpedicion || ''
   248→   248→      }
   249→   249→      if (rest[0] === 'lugar') {
   250→   250→        return data.rawDataMap?.['EXPEDICION.LUGAR'] || data.lugar || ''
   251→   251→      }
   252→   252→      return ''
   253→   253→    }
   254→   254→    default: return ''
   255→   255→  }
   256→   256→}

   258→   258→export function resolveBinding(path: string, data: DisplayData, gridConfig?: GridConfig, dateFormat?: string): string {
   259→   259→  if (!path || !data) return ''
   260→   260→
   261→   261→  const applyFormat = (val: string): string => {
   262→   262→    if (dateFormat && val) return formatDateWithStyle(val, dateFormat)
   263→   263→    return val
   264→   264→  }
   265→   265→
   266→   266→  // Soporte para texto libre con bindings embebidos: "Serial: {{doc.acta}} — Fecha: {{doc.actaFecha}}"
   267→   267→  if (path.includes('{{')) {
   268→   268→    return path.replace(/\{\{([^}]+)\}\}/g, (_match, bindingPath: string) => {
   269→   269→      const trimmed = bindingPath.trim()
   270→   270→      const val = resolveSingleBinding(trimmed, data, gridConfig)
   271→   271→      if (val && COMBINED_DATE_FIELDS.some(f => trimmed.endsWith(f))) {
   272→   272→        return applyFormat(formatDateLong(val))
   273→   273→      }
   274→   274→      return applyFormat(val)
   275→   275→    })
   276→   276→  }
   277→   277→
   278→   278→  // Soporte para multiples bindings/texto separados por coma
   279→   279→  // Textos literales van entre comillas: binding1, " — ", binding2
   280→   280→  if (path.includes(',')) {
   281→   281→    const parts = path.split(',').map(p => p.trim()).filter(Boolean)
   282→   282→    const resolved = parts.map(p => {
   283→   283→      // Texto literal entre comillas (comillas simples o dobles)
   284→   284→      if ((p.startsWith('"') && p.endsWith('"')) || (p.startsWith("'") && p.endsWith("'"))) {
   285→   285→        return p.slice(1, -1)
   286→   286→      }
   287→   287→      const val = resolveSingleBinding(p, data, gridConfig)
   288→   288→      if (val && COMBINED_DATE_FIELDS.some(f => p.endsWith(f))) {
   289→   289→        return applyFormat(formatDateLong(val))
   290→   290→      }
   291→   291→      return applyFormat(val)
   292→   292→    })
   293→   293→    return resolved.join('')
   294→   294→  }
   295→   295→
   296→   296→  // Binding simple (sin {{ }} ni coma)
   297→   297→  return applyFormat(resolveSingleBinding(path, data, gridConfig))
   298→   298→}
