   141→   141→  '1': 'Primer Año', '2': 'Segundo Año', '3': 'Tercer Año',

   145→   145→export function resolveBinding(path: string, data: DisplayData): string {
   146→   146→  if (!path || !data) return ''
   147→   147→  const [domain, ...rest] = path.split('.')
   148→   148→  switch (domain) {
   149→   149→    case 'student': return rest.reduce((o: any, k: string) => o?.[k], data.estudiante) || ''
   150→   150→    case 'school': {
   151→   151→      const schoolMap: Record<string, string> = {
   152→   152→        codigo: data.od, denominacion: data.denominacion, direccion: data.direccion,
   153→   153→        telefono: data.telefono, municipio: data.municipio, estado: data.estado, cdcce: data.cdcce
   154→   154→      }
   155→   155→      return schoolMap[rest[0]] || ''
   156→   156→    }
   157→   157→    case 'doc': {
   158→   158→      const docMap: Record<string, string> = {
   159→   159→        planEstudio: data.planEstudio, codigo: data.planCodigo,
   160→   160→        lugar: data.lugar, fechaExpedicion: data.fechaExpedicion,
   161→   161→        observaciones: data.observaciones, promedioAcumulado: data.promedioAcumulado
   162→   162→      }
   163→   163→      return docMap[rest[0]] || ''
   164→   164→    }
   165→   165→    case 'director':
   166→   166→      return rest[0] === 'nombre' ? (data.director?.apellidosNombres || '') : (data.director?.cedula || '')
   167→   167→    case 'cdcee':
   168→   168→      return rest[0] === 'nombre' ? (data.directorCdcce?.apellidosNombres || '') : (data.directorCdcce?.cedula || '')
   169→   169→    case 'inst': {
   170→   170→      const idx = parseInt(rest[0])
   171→   171→      const field = rest[1]
   172→   172→      return data.instituciones[idx]?.[field as keyof InstitucionEducativa] || '*'
   173→   173→    }
   174→   174→    case 'calif': {
   175→   175→      const yearKey = YEAR_NAME_MAP[rest[0]] || rest[0]
   176→   176→      const idx = parseInt(rest[1])
   177→   177→      const field = rest[2]
   178→   178→      const yearData = data.calificaciones?.[yearKey]
   179→   179→      if (!yearData) return ''
   180→   180→      const subj = yearData[idx]
   181→   181→      if (!subj) return ''
   182→   182→      const map: Record<string, string> = {
   183→   183→        materia: subj.materia, numero: String(subj.numero),
   184→   184→        nota: subj.nota, literal: subj.literal, te: subj.tipoEvaluacion,
   185→   185→        mes: subj.fechaMes, anio: subj.fechaAnio, inst: subj.instEduc,
   186→   186→      }
   187→   187→      return map[field] || ''
   188→   188→    }
   189→   189→    case 'orient': {
   190→   190→      const idx = parseInt(rest[0])
   191→   191→      const entry = data.orientacion?.[idx]
   192→   192→      if (!entry) return ''
   193→   193→      if (rest[1] === 'literal') return entry.literal
   194→   194→      return entry.anio || ''
   195→   195→    }
   196→   196→    case 'grupo': {
   197→   197→      const idx = parseInt(rest[0])
   198→   198→      const entry = data.grupos?.[idx]
   199→   199→      if (!entry) return ''
   200→   200→      if (rest[1] === 'literal') return entry.literal
   201→   201→      if (rest[1] === 'grupo') return entry.grupo
   202→   202→      return entry.anio || ''
   203→   203→    }
   204→   204→    default: return ''
   205→   205→  }
   206→   206→}
