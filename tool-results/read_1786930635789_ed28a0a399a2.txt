    11→    11→function generateCalifBindings() {
    12→    12→  const groups: { group: string; bindings: { value: string; label: string }[] }[] = []
    13→    13→  for (let y = 1; y <= 5; y++) {
    14→    14→    const materias = MATERIAS_VIGENTE[y] || []
    15→    15→    const bindings: { value: string; label: string }[] = []
    16→    16→    for (let s = 0; s < materias.length; s++) {
    17→    17→      const nombre = materias[s]
    18→    18→      bindings.push(
    19→    19→        { value: `calif.${y}.${s}.materia`, label: `${s + 1}. ${nombre}` },
    20→    20→        { value: `calif.${y}.${s}.nota`, label: `${s + 1}. ${nombre} — Nota` },
    21→    21→        { value: `calif.${y}.${s}.literal`, label: `${s + 1}. ${nombre} — Literal` },
    22→    22→        { value: `calif.${y}.${s}.te`, label: `${s + 1}. ${nombre} — T-E` },
    23→    23→        { value: `calif.${y}.${s}.mes`, label: `${s + 1}. ${nombre} — Mes` },
    24→    24→        { value: `calif.${y}.${s}.anio`, label: `${s + 1}. ${nombre} — Año` },
    25→    25→        { value: `calif.${y}.${s}.inst`, label: `${s + 1}. ${nombre} — Inst. Educ.` },
    26→    26→        { value: `calif.${y}.${s}.numero`, label: `${s + 1}. ${nombre} — N°` },
    27→    27→      )
    28→    28→    }
    29→    29→    groups.push({ group: `Calificaciones — ${YEAR_NAMES[y - 1]}`, bindings })
    30→    30→  }
    31→    31→  return groups
    32→    32→}
