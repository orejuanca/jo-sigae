     3→export interface CellConfig {
     4→  content: string           // Static text label (e.g., "Código:")
     5→  dataBinding: string       // Dot-path to data field (e.g., "student.cedula", "school.denominacion")
     6→  colspan: number           // Default 1
     7→  rowspan: number           // Default 1
     8→  width: string             // e.g., '10%', '100px', '' (auto)
     9→  height: string            // e.g., '20px', 'auto'
    10→  fontSize: number          // Default 9
    11→  fontWeight: 'normal' | 'bold'
    12→  textAlign: 'left' | 'center' | 'right'
    13→  verticalAlign: 'top' | 'middle' | 'bottom'
    14→  borderTop: boolean        // Default true
    15→  borderRight: boolean      // Default true
    16→  borderBottom: boolean     // Default true
    17→  borderLeft: boolean       // Default true
    18→  borderColor: string       // Default '#000000'
    19→  bgColor: string           // Default '' (transparent)
    20→  color: string             // Default '' (inherit)
    21→  whiteSpace: 'normal' | 'nowrap' | 'pre'
    22→  padding: string           // Default '1px 2px'
    23→  fontStyle: 'normal' | 'italic'
    24→}
    25→
    26→export interface GridRow {
    27→  cells: Record<number, CellConfig>  // colIndex -> CellConfig
    28→}

    30→export interface GridConfig {
    31→  totalCols: number
    32→  columnWidths: string[]   // Width per column index, e.g., ['4.57%','13%','13%',...]
    33→  rows: GridRow[]
    34→}
    35→
    36→// Factory for empty cell with defaults
    37→export function emptyCell(overrides?: Partial<CellConfig>): CellConfig {
    38→  return {
    39→    content: '', dataBinding: '', colspan: 1, rowspan: 1,
    40→    width: '', height: '', fontSize: 9, fontWeight: 'normal',
    41→    textAlign: 'left', verticalAlign: 'middle',
    42→    borderTop: true, borderRight: true, borderBottom: true, borderLeft: true,
    43→    borderColor: '#000000', bgColor: '', color: '', whiteSpace: 'normal',
    44→    padding: '1px 2px', fontStyle: 'normal',
    45→    ...overrides,
    46→  }
    47→}
    48→
    49→// Factory for empty row
    50→export function emptyRow(totalCols: number): GridRow {
    51→  const cells: Record<number, CellConfig> = {}
    52→  for (let c = 0; c < totalCols; c++) {
    53→    cells[c] = emptyCell()
    54→  }
    55→  return { cells }
    56→}

   149→// === Data Resolution for Preview ===
   150→const YEAR_NAME_MAP: Record<string, string> = {
   151→  '1': 'Primer Año', '2': 'Segundo Año', '3': 'Tercer Año',
   152→  '4': 'Cuarto Año', '5': 'Quinto Año',
   153→}

   155→export function resolveBinding(path: string, data: DisplayData): string {
   156→  if (!path || !data) return ''
   157→  const [domain, ...rest] = path.split('.')
   158→  switch (domain) {
   159→    case 'student': return rest.reduce((o: any, k: string) => o?.[k], data.estudiante) || ''
   160→    case 'school': {
   161→      const schoolMap: Record<string, string> = {
   162→        codigo: data.od, denominacion: data.denominacion, direccion: data.direccion,
   163→        telefono: data.telefono, municipio: data.municipio, estado: data.estado, cdcce: data.cdcce
   164→      }
   165→      return schoolMap[rest[0]] || ''
   166→    }
   167→    case 'doc': {
   168→      const docMap: Record<string, string> = {
   169→        planEstudio: data.planEstudio, codigo: data.planCodigo,
   170→        lugar: data.lugar, fechaExpedicion: data.fechaExpedicion,
   171→        observaciones: data.observaciones, promedioAcumulado: data.promedioAcumulado
   172→      }
   173→      return docMap[rest[0]] || ''
   174→    }
   175→    case 'director':
   176→      return rest[0] === 'nombre' ? (data.director?.apellidosNombres || '') : (data.director?.cedula || '')
   177→    case 'cdcee':
   178→      return rest[0] === 'nombre' ? (data.directorCdcce?.apellidosNombres || '') : (data.directorCdcce?.cedula || '')
   179→    case 'inst': {
   180→      const idx = parseInt(rest[0])
   181→      const field = rest[1]
   182→      return data.instituciones[idx]?.[field as keyof InstitucionEducativa] || '*'
   183→    }
   184→    case 'calif': {
   185→      const yearKey = YEAR_NAME_MAP[rest[0]] || rest[0]
   186→      const idx = parseInt(rest[1])
   187→      const field = rest[2]
   188→      const yearData = data.calificaciones?.[yearKey]
   189→      if (!yearData) return ''
   190→      const subj = yearData[idx]
   191→      if (!subj) return ''
   192→      const map: Record<string, string> = {
   193→        materia: subj.materia, numero: String(subj.numero),
   194→        nota: subj.nota, literal: subj.literal, te: subj.tipoEvaluacion,
   195→        mes: subj.fechaMes, anio: subj.fechaAnio, inst: subj.instEduc,
   196→      }
   197→      return map[field] || ''
   198→    }
   199→    case 'orient': {
   200→      const idx = parseInt(rest[0])
   201→      const entry = data.orientacion?.[idx]
   202→      if (!entry) return ''
   203→      if (rest[1] === 'literal') return entry.literal
   204→      return entry.anio || ''
   205→    }
   206→    case 'grupo': {
   207→      const idx = parseInt(rest[0])
   208→      const entry = data.grupos?.[idx]
   209→      if (!entry) return ''
   210→      if (rest[1] === 'literal') return entry.literal
   211→      if (rest[1] === 'grupo') return entry.grupo
   212→      return entry.anio || ''
   213→    }
   214→    case 'obsLine': {
   215→      const idx = parseInt(rest[0])
   216→      return data.observacionesLines?.[idx] || ''
   217→    }
   218→    default: return ''
   219→  }
   220→}

   227→function row(cells: [number, CellConfig][]): GridRow {
   228→  const allCells: Record<number, CellConfig> = {}
   229→  cells.forEach(([col, cell]) => { allCells[col] = cell })
   230→  return { cells: allCells }
   231→}

   233→// Bold cell shorthand
   234→function b(content: string, dataBinding?: string, overrides?: Partial<CellConfig>): CellConfig {

   236→}

   240→  return emptyCell({ content, fontWeight: 'bold', textAlign: 'center', verticalAlign: 'middle', ...overrides })
