     3→export interface CellConfig {
     4→  content: string           // Static text label (e.g., "Código:")
     5→  dataBinding: string       // Dot-path to data field (e.g., "student.cedula", "school.denominacion")
     6→  colspan: number           // Default 1
     7→  rowspan: number           // Default 1
     8→  width: string             // e.g., '10%', '100px', '' (auto)
     9→  height: string            // e.g., '20px', 'auto'
    10→  fontSize: number          // Default 9
    11→  fontWeight: 'normal' | 'bold'
    12→  textAlign: 'left' | 'center' | 'right'
    13→  verticalAlign: 'top' | 'middle' | 'bottom'
    14→  borderTop: boolean        // Default true
    15→  borderRight: boolean      // Default true
    16→  borderBottom: boolean     // Default true
    17→  borderLeft: boolean       // Default true
    18→  borderColor: string       // Default '#000000'
    19→  bgColor: string           // Default '' (transparent)
    20→  color: string             // Default '' (inherit)
    21→  whiteSpace: 'normal' | 'nowrap' | 'pre'
    22→  padding: string           // Default '1px 2px'
    23→  fontStyle: 'normal' | 'italic'
    24→}
    25→
    26→export interface GridRow {
    27→  cells: Record<number, CellConfig>  // colIndex -> CellConfig
    28→}
    29→
    30→export interface GridConfig {
    31→  totalCols: number
    32→  columnWidths: string[]   // Width per column index, e.g., ['4.57%','13%','13%',...]
    33→  rows: GridRow[]
    34→}
    35→
    36→// Factory for empty cell with defaults
    37→export function emptyCell(overrides?: Partial<CellConfig>): CellConfig {
    38→  return {
    39→    content: '', dataBinding: '', colspan: 1, rowspan: 1,
    40→    width: '', height: '', fontSize: 9, fontWeight: 'normal',
    41→    textAlign: 'left', verticalAlign: 'middle',
    42→    borderTop: true, borderRight: true, borderBottom: true, borderLeft: true,
    43→    borderColor: '#000000', bgColor: '', color: '', whiteSpace: 'normal',
    44→    padding: '1px 2px', fontStyle: 'normal',
    45→    ...overrides,
    46→  }
    47→}
    48→
    49→// Factory for empty row
    50→export function emptyRow(totalCols: number): GridRow {
    51→  const cells: Record<number, CellConfig> = {}
    52→  for (let c = 0; c < totalCols; c++) {
    53→    cells[c] = emptyCell()
    54→  }
    55→  return { cells }
    56→}

   140→const YEAR_NAME_MAP: Record<string, string> = {
   141→  '1': 'Primer Año', '2': 'Segundo Año', '3': 'Tercer Año',
   142→  '4': 'Cuarto Año', '5': 'Quinto Año',

   145→export function resolveBinding(path: string, data: DisplayData): string {
   146→  if (!path || !data) return ''
   147→  const [domain, ...rest] = path.split('.')
   148→  switch (domain) {
   149→    case 'student': return rest.reduce((o: any, k: string) => o?.[k], data.estudiante) || ''
   150→    case 'school': {
   151→      const schoolMap: Record<string, string> = {
   152→        codigo: data.od, denominacion: data.denominacion, direccion: data.direccion,
   153→        telefono: data.telefono, municipio: data.municipio, estado: data.estado, cdcce: data.cdcce
   154→      }
   155→      return schoolMap[rest[0]] || ''
   156→    }
   157→    case 'doc': {
   158→      const docMap: Record<string, string> = {
   159→        planEstudio: data.planEstudio, codigo: data.planCodigo,
   160→        lugar: data.lugar, fechaExpedicion: data.fechaExpedicion,
   161→        observaciones: data.observaciones, promedioAcumulado: data.promedioAcumulado
   162→      }
   163→      return docMap[rest[0]] || ''
   164→    }
   165→    case 'director':
   166→      return rest[0] === 'nombre' ? (data.director?.apellidosNombres || '') : (data.director?.cedula || '')
   167→    case 'cdcee':
   168→      return rest[0] === 'nombre' ? (data.directorCdcce?.apellidosNombres || '') : (data.directorCdcce?.cedula || '')
   169→    case 'inst': {
   170→      const idx = parseInt(rest[0])
   171→      const field = rest[1]
   172→      return data.instituciones[idx]?.[field as keyof InstitucionEducativa] || '*'
   173→    }
   174→    case 'calif': {
   175→      const yearKey = YEAR_NAME_MAP[rest[0]] || rest[0]
   176→      const idx = parseInt(rest[1])
   177→      const field = rest[2]
   178→      const yearData = data.calificaciones?.[yearKey]
   179→      if (!yearData) return ''
   180→      const subj = yearData[idx]
   181→      if (!subj) return ''
   182→      const map: Record<string, string> = {
   183→        materia: subj.materia, numero: String(subj.numero),
   184→        nota: subj.nota, literal: subj.literal, te: subj.tipoEvaluacion,
   185→        mes: subj.fechaMes, anio: subj.fechaAnio, inst: subj.instEduc,
   186→      }
   187→      return map[field] || ''
   188→    }
   189→    case 'orient': {
   190→      const idx = parseInt(rest[0])
   191→      const entry = data.orientacion?.[idx]
   192→      if (!entry) return ''
   193→      if (rest[1] === 'literal') return entry.literal
   194→      return entry.anio || ''
   195→    }
   196→    case 'grupo': {
   197→      const idx = parseInt(rest[0])
   198→      const entry = data.grupos?.[idx]
   199→      if (!entry) return ''
   200→      if (rest[1] === 'literal') return entry.literal
   201→      if (rest[1] === 'grupo') return entry.grupo
   202→      return entry.anio || ''
   203→    }
   204→    default: return ''
   205→  }
   206→}

   208→// === Helpers for building template ===
   209→function c(content: string, dataBinding?: string, overrides?: Partial<CellConfig>): CellConfig {
   210→  return emptyCell({ content, dataBinding, ...overrides })
   211→}
   212→
   213→function row(cells: [number, CellConfig][]): GridRow {
   214→  const allCells: Record<number, CellConfig> = {}
   215→  cells.forEach(([col, cell]) => { allCells[col] = cell })
   216→  return { cells: allCells }
   217→}
   218→
   219→// Bold cell shorthand
   220→function b(content: string, dataBinding?: string, overrides?: Partial<CellConfig>): CellConfig {
   221→  return emptyCell({ content, dataBinding, fontWeight: 'bold', ...overrides })
   222→}
   223→
   224→// Header cell (bold, centered, middle)
   225→function h(content: string, overrides?: Partial<CellConfig>): CellConfig {
   226→  return emptyCell({ content, fontWeight: 'bold', textAlign: 'center', verticalAlign: 'middle', ...overrides })
   227→}
