    64→// === Grid helpers ===
    65→function loadGridConfig(): GridConfig {
    66→  if (typeof window === 'undefined') return createDefaultTemplate()
    67→  try {
    68→    const raw = localStorage.getItem(STORAGE_KEY)
    69→    if (raw) {
    70→      const parsed = JSON.parse(raw) as GridConfig
    71→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) return patchDataBindings(parsed)
    72→    }
    73→  } catch { /* ignore */ }
    74→  return createDefaultTemplate()
    75→}
    76→
    77→function saveGridConfig(config: GridConfig) {
    78→  try {
    79→    const patched = patchDataBindings(config)
    80→    const json = JSON.stringify(patched)
    81→    console.log('[saveGridConfig] size:', (json.length / 1024).toFixed(1), 'KB')
    82→    localStorage.setItem(STORAGE_KEY, json)
    83→  } catch (e) {
    84→    console.error('[saveGridConfig] FAILED:', e)
    85→  }
    86→}

    88→function updateCellInConfig(
    89→  config: GridConfig,
    90→  rowIdx: number,
    91→  colIdx: number,
    92→  updates: Partial<CellConfig>
    93→): GridConfig {
    94→  const newRows = config.rows.map((r, i) => {
    95→    if (i !== rowIdx) return r
    96→    const cell = r.cells[colIdx] || emptyCell()
    97→    return { cells: { ...r.cells, [colIdx]: { ...cell, ...updates } } }
    98→  })
    99→  return { ...config, rows: newRows }
   100→}

   102→// === Grid Rendering Component ===
   103→function GridTable({
   104→  config,
   105→  selectedCell,
   106→  selectionRange,
   107→  onCellMouseDown,
   108→  onCellMouseEnter,
   109→  onCellMouseUp,
   110→  isPreview,
   111→  displayData,
   112→}: {
   113→  config: GridConfig
   114→  selectedCell: { row: number; col: number } | null
   115→  selectionRange: { r1: number; c1: number; r2: number; c2: number } | null
   116→  onCellMouseDown: (row: number, col: number) => void
   117→  onCellMouseEnter: (row: number, col: number) => void
   118→  onCellMouseUp: () => void
   119→  isPreview: boolean
   120→  displayData: DisplayData | null
   121→}) {
   122→  // Recompute occupied set on every render to track rowspan AND colspan correctly
   123→  const occupied = useMemo(() => {
   124→    const occ = new Set<string>()
   125→    for (let r = 0; r < config.rows.length; r++) {
   126→      const row = config.rows[r]
   127→      if (!row) continue
   128→      for (const [key, cell] of Object.entries(row.cells)) {
   129→        const c = Number(key)
   130→        const rs = cell.rowspan || 1
   131→        const cs = cell.colspan || 1
   132→        // Mark cells consumed by rowspan (rows below)
   133→        if (rs > 1) {
   134→          for (let dr = 1; dr < rs; dr++) {
   135→            occ.add(`${r + dr}-${c}`)
   136→          }
   137→        }
   138→        // Mark cells consumed by colspan (columns to the right, same row)
   139→        if (cs > 1) {
   140→          for (let dc = 1; dc < cs; dc++) {
   141→            occ.add(`${r}-${c + dc}`)
   142→          }
   143→        }
   144→      }
   145→    }
   146→    return occ
   147→  }, [config.rows])
   148→
   149→  const inRange = (r: number, c: number) => {
   150→    if (!selectionRange) return false
   151→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   152→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   153→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   154→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   155→    return r >= minR && r <= maxR && c >= minC && c <= maxC
   156→  }
   157→
   158→  const isRangeActive = selectionRange !== null
   159→  const isSingleSelected = selectedCell && !isRangeActive
   160→
   161→  return (
   162→    <div
   163→      className="overflow-auto flex-1 bg-white p-2 rounded border"
   164→      style={{ maxHeight: 'calc(100vh - 200px)' }}
   165→      onMouseUp={() => !isPreview && onCellMouseUp()}
   166→    >
   167→      <div style={{ width: '816px', height: '1344px', maxWidth: '100%', margin: '0 auto', boxShadow: '0 1px 3px rgba(0,0,0,0.12)', position: 'relative', overflow: 'hidden' }}>
   168→        <table style={{ borderCollapse: 'collapse', width: '100%', height: '100%', fontSize: '9pt', fontFamily: 'Arial, sans-serif', lineHeight: '1.2', tableLayout: 'fixed' }}>
   169→          <colgroup>
   170→            {config.columnWidths.map((w, i) => (
   171→              <col key={i} style={{ width: w || `${100 / config.totalCols}%` }} />
   172→            ))}
   173→          </colgroup>
   174→          <tbody>
   175→            {config.rows.map((gridRow, r) => {
   176→              const cells: React.ReactNode[] = []
   177→              for (let c = 0; c < config.totalCols; c++) {
   178→                const key = `${r}-${c}`
   179→                if (occupied.has(key)) continue
   180→
   181→                const cell = gridRow.cells[c] || emptyCell()
   182→
   183→                let displayContent = cell.content
   184→                if (isPreview && cell.dataBinding && displayData) {
   185→                  displayContent = resolveBinding(cell.dataBinding, displayData)
   186→                }
   187→
   188→                const borderStyle = (enabled: boolean) =>
   189→                  enabled ? `1px solid ${cell.borderColor}` : 'none'
   190→
   191→                const cellIsSelected = isSingleSelected && selectedCell?.row === r && selectedCell?.col === c
   192→                const cellInRange = isRangeActive && inRange(r, c)
   193→                const hasContent = cell.content || cell.dataBinding
   194→
   195→                cells.push(
   196→                  <td
   197→                    key={key}
   198→                    colSpan={cell.colspan > 1 ? cell.colspan : undefined}
   199→                    rowSpan={cell.rowspan > 1 ? cell.rowspan : undefined}
   200→                    style={{
   201→                      borderTop: borderStyle(cell.borderTop),
   202→                      borderRight: borderStyle(cell.borderRight),
   203→                      borderBottom: borderStyle(cell.borderBottom),
   204→                      borderLeft: borderStyle(cell.borderLeft),
   205→                      width: cell.width || undefined,
   206→                      height: cell.height || undefined,
   207→                      fontSize: `${cell.fontSize}pt`,
   208→                      fontWeight: cell.fontWeight,
   209→                      fontStyle: cell.fontStyle,
   210→                      textAlign: cell.textAlign,
   211→                      verticalAlign: cell.verticalAlign,
   212→                      color: cell.color || undefined,
   213→                      whiteSpace: cell.whiteSpace,
   214→                      padding: cell.padding,
   215→                      cursor: isPreview ? 'default' : 'cell',
   216→                      outline: cellIsSelected ? '2px solid #3b82f6' : undefined,
   217→                      outlineOffset: '-2px',
   218→                      minWidth: hasContent ? undefined : '0px',
   219→                      background: cellInRange
   220→                        ? 'rgba(59,130,246,0.15)'
   221→                        : cellIsSelected
   222→                          ? 'rgba(59,130,246,0.06)'
   223→                          : cell.bgColor || undefined,
   224→                      userSelect: 'none',
   225→                      position: 'relative',
   226→                    }}
   227→                    onMouseDown={() => !isPreview && onCellMouseDown(r, c)}
   228→                    onMouseEnter={() => !isPreview && onCellMouseEnter(r, c)}
   229→                    title={cell.dataBinding ? `[${cell.dataBinding}]` : undefined}
   230→                  >
   231→                    {!isPreview && cell.dataBinding && (
   232→                      <span style={{
   233→                        position: 'absolute', top: '1px', right: '2px',
   234→                        width: '5px', height: '5px', borderRadius: '50%',
   235→                        background: '#f97316', display: 'inline-block',
   236→                        pointerEvents: 'none',
   237→                      }} />
   238→                    )}
   239→                    {isPreview && cell.dataBinding && !displayContent ? (
   240→                      <span style={{ color: '#ccc' }}>—</span>
   241→                    ) : (
   242→                      displayContent
   243→                    )}
   244→                  </td>
   245→                )
   246→              }
   247→              return <tr key={r}>{cells}</tr>
   248→            })}
   249→          </tbody>
   250→        </table>
   251→      </div>
   252→    </div>
   253→  )
   254→}

   256→// === Column Width Editor ===
   257→function ColumnWidthEditor({
   258→  config,
   259→  onUpdate,
   260→}: {
   261→  config: GridConfig
   262→  onUpdate: (config: GridConfig) => void
   263→}) {
   264→  const [editing, setEditing] = useState<number | null>(null)
   265→  const [value, setValue] = useState('')
   266→
   267→  const startEdit = (idx: number) => {
   268→    setEditing(idx)
   269→    setValue(config.columnWidths[idx] || '')
   270→  }
   271→
   272→  const commitEdit = () => {
   273→    if (editing !== null && value.trim()) {
   274→      const newWidths = [...config.columnWidths]
   275→      newWidths[editing] = value.trim()
   276→      onUpdate({ ...config, columnWidths: newWidths })
   277→    }
   278→    setEditing(null)
   279→  }
   280→
   281→  return (
   282→    <div className="flex gap-0.5 flex-wrap p-1">
   283→      {config.columnWidths.map((w, i) => (
   284→        <div
   285→          key={i}
   286→          className="text-center border rounded px-0.5 py-0.5 min-w-[44px]"
   287→          style={{ fontSize: '7pt', backgroundColor: '#f8f8f8' }}
   288→        >
   289→          <div className="text-muted-foreground" style={{ fontSize: '6pt' }}>C{i}</div>
   290→          {editing === i ? (
   291→            <input
   292→              autoFocus
   293→              value={value}
   294→              onChange={(e) => setValue(e.target.value)}
   295→              onBlur={commitEdit}
   296→              onKeyDown={(e) => { if (e.key === 'Enter') commitEdit() }}
   297→              className="w-10 h-4 text-center text-[7pt] border rounded"
   298→              style={{ padding: '0 1px' }}
   299→            />
   300→          ) : (
   301→            <div
   302→              onClick={() => startEdit(i)}
   303→              className="cursor-pointer hover:bg-accent rounded"
   304→              title="Clic para editar"
   305→            >
   306→              {w}
   307→            </div>
   308→          )}
   309→        </div>
   310→      ))}
   311→    </div>
   312→  )
   313→}

   537→  const clearRangeSelection = useCallback(() => {
   538→    setSelAnchor(null)
   539→    setSelEnd(null)
   540→    setIsDragging(false)
   541→  }, [])
   542→
   543→  const handleCellUpdate = useCallback((updates: Partial<CellConfig>) => {
   544→    if (!selectedCell) return
   545→    setGridConfig((prev) => updateCellInConfig(prev, selectedCell.row, selectedCell.col, updates))
   546→  }, [selectedCell])
   547→
   548→  const handleApplyColumns = () => {
   549→    const n = parseInt(colInput)
   550→    if (isNaN(n) || n < 1 || n > 100) {
   551→      toast({ title: 'Error', description: 'Ingrese un número válido de columnas (1-100).', variant: 'destructive' })
   552→      return
   553→    }
   554→    const newConfig: GridConfig = {
   555→      totalCols: n,
   556→      columnWidths: Array(n).fill(`${(100 / n).toFixed(2)}%`),
   557→      rows: [emptyRow(n)],
   558→    }
   559→    setGridConfig(newConfig)
   560→    setSelectedCell(null)
   561→    toast({ title: `Grilla reiniciada con ${n} columnas` })
   562→  }
   563→
   564→  const handleAddRow = () => {
   565→    setGridConfig((prev) => ({
   566→      ...prev,
   567→      rows: [...prev.rows, emptyRow(prev.totalCols)],
   568→    }))
   569→  }
   570→
   571→  const handleInsertRowAbove = () => {
   572→    if (!selectedCell) {
   573→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar fila arriba.', variant: 'destructive' })
   574→      return
   575→    }
   576→    setGridConfig((prev) => {
   577→      const newRow = emptyRow(prev.totalCols)
   578→      const newRows = [...prev.rows]
   579→
   580→      // Adjust rowspan of cells in the row above (selectedCell.row - 1)
   581→      // that span across the insertion point
   582→      const aboveIdx = selectedCell.row - 1
   583→      if (aboveIdx >= 0 && aboveIdx < prev.rows.length) {
   584→        const aboveRow = prev.rows[aboveIdx]
   585→        const adjustedCells: Record<number, CellConfig> = {}
   586→        for (const [key, cell] of Object.entries(aboveRow.cells)) {
   587→          const col = Number(key)
   588→          const rs = cell.rowspan || 1
   589→          // If this cell spans to or past the insertion row, reduce rowspan
   590→          if (col === 0 || true) { // check all cells
   591→            if (rs > 1 && (aboveIdx + rs) > selectedCell.row) {
   592→              adjustedCells[col] = { ...cell, rowspan: rs - 1 }
   593→              continue
   594→            }
   595→          }
   596→          adjustedCells[col] = cell
   597→        }
   598→        newRows[aboveIdx] = { cells: adjustedCells }
   599→      }
   600→
   601→      // Also adjust rowspan of cells in the CURRENT row (selectedCell.row)
   602→      // since it shifts down by 1, but their rowspan origin stays the same
   603→      const currentRow = prev.rows[selectedCell.row]
   604→      if (currentRow) {
   605→        const adjustedCurrent: Record<number, CellConfig> = {}
   606→        for (const [key, cell] of Object.entries(currentRow.cells)) {
   607→          const col = Number(key)
   608→          const rs = cell.rowspan || 1
   609→          // Cells that started in a row above and were part of a rowspan
   610→          // don't need adjustment - they're already handled by the row above
   611→          adjustedCurrent[col] = cell
   612→        }
   613→        // Keep the current row as-is (it just moves down)
   614→      }
   615→
   616→      newRows.splice(selectedCell.row, 0, newRow)
   617→      setSelectedCell({ row: selectedCell.row + 1, col: selectedCell.col })
   618→      return { ...prev, rows: newRows }
   619→    })
   620→  }
   621→
   622→  const handleInsertRowBelow = () => {
   623→    if (!selectedCell) {
   624→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar fila abajo.', variant: 'destructive' })
   625→      return
   626→    }
   627→    setGridConfig((prev) => {
   628→      const newRow = emptyRow(prev.totalCols)
   629→      const insertAt = selectedCell.row + 1
   630→      const newRows = [...prev.rows]
   631→
   632→      // Adjust rowspan of cells in selectedCell.row that span across insertAt
   633→      const currentRow = prev.rows[selectedCell.row]
   634→      if (currentRow) {
   635→        const adjustedCells: Record<number, CellConfig> = {}
   636→        for (const [key, cell] of Object.entries(currentRow.cells)) {
   637→          const col = Number(key)
   638→          const rs = cell.rowspan || 1
   639→          if (rs > 1 && (selectedCell.row + rs) > insertAt) {
   640→            adjustedCells[col] = { ...cell, rowspan: rs - 1 }
   641→            continue
   642→          }
   643→          adjustedCells[col] = cell
   644→        }
   645→        newRows[selectedCell.row] = { cells: adjustedCells }
   646→      }
   647→
   648→      newRows.splice(insertAt, 0, newRow)
   649→      return { ...prev, rows: newRows }
   650→    })
   651→  }
   652→
   653→  const handleDeleteLastRow = () => {
   654→    setGridConfig((prev) => {
   655→      if (prev.rows.length <= 1) {
   656→        toast({ title: 'No se puede eliminar la última fila', variant: 'destructive' })
   657→        return prev
   658→      }
   659→      const newRows = prev.rows.slice(0, -1)
   660→      // Clear selection if it was on the deleted row
   661→      if (selectedCell && selectedCell.row >= newRows.length) {
   662→        setSelectedCell(null)
   663→      }
   664→      return { ...prev, rows: newRows }
   665→    })
   666→  }
   667→
   668→  // === Column Operations ===
   669→  const handleInsertColLeft = () => {
   670→    if (!selectedCell) {
   671→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar columna a la izquierda.', variant: 'destructive' })
   672→      return
   673→    }
   674→    setGridConfig((prev) => {
   675→      const insertAt = selectedCell.col
   676→      // Shift cell indices >= insertAt in every row
   677→      const newRows = prev.rows.map((row) => {
   678→        const newCells: Record<number, CellConfig> = {}
   679→        for (const [key, cell] of Object.entries(row.cells)) {
   680→          const idx = Number(key)
   681→          if (idx >= insertAt) {
   682→            newCells[idx + 1] = cell
   683→          } else {
   684→            newCells[idx] = cell
   685→          }
   686→        }
   687→        newCells[insertAt] = emptyCell()
   688→        return { cells: newCells }
   689→      })
   690→      // Insert column width
   691→      const newWidths = [...prev.columnWidths]
   692→      newWidths.splice(insertAt, 0, '3%')
   693→      // Adjust selection
   694→      setSelectedCell({ row: selectedCell.row, col: selectedCell.col + 1 })
   695→      return { ...prev, totalCols: prev.totalCols + 1, columnWidths: newWidths, rows: newRows }
   696→    })
   697→    setColInput(String(gridConfig.totalCols + 1))
   698→  }
   699→
   700→  const handleInsertColRight = () => {
   701→    if (!selectedCell) {
   702→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero para insertar columna a la derecha.', variant: 'destructive' })
   703→      return
   704→    }
   705→    setGridConfig((prev) => {
   706→      const insertAt = selectedCell.col + 1
   707→      // Shift cell indices >= insertAt in every row
   708→      const newRows = prev.rows.map((row) => {
   709→        const newCells: Record<number, CellConfig> = {}
   710→        for (const [key, cell] of Object.entries(row.cells)) {
   711→          const idx = Number(key)
   712→          if (idx >= insertAt) {
   713→            newCells[idx + 1] = cell
   714→          } else {
   715→            newCells[idx] = cell
   716→          }
   717→        }
   718→        newCells[insertAt] = emptyCell()
   719→        return { cells: newCells }
   720→      })
   721→      // Insert column width
   722→      const newWidths = [...prev.columnWidths]
   723→      newWidths.splice(insertAt, 0, '3%')
   724→      // Selection stays on same column
   725→      return { ...prev, totalCols: prev.totalCols + 1, columnWidths: newWidths, rows: newRows }
   726→    })
   727→    setColInput(String(gridConfig.totalCols + 1))
   728→  }
   729→
   730→  const handleDeleteLastCol = () => {
   731→    setGridConfig((prev) => {
   732→      if (prev.totalCols <= 1) {
   733→        toast({ title: 'No se puede eliminar la última columna', variant: 'destructive' })
   734→        return prev
   735→      }
   736→      const lastCol = prev.totalCols - 1
   737→      const newRows = prev.rows.map((row) => {
   738→        const { [lastCol]: _removed, ...rest } = row.cells
   739→        return { cells: rest }
   740→      })
   741→      const newWidths = prev.columnWidths.slice(0, -1)
   742→      if (selectedCell && selectedCell.col >= prev.totalCols - 1) {
   743→        setSelectedCell(null)
   744→      }
   745→      return { ...prev, totalCols: prev.totalCols - 1, columnWidths: newWidths, rows: newRows }
   746→    })
   747→    setColInput(String(gridConfig.totalCols - 1))
   748→  }
   749→
   750→  // === Merge Selection (drag range) ===
   751→  const handleMergeSelection = () => {
   752→    if (!selectionRange) {
   753→      toast({ title: 'Sin selección', description: 'Selecciona varias celdas arrastrando primero.', variant: 'destructive' })
   754→      return
   755→    }
   756→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   757→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   758→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   759→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   760→    const spanCols = maxC - minC + 1
   761→    const spanRows = maxR - minR + 1
   762→    if (spanCols === 1 && spanRows === 1) {
   763→      toast({ title: 'Una sola celda', description: 'Selecciona al menos 2 celdas para combinar.', variant: 'destructive' })
   764→      return
   765→    }
   766→    setGridConfig((prev) => {
   767→      const updated = updateCellInConfig(prev, minR, minC, {
   768→        colspan: spanCols,
   769→        rowspan: spanRows,
   770→      })
   771→      const newRows = updated.rows.map((row, ri) => {
   772→        if (ri < minR || ri > maxR) return row
   773→        const newCells = { ...row.cells }
   774→        for (let ci = minC; ci <= maxC; ci++) {
   775→          if (ri === minR && ci === minC) continue
   776→          delete newCells[ci]
   777→        }
   778→        return { cells: newCells }
   779→      })
   780→      return { ...updated, rows: newRows }
   781→    })
   782→    setSelectedCell({ row: minR, col: minC })
   783→    clearRangeSelection()
   784→    toast({ title: 'Celdas combinadas', description: `${spanCols}x${spanRows} celdas fusionadas en una.` })
   785→  }
   786→
   787→  // === Merge / Split Operations ===
   788→  // Combina columnas: elimina la columna de la derecha y suma su ancho a la actual.
   789→  // El total de columnas baja (ej: 27 → 26).
   790→  const handleMergeColumns = () => {
   791→    if (!selectedCell) {
   792→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   793→      return
   794→    }
   795→    if (gridConfig.totalCols <= 1) {
   796→      toast({ title: 'No se puede', description: 'Solo queda 1 columna.', variant: 'destructive' })
   797→      return
   798→    }
   799→    const mergeCol = selectedCell.col
   800→    if (mergeCol >= gridConfig.totalCols - 1) {
   801→      toast({ title: 'No se puede', description: 'No hay columna a la derecha para combinar.', variant: 'destructive' })
   802→      return
   803→    }
   804→    setGridConfig((prev) => {
   805→      // Sumar ancho de la columna derecha a la actual
   806→      const newWidths = [...prev.columnWidths]
   807→      const leftW = parseFloat(newWidths[mergeCol]) || 0
   808→      const rightW = parseFloat(newWidths[mergeCol + 1]) || 0
   809→      newWidths[mergeCol] = `${leftW + rightW}%`
   810→      newWidths.splice(mergeCol + 1, 1)
   811→      // En cada fila, remover la columna mergeCol+1 y reindexar
   812→      const newRows = prev.rows.map((row) => {
   813→        const newCells: Record<number, CellConfig> = {}
   814→        for (const [key, cell] of Object.entries(row.cells)) {
   815→          const idx = Number(key)
   816→          if (idx === mergeCol + 1) continue // descartar columna derecha
   817→          if (idx > mergeCol + 1) {
   818→            newCells[idx - 1] = cell // reindexar
   819→          } else {
   820→            newCells[idx] = cell
   821→          }
   822→        }
   823→        return { cells: newCells }
   824→      })
   825→      // Ajustar seleccion si estaba mas alla de la columna eliminada
   826→      if (selectedCell.col > mergeCol) {
   827→        setSelectedCell({ row: selectedCell.row, col: Math.max(0, selectedCell.col - 1) })
   828→      }
   829→      const newTotal = prev.totalCols - 1
   830→      return { ...prev, totalCols: newTotal, columnWidths: newWidths, rows: newRows }
   831→    })
   832→    setColInput(String(gridConfig.totalCols - 1))
   833→  }
   834→
   835→  // Combina filas: elimina la fila de abajo y la suma a la actual.
   836→  // El total de filas baja en 1.
   837→  const handleMergeRows = () => {
   838→    if (!selectedCell) {
   839→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   840→      return
   841→    }
   842→    if (gridConfig.rows.length <= 1) {
   843→      toast({ title: 'No se puede', description: 'Solo queda 1 fila.', variant: 'destructive' })
   844→      return
   845→    }
   846→    const mergeRow = selectedCell.row
   847→    if (mergeRow >= gridConfig.rows.length - 1) {
   848→      toast({ title: 'No se puede', description: 'No hay fila debajo para combinar.', variant: 'destructive' })
   849→      return
   850→    }
   851→    setGridConfig((prev) => {
   852→      const currentRow = prev.rows[mergeRow]
   853→      const nextRow = prev.rows[mergeRow + 1]
   854→      // Fusionar contenido de celdas: si la celda de abajo tiene contenido y la actual no, copiarlo
   855→      const mergedCells: Record<number, CellConfig> = { ...currentRow.cells }
   856→      for (const [key, cell] of Object.entries(nextRow.cells)) {
   857→        const idx = Number(key)
   858→        const existing = mergedCells[idx]
   859→        if (!existing || (!existing.content && !existing.dataBinding)) {
   860→          mergedCells[idx] = { ...cell }
   861→        } else if (cell.content && !existing.content) {
   862→          mergedCells[idx] = { ...existing, content: existing.content + ' ' + cell.content }
   863→        }
   864→      }
   865→      const newRows = [...prev.rows]
   866→      newRows[mergeRow] = { cells: mergedCells }
   867→      newRows.splice(mergeRow + 1, 1)
   868→      if (selectedCell.row >= newRows.length) {
   869→        setSelectedCell(null)
   870→      }
   871→      return { ...prev, rows: newRows }
   872→    })
   873→  }
   874→
   875→  // Separar celda: reset colspan/rowspan a 1
   876→  const handleSplitCell = () => {
   877→    if (!selectedCell) {
   878→      toast({ title: 'Sin selección', description: 'Selecciona una celda primero.', variant: 'destructive' })
   879→      return
   880→    }
   881→    const cell = gridConfig.rows[selectedCell.row]?.cells[selectedCell.col]
   882→    if (!cell || (cell.colspan === 1 && cell.rowspan === 1)) {
   883→      toast({ title: 'Sin combinación', description: 'Esta celda no está combinada (colspan/rowspan ya es 1).', variant: 'destructive' })
   884→      return
   885→    }
   886→    setGridConfig((prev) => updateCellInConfig(prev, selectedCell.row, selectedCell.col, { colspan: 1, rowspan: 1 }))
   887→  }
   888→
   889→  // === Save to DB (opens dialog for name) ===
   890→  const handleOpenSaveDialog = () => {
   891→    setSaveName('')
   892→    setShowSaveDialog(true)
   893→  }
   894→
   895→  const handleConfirmSave = async () => {
   896→    if (!saveName.trim()) {
   897→      toast({ title: 'Nombre requerido', description: 'Ingresa un nombre para el layout.', variant: 'destructive' })
   898→      return
   899→    }
   900→
   901→    setSaving(true)
   902→    try {
   903→      const res = await fetch('/api/cert-layouts', {
   904→        method: 'POST',
   905→        headers: { 'Content-Type': 'application/json' },
   906→        body: JSON.stringify({
   907→          nombre: saveName.trim(),
   908→          datos: gridConfig,
   909→        }),
   910→      })
   911→
   912→      if (!res.ok) {
   913→        const err = await res.json()
   914→        throw new Error(err.error || 'Error al guardar')
   915→      }
   916→
   917→      setShowSaveDialog(false)
   918→      toast({
   919→        title: '¡Guardado exitoso!',
   920→        description: `El layout "${saveName.trim()}" se guardó correctamente en la base de datos.`,
   921→      })
   922→    } catch (error: unknown) {
   923→      const msg = error instanceof Error ? error.message : 'Error desconocido'
   924→      toast({ title: 'Error al guardar', description: msg, variant: 'destructive' })
   925→    } finally {
   926→      setSaving(false)
   927→    }
   928→  }
   929→
   930→  // === Load saved layouts ===
   931→  const handleOpenLayoutsDialog = async () => {
   932→    setShowLayoutsDialog(true)
   933→    setLoadingLayouts(true)
   934→    try {
   935→      const res = await fetch('/api/cert-layouts')
   936→      if (res.ok) {
   937→        const data = await res.json()
   938→        setSavedLayouts(data)
   939→      }
   940→    } catch {
   941→      toast({ title: 'Error', description: 'No se pudieron cargar los layouts.', variant: 'destructive' })
   942→    } finally {
   943→      setLoadingLayouts(false)
   944→    }
   945→  }
   946→
   947→  const handleLoadLayoutFromDB = async (id: string) => {
   948→    setLoadingLayout(true)
   949→    try {
   950→      const res = await fetch(`/api/cert-layouts/${id}`)
   951→      if (!res.ok) {
   952→        throw new Error('Layout no encontrado')
   953→      }
   954→      const layout = await res.json()
   955→      const parsed: GridConfig = typeof layout.datos === 'string'
   956→        ? JSON.parse(layout.datos)
   957→        : layout.datos
   958→
   959→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) {
   960→        setGridConfig(patchDataBindings(parsed))
   961→        setSelectedCell(null)
   962→        setShowLayoutsDialog(false)
   963→        toast({ title: 'Layout cargado', description: `"${layout.nombre}" se cargó correctamente.` })
   964→      } else {
   965→        throw new Error('Datos de layout inválidos')
   966→      }
   967→    } catch (error: unknown) {
   968→      const msg = error instanceof Error ? error.message : 'Error desconocido'
   969→      toast({ title: 'Error al cargar', description: msg, variant: 'destructive' })
   970→    } finally {
   971→      setLoadingLayout(false)
   972→    }
   973→  }
   974→
   975→  const handleDeleteLayout = async (id: string) => {
   976→    try {
   977→      const res = await fetch(`/api/cert-layouts/${id}`, { method: 'DELETE' })
   978→      if (res.ok) {
   979→        setSavedLayouts((prev) => prev.filter((l) => l.id !== id))
   980→        toast({ title: 'Layout eliminado' })
   981→      }
   982→    } catch {
   983→      toast({ title: 'Error', description: 'No se pudo eliminar el layout.', variant: 'destructive' })
   984→    }
   985→  }
   986→
   987→  // Local save/load/reset
   988→  const handleLoadLocal = () => {
   989→    const loaded = loadGridConfig()
   990→    setGridConfig(loaded)
   991→    setSelectedCell(null)
   992→    toast({ title: 'Diseño cargado', description: 'Se restauró desde el almacenamiento local.' })
   993→  }
   994→
   995→  const handleReset = () => {
   996→    const def = createDefaultTemplate()
   997→    setGridConfig(def)
   998→    setSelectedCell(null)
   999→    setColInput('27')
  1000→    toast({ title: 'Diseño restablecido', description: 'Se restauró la plantilla por defecto.' })
  1001→  }
  1002→
  1003→  // Selected cell data
  1004→  const selectedCellData = useMemo(() => {
  1005→    if (!selectedCell) return null
  1006→    const row = gridConfig.rows[selectedCell.row]
  1007→    return row?.cells[selectedCell.col] || emptyCell()
  1008→  }, [selectedCell, gridConfig])

  1011→    <AppShell>
  1012→      <div className="space-y-3 print:hidden">
  1013→        {/* Header */}
  1014→        <div>
  1015→          <h1 className="text-2xl font-bold">Editor de Grilla — Certificaciones</h1>
