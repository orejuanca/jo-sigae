    34→// === Auto-fit: reduce fontSize para que el texto quepa en la celda ===
    35→function AutoFitCell({ children, fontSize, fontWeight, autoFit }: {
    36→  children: React.ReactNode; fontSize: number; fontWeight: string; autoFit: boolean
    37→}) {
    38→  const ref = useRef<HTMLSpanElement>(null)
    39→  const scaledRef = useRef<number | null>(null)
    40→
    41→  useLayoutEffect(() => {
    42→    const span = ref.current
    43→    if (!span) return
    44→
    45→    if (!autoFit) {
    46→      if (scaledRef.current !== null) {
    47→        scaledRef.current = null
    48→        span.style.cssText = ''
    49→      }
    50→      return
    51→    }
    52→
    53→    const td = span.closest('td') as HTMLTableCellElement | null
    54→    if (!td) return
    55→
    56→    const text = span.textContent || ''
    57→    if (!text.trim()) return
    58→
    59→    const cs = window.getComputedStyle(td)
    60→    const pl = parseFloat(cs.paddingLeft) || 0
    61→    const pr = parseFloat(cs.paddingRight) || 0
    62→    const available = td.clientWidth - pl - pr
    63→    if (available <= 0) return
    64→
    65→    // Resetear al fontSize original y forzar nowrap para medir ancho natural
    66→    span.style.fontSize = `${fontSize}pt`
    67→    span.style.whiteSpace = 'nowrap'
    68→    span.style.display = 'inline-block'
    69→    span.style.maxWidth = 'none'
    70→    span.style.overflow = 'visible'
    71→
    72→    // offsetWidth = layout width, NO afectado por transform:scale del padre
    73→    const textW = span.offsetWidth
    74→
    75→    if (textW > available) {
    76→      const ratio = (available * 0.99) / textW
    77→      const newSize = Math.max(Math.round(fontSize * ratio * 10) / 10, fontSize * 0.5)
    78→      scaledRef.current = newSize
    79→      span.style.fontSize = `${newSize}pt`
    80→      span.style.whiteSpace = 'nowrap'
    81→      span.style.overflow = 'hidden'
    82→      span.style.maxWidth = `${available}px`
    83→    } else {
    84→      scaledRef.current = null
    85→      span.style.cssText = ''
    86→    }
    87→  })
    88→
    89→  return <span ref={ref}>{children}</span>
    90→}

   140→// === Grid helpers ===
   141→function loadGridConfig(): GridConfig {
   142→  if (typeof window === 'undefined') return createDefaultTemplate()
   143→  try {
   144→    const raw = localStorage.getItem(STORAGE_KEY)
   145→    if (raw) {
   146→      const parsed = JSON.parse(raw) as GridConfig
   147→      if (parsed.rows && Array.isArray(parsed.rows) && parsed.totalCols) return patchDataBindings(parsed)
   148→    }
   149→  } catch { /* ignore */ }
   150→  return createDefaultTemplate()
   151→}

   164→function updateCellInConfig(
   165→  config: GridConfig,
   166→  rowIdx: number,
   167→  colIdx: number,
   168→  updates: Partial<CellConfig>
   169→): GridConfig {
   170→  const newRows = config.rows.map((r, i) => {
   171→    if (i !== rowIdx) return r
   172→    const cell = r.cells[colIdx] || emptyCell()
   173→    return { cells: { ...r.cells, [colIdx]: { ...cell, ...updates } } }
   174→  })
   175→  return { ...config, rows: newRows }
   176→}

   178→// === Grid Rendering Component ===
   179→function GridTable({
   180→  config,
   181→  selectedCell,
   182→  selectionRange,
   183→  onCellMouseDown,
   184→  onCellMouseEnter,
   185→  onCellMouseUp,
   186→  isPreview,
   187→  displayData,
   188→  onCellEdit,
   189→  savingDraft,
   190→  draftOverrides,
   191→  pageWidth,
   192→  pageHeight,
   193→}: {
   194→  config: GridConfig
   195→  selectedCell: { row: number; col: number } | null
   196→  selectionRange: { r1: number; c1: number; r2: number; c2: number } | null
   197→  onCellMouseDown: (row: number, col: number) => void
   198→  onCellMouseEnter: (row: number, col: number) => void
   199→  onCellMouseUp: () => void
   200→  isPreview: boolean
   201→  displayData: DisplayData | null
   202→  onCellEdit?: (binding: string, newValue: string) => void
   203→  savingDraft?: boolean
   204→  draftOverrides?: Record<string, string>
   205→  pageWidth?: number
   206→  pageHeight?: number
   207→}) {
   208→  // Recompute occupied set on every render to track rowspan AND colspan correctly
   209→  const occupied = useMemo(() => {
   210→    const occ = new Set<string>()
   211→    for (let r = 0; r < config.rows.length; r++) {
   212→      const row = config.rows[r]
   213→      if (!row) continue
   214→      for (const [key, cell] of Object.entries(row.cells)) {
   215→        const c = Number(key)
   216→        const rs = cell.rowspan || 1
   217→        const cs = cell.colspan || 1
   218→        // Mark cells consumed by rowspan (rows below)
   219→        if (rs > 1) {
   220→          for (let dr = 1; dr < rs; dr++) {
   221→            occ.add(`${r + dr}-${c}`)
   222→          }
   223→        }
   224→        // Mark cells consumed by colspan (columns to the right, same row)
   225→        if (cs > 1) {
   226→          for (let dc = 1; dc < cs; dc++) {
   227→            occ.add(`${r}-${c + dc}`)
   228→          }
   229→        }
   230→      }
   231→    }
   232→    return occ
   233→  }, [config.rows])
   234→
   235→  const inRange = (r: number, c: number) => {
   236→    if (!selectionRange) return false
   237→    const minR = Math.min(selectionRange.r1, selectionRange.r2)
   238→    const maxR = Math.max(selectionRange.r1, selectionRange.r2)
   239→    const minC = Math.min(selectionRange.c1, selectionRange.c2)
   240→    const maxC = Math.max(selectionRange.c1, selectionRange.c2)
   241→    return r >= minR && r <= maxR && c >= minC && c <= maxC
   242→  }
   243→
   244→  const isRangeActive = selectionRange !== null
   245→  const isSingleSelected = selectedCell && !isRangeActive
   246→
   247→  return (
   248→    <div
   249→      className="bg-white p-2 rounded border"
   250→      style={{ maxWidth: '860px', margin: '0 auto', overflow: 'visible' }}
   251→      onMouseUp={() => !isPreview && onCellMouseUp()}
   252→    >
   253→      <div style={{ width: `${pageWidth || 816}px`, height: `${pageHeight || 1344}px`, maxWidth: '100%', margin: '0 auto', boxShadow: '0 1px 3px rgba(0,0,0,0.12)', position: 'relative', overflowY: 'auto', overflowX: 'hidden', background: 'white' }}>
   254→        <table style={{ borderCollapse: 'collapse', width: '100%', fontSize: '9pt', fontFamily: 'Arial, sans-serif', lineHeight: '1.2', tableLayout: 'fixed' }}>
   255→          <colgroup>
   256→            {config.columnWidths.map((w, i) => (
   257→              <col key={i} style={{ width: w || `${100 / config.totalCols}%` }} />
   258→            ))}
   259→          </colgroup>
   260→          <tbody>
   261→            {config.rows.map((gridRow, r) => {
   262→              const cells: React.ReactNode[] = []
   263→              for (let c = 0; c < config.totalCols; c++) {
   264→                const key = `${r}-${c}`
   265→                if (occupied.has(key)) continue
   266→
   267→                const cell = gridRow.cells[c] || emptyCell()
   268→
   269→                let displayContent = cell.content
   270→                if (isPreview && cell.dataBinding && displayData) {
   271→                  const ov = draftOverrides?.[cell.dataBinding]
   272→                  const resolved = ov !== undefined && ov !== '' ? ov : resolveBinding(cell.dataBinding, displayData, config, cell.dateFormat)
   273→                  displayContent = resolved || cell.content
   274→                }
   275→
   276→                const borderStyle = (enabled: boolean) =>
   277→                        enabled ? `${(cell.borderStyle === 'double' || cell.borderStyle === 'groove' || cell.borderStyle === 'ridge') ? '3px' : '1px'} ${cell.borderStyle || 'solid'} ${cell.borderColor}` : 'none'
   278→
   279→                const cellIsSelected = isSingleSelected && selectedCell?.row === r && selectedCell?.col === c
   280→                const cellInRange = isRangeActive && inRange(r, c)
   281→                const hasContent = cell.content || cell.dataBinding
   282→
   283→                cells.push(
   284→                  <td
   285→                    key={key}
   286→                    colSpan={cell.colspan > 1 ? cell.colspan : undefined}
   287→                    rowSpan={cell.rowspan > 1 ? cell.rowspan : undefined}
   288→                    style={{
   289→                      borderTop: borderStyle(cell.borderTop),
   290→                      borderRight: borderStyle(cell.borderRight),
   291→                      borderBottom: borderStyle(cell.borderBottom),
   292→                      borderLeft: borderStyle(cell.borderLeft),
   293→                      width: cell.width || undefined,
   294→                      height: cell.height || '24px',
   295→                      fontSize: `${cell.fontSize}pt`,
   296→                      fontWeight: cell.fontWeight,
   297→                      fontStyle: cell.fontStyle,
   298→                      textDecoration: cell.textDecoration === 'underline' ? 'underline' : undefined,
   299→                      textAlign: cell.textAlign,
   300→                      verticalAlign: cell.verticalAlign,
   301→                      color: cell.color || undefined,
   302→                      whiteSpace: cell.whiteSpace,
   303→                      padding: cell.padding,
   304→                      cursor: isPreview ? 'default' : 'cell',
   305→                      outline: cellIsSelected ? '2px solid #3b82f6' : undefined,
   306→                      outlineOffset: '-2px',
   307→                      minWidth: hasContent ? undefined : '0px',
   308→                      minHeight: '24px',
   309→                      background: cellInRange
   310→                        ? 'rgba(59,130,246,0.15)'
   311→                        : cellIsSelected
   312→                          ? 'rgba(59,130,246,0.06)'
   313→                          : cell.bgColor || undefined,
   314→                      writingMode: cell.writingMode || undefined,
   315→                      transform: cell.writingMode === 'rotate-180' ? 'rotate(180deg)' : undefined,
   316→                      userSelect: 'none',                      
   317→                      position: 'relative',
   318→                      overflow: 'hidden',
   319→                    }}
   320→                    onMouseDown={() => !isPreview && onCellMouseDown(r, c)}
   321→                    onMouseEnter={() => !isPreview && onCellMouseEnter(r, c)}
   322→                    title={cell.dataBinding ? `[${cell.dataBinding}]` : undefined}
   323→                    onDoubleClick={(e) => {
   324→                      if (isPreview && cell.dataBinding && onCellEdit) {
   325→                        e.stopPropagation()
   326→                        const target = e.currentTarget.querySelector('[data-editable]') as HTMLElement | null
   327→                        if (target) {
   328→                          target.setAttribute('contentEditable', 'true')
   329→                          target.focus()
   330→                          // Select all text
   331→                          const range = document.createRange()
   332→                          range.selectNodeContents(target)
   333→                          const sel = window.getSelection()
   334→                          sel?.removeAllRanges()
   335→                          sel?.addRange(range)
   336→                        }
   337→                      }
   338→                    }}
   339→                  >
   340→                    {!isPreview && cell.dataBinding && (
   341→                      <span style={{
   342→                        position: 'absolute', top: '1px', right: '2px',
   343→                        width: '5px', height: '5px', borderRadius: '50%',
   344→                        background: '#f97316', display: 'inline-block',
   345→                        pointerEvents: 'none',
   346→                      }} />
   347→                    )}
   348→                    {isPreview && cell.dataBinding && !displayContent ? (
   349→                      <span style={{ color: '#ccc' }}>—</span>
   350→                    ) : isPreview && cell.dataBinding && onCellEdit ? (
   351→                      <AutoFitCell fontSize={cell.fontSize} fontWeight={cell.fontWeight} autoFit={!!cell.autoFit}>
   352→                      <span
   353→                        data-editable
   354→                        style={{ cursor: 'text', outline: 'none', minWidth: '20px', display: 'inline-block', minHeight: '1em' }}
   355→                        onBlur={(e) => {
   356→                          e.currentTarget.setAttribute('contentEditable', 'false')
   357→                          const newVal = e.currentTarget.textContent || ''
   358→                          if (newVal !== displayContent) {
   359→                            onCellEdit(cell.dataBinding, newVal)
   360→                          }
   361→                        }}
   362→                        onKeyDown={(e) => {
   363→                          if (e.key === 'Enter' && !e.shiftKey) {
   364→                            e.preventDefault()
   365→                            e.currentTarget.blur()
   366→                          }
   367→                          if (e.key === 'Escape') {
   368→                            e.currentTarget.blur()
   369→                          }
   370→                        }}
   371→                        suppressContentEditableWarning
   372→                      >
   373→                        {displayContent}
   374→                      </span>
   375→                      </AutoFitCell>
   376→                    ) : displayContent.startsWith('##LOGO_') && displayContent.endsWith('##') ? (
   377→                      <div style={{ width: '100%', height: '100%', display: 'flex', alignItems: 'center', justifyContent: 'center', overflow: 'hidden' }}>
   378→                        <img
   379→                          src="/logo-gob-mppe.png"
   380→                          alt="Logo Gobierno Bolivariano"
   381→                          style={{ maxWidth: '100%', maxHeight: '100%', width: '100%', height: 'auto', objectFit: 'contain', display: 'block' }}
   382→                        />
   383→                      </div>
   384→                    ) : (
   385→                      <AutoFitCell fontSize={cell.fontSize} fontWeight={cell.fontWeight} autoFit={!!cell.autoFit}>
   386→                        {displayContent}
   387→                      </AutoFitCell>
   388→                    )}
   389→                  </td>
   390→                )
   391→              }
   392→              return <tr key={r}>{cells}</tr>
   393→            })}
   394→          </tbody>
   395→        </table>
   396→      </div>
   397→    </div>
   398→  )
   399→}

   401→// === Column Width Editor ===
   402→function ColumnWidthEditor({
   403→  config,
   404→  onUpdate,
   405→}: {
   406→  config: GridConfig
   407→  onUpdate: (config: GridConfig) => void
   408→}) {
   409→  const [editing, setEditing] = useState<number | null>(null)
   410→  const [value, setValue] = useState('')
   411→
   412→  const startEdit = (idx: number) => {
   413→    setEditing(idx)
   414→    setValue(config.columnWidths[idx] || '')
   415→  }
   416→
   417→  const commitEdit = () => {
   418→    if (editing !== null && value.trim()) {
   419→      const newWidths = [...config.columnWidths]
   420→      newWidths[editing] = value.trim()
   421→      onUpdate({ ...config, columnWidths: newWidths })
   422→    }
   423→    setEditing(null)
   424→  }
   425→
   426→  return (
   427→    <div className="flex gap-0.5 overflow-x-auto p-1" style={{ flexWrap: 'nowrap' }}>
   428→      {config.columnWidths.map((w, i) => (
   429→        <div
   430→          key={i}
   431→          className="text-center border rounded px-0.5 py-0.5 min-w-[44px]"
   432→          style={{ fontSize: '7pt', backgroundColor: '#f8f8f8' }}
   433→        >
   434→          <div className="text-muted-foreground" style={{ fontSize: '6pt' }}>C{i}</div>
   435→          {editing === i ? (
   436→            <input
   437→              autoFocus
   438→              value={value}
   439→              onChange={(e) => setValue(e.target.value)}
   440→              onBlur={commitEdit}
   441→              onKeyDown={(e) => { if (e.key === 'Enter') commitEdit() }}
   442→              className="w-10 h-4 text-center text-[7pt] border rounded"
   443→              style={{ padding: '0 1px' }}
   444→            />
   445→          ) : (
   446→            <div
   447→              onClick={() => startEdit(i)}
   448→              className="cursor-pointer hover:bg-accent rounded"
   449→              title="Clic para editar"
   450→            >
   451→              {w}
   452→            </div>
   453→          )}
   454→        </div>
   455→      ))}
   456→    </div>
   457→  )
   458→}

   794→  function applyArrayOverrides(
   795→    arr: Record<string, any>[] | undefined, ov: Record<string, string>, prefix: string, fields: string[]
   796→  ): Record<string, any>[] {
   797→    if (!arr) return []
   798→    return arr.map((item, idx) => {
   799→      const patched = { ...item }
   800→      for (const field of fields) {
   801→        const key = `${prefix}.${idx}.${field}`
   802→        const val = ov[key]
   803→        if (val !== undefined && val !== null && val !== '') {
   804→          patched[field] = val
   805→        }
   806→      }
   807→      return patched
   808→    })

   811→  // Helper: build instituciones array from rawDataMap flat keys (plan derogado)
   812→  // rawDataMap has INST.BASICA.1-5, LOCAL.BASICA.1-5, EF.BASICA.1-5
   813→  // and INST.DIV.1-5, LOCAL.DIV.1-5, EF.DIV.1-5
   814→  function buildInstitucionesFromRaw(rm: Record<string, string>): { numero: number; denominacion: string; localidad: string; ef: string }[] {
   815→    const result: { numero: number; denominacion: string; localidad: string; ef: string }[] = []
   816→    // Collect all institution entries (BASICA first, then DIV)
   817→    const allEntries: { denominacion: string; localidad: string; ef: string }[] = []
   818→    for (let i = 1; i <= 5; i++) {
   819→      allEntries.push({
   820→        denominacion: rm[`INST.BASICA.${i}`] || '',
   821→        localidad: rm[`LOCAL.BASICA.${i}`] || '',
   822→        ef: rm[`EF.BASICA.${i}`] || '',
   823→      })
   824→    }
   825→    for (let i = 1; i <= 5; i++) {
   826→      const denom = rm[`INST.DIV.${i}`] || ''
   827→      const loc = rm[`LOCAL.DIV.${i}`] || ''
   828→      const ef = rm[`EF.DIV.${i}`] || ''
   829→      // Only add DIV entries if they have data and aren't duplicates of BASICA
   830→      if (denom && !allEntries.some(e => e.denominacion === denom)) {
   831→        allEntries.push({ denominacion: denom, localidad: loc, ef })
   832→      }
   833→    }
   834→    // Fallback: check top-level flat keys (INSTITUCION1-5, LOCALIDAD1-5, EF1-5)
   835→    if (allEntries.every(e => !e.denominacion)) {
   836→      for (let i = 1; i <= 5; i++) {
   837→        allEntries.push({
   838→          denominacion: rm[`INSTITUCION${i}`] || '',
   839→          localidad: rm[`LOCALIDAD${i}`] || '',
   840→          ef: rm[`EF${i}`] || '',
   841→        })
   842→      }
   843→    }
   844→    // Build final result with numbering, skip empty entries
   845→    let num = 1
   846→    for (const entry of allEntries) {
   847→      if (entry.denominacion) {
   848→        result.push({ numero: num++, ...entry })
   849→      }
   850→      if (result.length >= 5) break
   851→    }
   852→    return result
   853→  }

   855→  // Helper: apply draft overrides to calificaciones (nested by year key)
   856→  function applyCalifOverrides(
   857→    cals: Record<string, any[]> | undefined, ov: Record<string, string>
   858→  ): Record<string, any[]> {
   859→    if (!cals) return {}
   860→    const YEAR_NAME_MAP: Record<string, string> = {
   861→      '1': 'Primer Año', '2': 'Segundo Año', '3': 'Tercer Año',
   862→      '4': 'Cuarto Año', '5': 'Quinto Año',
   863→    }
   864→    const CALIF_FIELD_MAP: Record<string, string> = {
   865→      materia: 'materia', numero: 'numero', nota: 'nota', literal: 'literal',
   866→      te: 'tipoEvaluacion', mes: 'fechaMes', anio: 'fechaAnio', inst: 'instEduc',
   867→    }
   868→    const result: Record<string, any[]> = {}
   869→    for (const [yearKey, subjects] of Object.entries(cals)) {
   870→      result[yearKey] = subjects.map((subj: any, sIdx: number) => {
   871→        const patched = { ...subj }
   872→        // Check both numeric key (e.g., "calif.1.0.nota") and year-name key
   873→        for (let yNum = 1; yNum <= 5; yNum++) {
   874→          if (YEAR_NAME_MAP[String(yNum)] !== yearKey) continue
   875→          for (const [shortField, realField] of Object.entries(CALIF_FIELD_MAP)) {
   876→            const key = `calif.${yNum}.${sIdx}.${shortField}`
   877→            const val = ov[key]
   878→            if (val !== undefined && val !== null && val !== '') {
   879→              patched[realField] = val
   880→            }
   881→          }
   882→        }
   883→        return patched
   884→      })
   885→    }
   886→    return result
   887→  }
